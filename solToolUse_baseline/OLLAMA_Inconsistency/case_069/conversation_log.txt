```python
# --------------------------------------------------------------
#  extract_c3d20r_connectivity_from_inp.py
# --------------------------------------------------------------
import sys
import re

def parse_inp(inp_path):
    elem_pattern = re.compile(r'^\s*\*Element,\s*type=C3D20R', re.IGNORECASE)
    data_pattern = re.compile(r'^\s*(\d+)\s*,\s*(.+)$')
    connectivity = []

    with open(inp_path, 'r') as f:
        lines = f.readlines()

    i = 0
    while i < len(lines):
        line = lines[i].strip()
        # look for the start of a C3D20R element block
        if elem_pattern.match(line):
            i += 1                     # skip the *Element line
            # read until a line that starts with *
            while i < len(lines) and not lines[i].lstrip().startswith('*'):
                # join continuation lines (Abaqus allows a back‑slash at line end)
                raw = lines[i].strip()
                while raw.endswith('\\'):
                    raw = raw[:-1] + lines[i+1].strip()
                    i += 1
                # split the element line
                m = data_pattern.match(raw)
                if m:
                    elem_id = int(m.group(1))
                    nodes = [int(v) for v in m.group(2).split(',')]
                    if len(nodes) != 20:
                        raise ValueError(f'Element {elem_id} does not have 20 nodes (found {len(nodes)})')
                    connectivity.append((elem_id, nodes))
                i += 1
        else:
            i += 1

    return connectivity


def main():
    if len(sys.argv) != 2:
        print('Usage: python extract_c3d20r_connectivity_from_inp.py <model.inp>')
        sys.exit(1)

    inp_path = sys.argv[1]
    conn = parse_inp(inp_path)

    # sort by element ID (optional, but makes the output tidy)
    conn.sort(key=lambda x: x[0])

    for elem_id, nodes in conn:
```
