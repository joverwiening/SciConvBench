## 4️⃣ Python converter (`convert_to_json.py`)

The script reads the **ASCII `.dat` file** produced by CalculiX (node displacements, reaction forces, and element stresses) and writes a single `results.json` file with a clean, hierarchical structure.

```python
#!/usr/bin/env python3
"""
convert_to_json.py
------------------
Parse a CalculiX *.dat file and write a JSON file with:
  - nodal_displacements: {node_id: {"UX":..., "UY":..., "UZ":...}}
  - nodal_reactions:    {node_id: {"FX":..., "FY":..., "FZ":...}}
  - element_stresses:   {elem_id: {"SXX":..., "SYY":..., "SZZ":..., "SXY":..., "SYZ":..., "SXZ":...}}
"""

import sys
import json
import re
from pathlib import Path

def parse_dat(dat_path: Path):
    """Extract the three blocks we care about."""
    nodal_disp = {}
    nodal_reac = {}
    elem_stress = {}

    # Regular expressions for the three sections
    disp_pat   = re.compile(r"^\s*DISPLACEMENT\s+NODAL\s+COORDINATES\s*$", re.I)
    reac_pat   = re.compile(r"^\s*REACTION\s+FORCES\s*$", re.I)
    stress_pat = re.compile(r"^\s*STRESS\s+IN\s+ELEMENTS\s*$", re.I)

    # Helper to read a block of numbers until a blank line or a line that starts with a non‑numeric char
    def read_block(it):
        block = []
        for line in it:
            if not line.strip():
                break
            # Stop when a line starts with a letter (new section header)
            if re.match(r"^[A-Za-z]", line):
                # push the line back for the outer loop to see
                it.push_back(line)
                break
            block.append(line)
        return block

    # Simple iterator that can push a line back (needed for block detection)
    class PushBackIter:
        def __init__(self, lines):
            self._lines = iter(lines)
            self._buffer = None
        def __iter__(self):
            return self
        def __next__(self):
            if self._buffer is not None:
                line = self._buffer
                self._buffer = None
                return line
            return next(self._lines)
        def push_back(self, line):
            self._buffer = line

    with dat_path.open() as f:
        it = PushBackIter(f.readlines())

        for line in it:
            # -------------------------------------------------
            # 1) NODAL DISPLACEMENTS
            # -------------------------------------------------
            if disp_pat.search(line):
                # Skip the header line that follows (node, ux, uy, uz)
                header = next(it)
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 4:
                        continue
                    nid = int(parts[0])
                    ux, uy, uz = map(float, parts[1:4])
                    nodal_disp[nid] = {"UX": ux, "UY": uy, "UZ": uz}
                continue

            # -------------------------------------------------
            # 2) REACTION FORCES
            # -------------------------------------------------
            if reac_pat.search(line):
                header = next(it)   # usually: node, fx, fy, fz
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 4:
                        continue
                    nid = int(parts[0])
                    fx, fy, fz = map(float, parts[1:4])
                    nodal_reac[nid] = {"FX": fx, "FY": fy, "FZ": fz}
                continue

            # -------------------------------------------------
            # 3) ELEMENT STRESSES (first integration point only)
            # -------------------------------------------------
            if stress_pat.search(line):
                header = next(it)   # usually: elem, sxx, syy, szz, sxy, syz, sxz
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 7:
                        continue
                    eid = int(parts[0])
                    sxx, syy, szz, sxy, syz, sxz = map(float, parts[1:7])
                    elem_stress[eid] = {
                        "SXX": sxx, "SYY": syy, "SZZ": szz,
                        "SXY": sxy, "SYZ": syz, "SXZ": sxz
                    }
                continue

    return {
        "nodal_displacements": nodal_disp,
        "nodal_reactions": nodal_reac,
        "element_stresses": elem_stress,
    }

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
        print(f"Error: {dat_file} does not exist.")
        sys.exit(1)

    results = parse_dat(dat_file)

    json_path = dat_file.with_name("results.json")
    json_path.write_text(json.dumps(results, indent=2))
    print(f"✅ JSON written to {json_path}")

if __name__ == "__main__":
    main()
```

**What the JSON looks like**

```json
  "nodal_displacements": {
    "1": { "UX": 0.0, "UY": 0.0, "UZ": 0.0 },
    "2": { "UX": 0.0, "UY": -0.00123, "UZ": 0.0 },
    "...": {}
  },
  "nodal_reactions": {
    "1": { "FX": 0.0, "FY": -1000.0, "FZ": 0.0 }
  },
  "element_stresses": {
    "1": { "SXX": 12.3, "SYY": -4.5, "SZZ": 0.0, "SXY": 0.0, "SYZ": 0.0, "SXZ": 0.0 },

  }
```

## 4️⃣ `convert_to_json.py` – parse `.dat` → `results.json`

```python
#!/usr/bin/env python3
"""
convert_to_json.py
Parse a CalculiX *.dat file and write a JSON file with:
  - nodal_displacements: {node_id: {"UX":..., "UY":..., "UZ":...}}
  - nodal_reactions:    {node_id: {"FX":..., "FY":..., "FZ":...}}
  - element_stresses:   {elem_id: {"SXX":..., "SYY":..., "SZZ":..., "SXY":..., "SYZ":..., "SXZ":...}}
"""

import sys
import json
import re
from pathlib import Path

def parse_dat(dat_path: Path):
    nodal_disp = {}
    nodal_reac = {}
    elem_stress = {}

    # -------------------------------------------------
    # Regular expressions for the three sections we care about
    # -------------------------------------------------
    disp_pat   = re.compile(r"^\s*DISPLACEMENT\s+NODAL\s+COORDINATES\s*$", re.I)
    reac_pat   = re.compile(r"^\s*REACTION\s+FORCES\s*$", re.I)
    stress_pat = re.compile(r"^\s*STRESS\s+IN\s+ELEMENTS\s*$", re.I)

    # Helper to read a block of numbers until a blank line or a new header
    def read_block(it):
        block = []
        for line in it:
            if not line.strip():
                break
            if re.match(r"^[A-Za-z]", line):
                it.push_back(line)
                break
            block.append(line)
        return block

    # Simple iterator that can push a line back (needed for block detection)
    class PushBackIter:
        def __init__(self, lines):
            self._lines = iter(lines)
            self._buffer = None
        def __iter__(self):
            return self
        def __next__(self):
            if self._buffer is not None:
                line = self._buffer
                self._buffer = None
                return line
            return next(self._lines)
        def push_back(self, line):
            self._buffer = line

    with dat_path.open() as f:
        it = PushBackIter(f.readlines())

        for line in it:
            # -------------------------------------------------
            # 1) NODAL DISPLACEMENTS
            # -------------------------------------------------
            if disp_pat.search(line):
                header = next(it)   # usually: node, ux, uy, uz
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 4:
                        continue
                    nid = int(parts[0])
                    ux, uy, uz = map(float, parts[1:4])
                    nodal_disp[nid] = {"UX": ux, "UY": uy, "UZ": uz}
                continue

            # -------------------------------------------------
            # 2) REACTION FORCES
            # -------------------------------------------------
            if reac_pat.search(line):
                header = next(it)   # usually: node, fx, fy, fz
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 4:
                        continue
                    nid = int(parts[0])
                    fx, fy, fz = map(float, parts[1:4])
                    nodal_reac[nid] = {"FX": fx, "FY": fy, "FZ": fz}
                continue

            # -------------------------------------------------
            # 3) ELEMENT STRESSES (first integration point only)
            # -------------------------------------------------
            if stress_pat.search(line):
                header = next(it)   # usually: elem, sxx, syy, szz, sxy, syz, sxz
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 7:
                        continue
                    eid = int(parts[0])
                    sxx, syy, szz, sxy, syz, sxz = map(float, parts[1:7])
                    elem_stress[eid] = {
                        "SXX": sxx, "SYY": syy, "SZZ": szz,
                        "SXY": sxy, "SYZ": syz, "SXZ": sxz,
                    }
                continue

    return {
        "nodal_displacements": nodal_disp,
        "nodal_reactions":    nodal_reac,
        "element_stresses":   elem_stress,
    }

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
        print(f"Error: {dat_file} does not exist.")
        sys.exit(1)

    results = parse_dat(dat_file)

    json_path = dat_file.with_name("results.json")
    json_path.write_text(json.dumps(results, indent=2))
    print(f"✅ JSON written to {json_path}")

if __name__ == "__main__":
    main()
```

```python
    # Helper to read a block of numbers until a blank line or a new header
    def read_block(it):
        block = []
        for line in it:
            if not line.strip():
                break
            if re.match(r"^[A-Za-z]", line):
                it.push_back(line)
                break
            block.append(line)
        return block

    # Iterator that can push a line back (needed for block detection)
    class PushBackIter:
        def __init__(self, lines):
            self._lines = iter(lines)
            self._buffer = None
        def __iter__(self):
            return self
        def __next__(self):
            if self._buffer is not None:
                line = self._buffer
                self._buffer = None
                return line
            return next(self._lines)
        def push_back(self, line):
            self._buffer = line

    with dat_path.open() as f:
        it = PushBackIter(f.readlines())

        for line in it:
            # -------------------------------------------------
            # 1) NODAL DISPLACEMENTS
            # -------------------------------------------------
            if disp_pat.search(line):
                next(it)  # skip header line (node, ux, uy, uz)
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 4:
                        continue
                    nid = int(parts[0])
                    ux, uy, uz = map(float, parts[1:4])
                    nodal_disp[nid] = {"UX": ux, "UY": uy, "UZ": uz}
                continue

            # -------------------------------------------------
            # 2) REACTION FORCES
            # -------------------------------------------------
            if reac_pat.search(line):
                next(it)  # skip header line (node, fx, fy, fz)
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 4:
                        continue
                    nid = int(parts[0])
                    fx, fy, fz = map(float, parts[1:4])
                    nodal_reac[nid] = {"FX": fx, "FY": fy, "FZ": fz}
                continue

            # -------------------------------------------------
            # 3) ELEMENT STRESSES (first integration point only)
            # -------------------------------------------------
            if stress_pat.search(line):
                next(it)  # skip header line (elem, sxx, syy, szz, sxy, syz, sxz)
                for blk_line in read_block(it):
                    parts = blk_line.split()
                    if len(parts) < 7:
                        continue
                    eid = int(parts[0])
                    sxx, syy, szz, sxy, syz, sxz = map(float, parts[1:7])
                    elem_stress[eid] = {
                        "SXX": sxx, "SYY": syy, "SZZ": szz,
                        "SXY": sxy, "SYZ": syz, "SXZ": sxz,
                    }
                continue

    return {
        "nodal_displacements": nodal_disp,
        "nodal_reactions":    nodal_reac,
        "element_stresses":   elem_stress,
    }

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
        print(f"Error: {dat_file} does not exist.")
        sys.exit(1)

    results = parse_dat(dat_file)

    json_path = dat_file.with_name("results.json")
    json_path.write_text(json.dumps(results, indent=2))
    print(f"✅ JSON written to {json_path}")

if __name__ == "__main__":
    main()
```

### `convert_to_json.py`  (Python 3.8+)

```python
#!/usr/bin/env python3
"""
convert_to_json.py – Parse a CalculiX *.dat file and write a tidy JSON file.

Usage:
    python convert_to_json.py <input.dat>   # writes results.json in the same folder
"""

import json
import re
import sys
from pathlib import Path

# ----------------------------------------------------------------------
# Helper regexes – adjust if your .dat headings differ
# ----------------------------------------------------------------------
RE_DISP = re.compile(r"^\s*DISPLACEMENT\s+NODAL\s+COORDINATES\s*$", re.I)
RE_STRESS = re.compile(r"^\s*STRESS\s+ELEMENT\s+NODES\s*$", re.I)
RE_REACT = re.compile(r"^\s*REACTION\s+FORCES\s*$", re.I)

def parse_section(lines, start_idx):
    """Read a numeric table until a blank line or a line that does not start with a digit."""
    data = []
    i = start_idx
    while i < len(lines):
        line = lines[i].strip()
        if not line or not line[0].isdigit():
            break
        # Split on whitespace, keep sign attached to numbers
        row = [float(val) for val in line.split()]
        data.append(row)
        i += 1
    return data, i

def read_dat(dat_path: Path):
    """Return a dict with parsed sections."""
    with dat_path.open("r", encoding="utf-8", errors="replace") as f:
        lines = f.readlines()

    out = {}
    i = 0
    while i < len(lines):
        line = lines[i].strip()
        if RE_DISP.match(line):
            i += 1  # skip header line(s) that follow the title
            # The displacement table usually starts with a line "NODE   U1   U2   U3"
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            disp, i = parse_section(lines, i)
            out["displacements"] = disp
        elif RE_STRESS.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            stress, i = parse_section(lines, i)
            out["stresses"] = stress
        elif RE_REACT.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            react, i = parse_section(lines, i)
            out["reactions"] = react
        else:
            i += 1
    return out

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
        print(f"❌  File not found: {dat_file}")
        sys.exit(1)

    print(f"🔎  Parsing {dat_file.name} …")
    data = read_dat(dat_file)

    json_path = dat_file.with_name("results.json")
    with json_path.open("w", encoding="utf-8") as jf:
        json.dump(data, jf, indent=2, ensure_ascii=False)

    print(f"✅  JSON written to {json_path}")

if __name__ == "__main__":
    main()
```

```python
"""
convert_to_json.py – Parse a CalculiX *.dat file and write a tidy JSON file.

Usage:
    python convert_to_json.py <input.dat>   # writes results.json in the same folder
"""

import json
import re
import sys
from pathlib import Path

# ------------------------------------------
# Helper regexes – adjust if your .dat headings differ
# ------------------------------------------
RE_DISP   = re.compile(r"^\s*DISPLACEMENT\s+NODAL\s+COORDINATES\s*$", re.I)
RE_STRESS = re.compile(r"^\s*STRESS\s+ELEMENT\s+NODES\s*$", re.I)
RE_REACT  = re.compile(r"^\s*REACTION\s+FORCES\s*$", re.I)

def parse_section(lines, start_idx):
    """Read a numeric table until a blank line or a line that does not start with a digit."""
    data = []
    i = start_idx
    while i < len(lines):
        line = lines[i].strip()
        if not line or not line[0].isdigit():
            break
        row = [float(val) for val in line.split()]
        data.append(row)
        i += 1
    return data, i

def read_dat(dat_path: Path):
    """Return a dict with parsed sections."""
    with dat_path.open("r", encoding="utf-8", errors="replace") as f:
        lines = f.readlines()

    out = {}
    i = 0
    while i < len(lines):
        line = lines[i].strip()
        if RE_DISP.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            disp, i = parse_section(lines, i)
            out["displacements"] = disp

                i += 1
            react, i = parse_section(lines, i)
            out["reactions"] = react
        else:
            i += 1
    return out

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
```

### `convert_to_json.py`  (Python 3.8+)

```python
#!/usr/bin/env python3
"""
convert_to_json.py – Parse a CalculiX *.dat file and write a tidy JSON file.

Usage:
    python convert_to_json.py <input.dat>   # writes results.json in the same folder
"""

import json
import re
import sys
from pathlib import Path
```

```python
# ----------------------------------
# Helper regexes – adjust if your .dat headings differ
# ----------------------------------
RE_DISP   = re.compile(r"^\s*DISPLACEMENT\s+NODAL\s+COORDINATES\s*$", re.I)
RE_STRESS = re.compile(r"^\s*STRESS\s+ELEMENT\s+NODES\s*$", re.I)
RE_REACT  = re.compile(r"^\s*REACTION\s+FORCES\s*$", re.I)

def parse_section(lines, start_idx):
    """Read a numeric table until a blank line or a line that does not start with a digit."""
    data = []
    i = start_idx
    while i < len(lines):
        line = lines[i].strip()
        if not line or not line[0].isdigit():
            break
        row = [float(val) for val in line.split()]
        data.append(row)
        i += 1
    return data, i

def read_dat(dat_path: Path):
    """Return a dict with parsed sections."""
    with dat_path.open("r", encoding="utf-8", errors="replace") as f:
        lines = f.readlines()

    out = {}
    i = 0
    while i < len(lines):
        line = lines[i].strip()
        if RE_DISP.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            disp, i = parse_section(lines, i)
            out["displacements"] = disp
        elif RE_STRESS.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            stress, i = parse_section(lines, i)
            out["stresses"] = stress
        elif RE_REACT.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            react, i = parse_section(lines, i)
            out["reactions"] = react
        else:
            i += 1
    return out

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
        print(f"❌  File not found: {dat_file}")
        sys.exit(1)

    print(f"🔎  Parsing {dat_file.name} …")
    data = read_dat(dat_file)

    json_path = dat_file.with_name("results.json")
    with json_path.open("w", encoding="utf-8") as jf:
        json.dump(data, jf, indent=2, ensure_ascii=False)

    print(f"✅  JSON written to {json_path}")

if __name__ == "__main__":
    main()
```

```bat
    echo ▶️  Converting model.dat → results.json ...
    python convert_to_json.py model.dat
    if errorlevel 1 (
        echo ❌  Python script failed.
        exit /b 1
    )
    echo ✅  Done! results.json is ready.
) else (
    echo ❌  model.dat not found – something went wrong with the solver.
    exit /b 1
)

endlocal
```  

### `convert_to_json.py`  (Python 3.8+)

```python
#!/usr/bin/env python3
"""
convert_to_json.py – Parse a CalculiX *.dat file and write a tidy JSON file.

Usage:
    python convert_to_json.py <input.dat>   # writes results.json in the same folder
"""

import json
import re
import sys
from pathlib import Path

# -------------------------------------------------
# Helper regexes – adjust if your .dat headings differ
# -------------------------------------------------
RE_DISP   = re.compile(r"^\s*DISPLACEMENT\s+NODAL\s+COORDINATES\s*$", re.I)
RE_STRESS = re.compile(r"^\s*STRESS\s+ELEMENT\s+NODES\s*$", re.I)
RE_REACT  = re.compile(r"^\s*REACTION\s+FORCES\s*$", re.I)

def parse_section(lines, start_idx):
    """Read a numeric table until a blank line or a line that does not start with a digit."""
    data = []
    i = start_idx
    while i < len(lines):
        line = lines[i].strip()
        if not line or not line[0].isdigit():
            break
        row = [float(val) for val in line.split()]
        data.append(row)
        i += 1
    return data, i

def read_dat(dat_path: Path):
    """Return a dict with parsed sections."""
    with dat_path.open("r", encoding="utf-8", errors="replace") as f:
        lines = f.readlines()

    out = {}
    i = 0
    while i < len(lines):
        line = lines[i].strip()
        if RE_DISP.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            disp, i = parse_section(lines, i)
            out["displacements"] = disp
        elif RE_STRESS.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            stress, i = parse_section(lines, i)
            out["stresses"] = stress
        elif RE_REACT.match(line):
            i += 1
            while i < len(lines) and not lines[i].strip().startswith("1"):
                i += 1
            react, i = parse_section(lines, i)
            out["reactions"] = react
        else:
            i += 1
    return out

def main():
    if len(sys.argv) != 2:
        print("Usage: python convert_to_json.py <model.dat>")
        sys.exit(1)

    dat_file = Path(sys.argv[1])
    if not dat_file.is_file():
        print(f"❌  File not found: {dat_file}")
        sys.exit(1)

    print(f"🔎  Parsing {dat_file.name} …")
    data = read_dat(dat_file)

    json_path = dat_file.with_name("results.json")
    with json_path.open("w", encoding="utf-8") as jf:
        json.dump(data, jf, indent=2, ensure_ascii=False)

    print(f"✅  JSON written to {json_path}")

if __name__ == "__main__":
    main()
```  

> **Tip:** If your `.dat` headings differ (e.g., “NODAL DISPLACEMENTS” instead of “DISPLACEMENT NODAL COORDINATES”), edit the three regular‑expression patterns at the top of `convert_to_json.py`.

## ❗ Quick‑Fix Checklist  

| Symptom | Likely cause | How to fix |
|---------|--------------|------------|
| `ccx: command not found` | CalculiX not installed or not on `PATH` | Install CalculiX (https://www.calculix.de) and add its `bin` directory to your system `PATH`. |
| Solver aborts with “*ERROR*” in `model.dat` | Invalid input (missing material, wrong element type, etc.) | Open `model.inp`, locate the line mentioned in the error, and correct the offending section. |
| `results.json` is empty / missing fields | Parser didn’t recognise the headings in `model.dat` | Open `model.dat`, verify the exact headings (e.g. “DISPLACEMENT NODAL COORDINATES”). If they differ, edit the regexes in `convert_to_json.py`. |
| Python crashes with `UnicodeDecodeError` | `.dat` contains non‑ASCII characters | Re‑save the `.dat` file as UTF‑8 (or run `python -X utf8 convert_to_json.py model.dat`). |
| Permission denied when running `run_calculix.sh` | Script not executable | Re‑run `chmod +x run_calculix.sh` and try again. |
| `model.dat` never appears | `ccx` failed silently (often due to missing material definitions) | Run `ccx model` manually in a terminal to see the full console output. |
