## How to extract it programmatically (Python example)

If you need to read the input file and obtain the list of nodes that belong to a `GENERATE`‑type NSET, you can do it with a few lines of Python.  
The code below works for the example you posted, but it also handles the general case where a `GENERATE` line may contain **start, end, increment** (or several ranges separated by new lines).

```python
import re
from pathlib import Path

def parse_generate_nset(file_path, set_name):
    """
    Reads an Abaqus .inp file and returns the list of node (or element) IDs
    that belong to a *NSET with the option GENERATE.

    Parameters
    ----------
    file_path : str or pathlib.Path
        Path to the .inp file.
    set_name : str
        The name of the NSET you want to extract (e.g. "Nall").

    Returns
    -------
    List[int]
        Sorted list of IDs belonging to the requested set.
    """
    ids = []
    # Regular expression that matches the *NSET line with GENERATE
    #   *NSET,NSET=SetName,GENERATE
    pattern = re.compile(rf'^\*NSET\s*,\s*NSET\s*=\s*{re.escape(set_name)}\s*,\s*GENERATE\s*$', re.IGNORECASE)

    with Path(file_path).open('r') as f:
        lines = f.readlines()

    i = 0
    while i < len(lines):
        line = lines[i].strip()
        # Look for the *NSET line that defines the set we want
        if pattern.match(line):
            i += 1                     # move to the first data line
            while i < len(lines):
                data_line = lines[i].strip()
                # Stop when we hit a new keyword (starts with '*')
                if data_line.startswith('*') or not data_line:
                    break
                # A data line may contain several comma‑separated ranges
                for token in data_line.split(','):
                    token = token.strip()
                    if not token:
                        continue
                    # Each token can be:
                    #   start,end               -> increment = 1
                    #   start,end,inc
                    #   start,end,inc,repeat   (repeat is ignored for NSET)
                    parts = [int(p) for p in token.split() if p]
                    if len(parts) == 2:               # start, end
                        start, end = parts
                        inc = 1
                    elif len(parts) >= 3:              # start, end, inc
                        start, end, inc = parts[:3]
                    else:
                        raise ValueError(f'Unexpected token "{token}"')
                    # Abaqus allows start > end when inc is negative
                    if inc == 0:
                        raise ValueError('Increment cannot be zero')
                    ids.extend(range(start, end + (1 if inc > 0 else -1), inc))
                i += 1
            break
        i += 1

    return sorted(set(ids))


# ----------------------------------------------------------------------

    print('First 10 nodes:', node_set[:10])
    print('Last 10 nodes :', node_set[-10:])
```
