## How to extract the element numbers automatically (Python)

If you have many such surfaces or you want a reusable tool, the short script below reads an Abaqus *.inp* file, finds the element set that is used for a given surface, and returns the list of element IDs.

```python
#!/usr/bin/env python3
"""
Extract the element numbers that belong to a given Abaqus surface.
"""

import re
from pathlib import Path

def _parse_generate(line: str):
    """Parse a line of the form  start, end, step  (all integers)."""
    start, end, step = (int(v) for v in line.split(','))
    return list(range(start, end + 1, step))

def _read_elset(lines, i):
    """
    Return the element IDs defined by an *Elset* block that starts at line i.
    The block can be either:
      * generate  (single line with start, end, step)
      * explicit   (one or more lines with comma‑separated numbers)
    """
    header = lines[i].strip().lower()
    generate = 'generate' in header

    # The element numbers start on the next line(s)
    elems = []
    j = i + 1
    while j < len(lines) and not lines[j].lstrip().startswith('*'):
        # remove comments and blank lines
        line = lines[j].split('!')[0].strip()
        if not line:
            j += 1
            continue

        if generate:
            # only one line is expected for a generated set
            elems = _parse_generate(line)
            break
        else:
            # explicit list – may span several lines
            elems.extend(int(v) for v in line.split(',') if v.strip())
        j += 1
    return elems

def extract_surface_elements(inp_path: Path, surface_name: str):
    """
    Return a list of element IDs that belong to the surface `surface_name`.
    """
    lines = inp_path.read_text().splitlines()

    # 1️⃣ Find the surface definition
    surf_pat = re.compile(rf'^\*Surface.*\bname\s*=\s*{re.escape(surface_name)}\b', re.IGNORECASE)
    surf_line_idx = None
    for i, line in enumerate(lines):
        if surf_pat.search(line):
            surf_line_idx = i
            break
    if surf_line_idx is None:
        raise ValueError(f'Surface "{surface_name}" not found in {inp_path}')

    # 2️⃣ The line after *Surface is the element‑set reference, e.g.
    #    _PickedSurf4_S2, S2
    set_ref_line = lines[surf_line_idx + 1].strip()
    elset_name = set_ref_line.split(',')[0].strip()

    # 3️⃣ Locate the *Elset block that defines this set
    elset_pat = re.compile(rf'^\*Elset\s*,\s*elset\s*=\s*{re.escape(elset_name)}', re.IGNORECASE)
    for i, line in enumerate(lines):
        if elset_pat.search(line):
            return _read_elset(lines, i)

    raise ValueError(f'Elset "{elset_name}" not found in {inp_path}')

# ----------------------------------------------------------------------
if __name__ == '__main__':
    # Example usage ----------------------------------------------------
    inp_file = Path('your_mesh_file.inp')          # <-- change to your file name
    surface = 'PickedSurf4'                        # <-- surface name as in *Surface
    elems = extract_surface_elements(inp_file, surface)
    print(f'Elements belonging to surface "{surface}":')
    print(elems)
    # ------------------------------------------------------------------
```

| 3️⃣   | `*Elset, elset=_PickedSurf4_S2` (or `*Elset, elset=_PickedSurf4_S2, generate`) | Reads the element IDs. If the header contains the word **generate**, the next line is interpreted as `start, end, step`; otherwise the IDs are read as an explicit list (possibly spanning several lines). |
