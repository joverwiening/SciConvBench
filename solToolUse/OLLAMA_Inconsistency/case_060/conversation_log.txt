```python
#!/usr/bin/env python3
import sys, re

def read_blocks(fname):
    with open(fname) as f:
        txt = f.read()
    # Displacements block starts with "DISPLACEMENT"
    disp = re.search(r'DISPLACEMENT\s+([\s\S]*?)\n\s*\n', txt)
    # Reaction forces block starts with "REACTION"
    reac = re.search(r'REACTION\s+([\s\S]*?)\n\s*\n', txt)
    # Element stress block starts with "ELEMENT STRESS"
    stress = re.search(r'ELEMENT STRESS\s+([\s\S]*?)\n\s*\n', txt)
    return disp, reac, stress

def parse_table(block):
    rows = []
    for line in block.strip().splitlines():
        if line.strip().startswith('**'):   # skip comment lines
            continue
        parts = [float(v) for v in line.split(',') if v.strip()]
        rows.append(parts)
    return rows

def main(datfile):
    disp, reac, stress = read_blocks(datfile)

    # ---- displacement of node 7 (y‑direction) ----
    if disp:
        drows = parse_table(disp.group(1))
        for r in drows:
            nid = int(r[0])
            if nid == 7:
                print(f'Node 7 displacement (U1,U2,U3) = {r[1]:.6e}, {r[2]:.6e}, {r[3]:.6e}')
                break

    # ---- reaction force at node 7 (y‑direction) ----
    if reac:
        rrows = parse_table(reac.group(1))
        for r in rrows:
            nid = int(r[0])
            if nid == 7:
                print(f'Node 7 reaction (RF1,RF2,RF3) = {r[1]:.6e}, {r[2]:.6e}, {r[3]:.6e}')
                break

    # ---- von Mises stress for each element (average of 8 Gauss points) ----
    if stress:
        srows = parse_table(stress.group(1))
        # columns: EL, S1, S2, …, S6 (six stress components) for each integration point
        # We'll compute von Mises per integration point and then average per element.
        from collections import defaultdict
        vm = defaultdict(list)
        for r in srows:
            el = int(r[0])
            s = r[1:7]   # S11,S22,S33,S12,S13,S23
            s11,s22,s33,s12,s13,s23 = s
            # von Mises formula (plane stress/3‑D)
            vm_i = ((s11 - s22)**2 + (s22 - s33)**2 + (s33 - s11)**2
                    + 6*(s12**2 + s13**2 + s23**2))**0.5 / (2**0.5)
            vm[el].append(vm_i)
        print('\nAverage von Mises stress per element:')
        for el, vals in sorted(vm.items()):
            print(f'  Element {el:3d}: {sum(vals)/len(vals):.2f} MPa')
    else:
        print('No stress block found.')

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print('Usage: python3 extract.py <job>.dat')
        sys.exit(1)
    main(sys.argv[1])
```

```python
#!/usr/bin/env python3
# --------------------------------------------------------------
# extract.py – pull out nodal displacements, reactions and
# average von Mises stress per element from a CalculiX .dat file
# --------------------------------------------------------------
import sys, re
from collections import defaultdict
import math

def read_blocks(fname):
    """Return the raw text of the three blocks we care about."""
    with open(fname) as f:
        txt = f.read()
    disp = re.search(r'DISPLACEMENT\s+([\s\S]*?)\n\s*\n', txt)
    reac = re.search(r'REACTION\s+([\s\S]*?)\n\s*\n', txt)
    stress = re.search(r'ELEMENT STRESS\s+([\s\S]*?)\n\s*\n', txt)
    return disp, reac, stress

def parse_table(block):
    """Convert a block of comma‑separated numbers into a list of rows."""
    rows = []
    for line in block.strip().splitlines():
        if line.strip().startswith('**'):   # skip comment lines
            continue
        # split on whitespace or commas, ignore empty fields
        parts = [float(v) for v in re.split(r'[,\s]+', line.strip()) if v]
        rows.append(parts)
    return rows

def von_mises(s):
    """Von Mises stress from the six stress components (3‑D)."""
    s11, s22, s33, s12, s13, s23 = s
    term1 = (s11 - s22) ** 2 + (s22 - s33) ** 2 + (s33 - s11) ** 2
    term2 = 6 * (s12 ** 2 + s13 ** 2 + s23 ** 2)
    return math.sqrt(0.5 * (term1 + term2))

def main(datfile):
    disp, reac, stress = read_blocks(datfile)

    # ---------- displacement of node 7 ----------
    if disp:
        for row in parse_table(disp.group(1)):
            nid = int(row[0])
            if nid == 7:
                print(f'Node 7 displacement (U1,U2,U3) = {row[1]:.6e}, {row[2]:.6e}, {row[3]:.6e}')
                break

    # ---------- reaction forces at node 7 ----------
    if reac:
        for row in parse_table(reac.group(1)):
            nid = int(row[0])
            if nid == 7:
                print(f'Node 7 reaction (RF1,RF2,RF3) = {row[1]:.6e}, {row[2]:.6e}, {row[3]:.6e}')
                break

    # ---------- von Mises per element (average over Gauss points) ----------
    if stress:
        vm_per_el = defaultdict(list)
        for row in parse_table(stress.group(1)):
            el = int(row[0])
            s = row[1:7]                     # six stress components
            vm_per_el[el].append(von_mises(s))

        print('\nAverage von Mises stress per element (MPa):')
        for el in sorted(vm_per_el):
            avg_vm = sum(vm_per_el[el]) / len(vm_per_el[el])
            print(f'  Element {el:4d}: {avg_vm:8.2f}')
    else:
        print('No ELEMENT STRESS block found.')

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print('Usage: python3 extract.py <job>.dat')
        sys.exit(1)
    main(sys.argv[1])
```

```python
#!/usr/bin/env python3
# --------------------------------------------------------------
# extract.py – pull out nodal displacements, reactions and
# average von Mises stress per element from a CalculiX .dat file
# --------------------------------------------------------------
import sys, re, math
from collections import defaultdict

def read_blocks(fname):
    """Return the raw text of the three blocks we care about."""
    with open(fname) as f:
        txt = f.read()
    disp   = re.search(r'DISPLACEMENT\s+([\s\S]*?)\n\s*\n', txt)
    reac   = re.search(r'REACTION\s+([\s\S]*?)\n\s*\n', txt)
    stress = re.search(r'ELEMENT STRESS\s+([\s\S]*?)\n\s*\n', txt)
    return disp, reac, stress

def parse_table(block):
    """Convert a block of numbers (comma/space separated) into rows."""
    rows = []
    for line in block.strip().splitlines():
        if line.strip().startswith('**'):   # skip comment lines
            continue
        parts = [float(v) for v in re.split(r'[,\s]+', line.strip()) if v]
        rows.append(parts)
    return rows

def von_mises(stress):
    """Von Mises stress from the six stress components (3‑D)."""
    s11, s22, s33, s12, s13, s23 = stress
    term1 = (s11 - s22) ** 2 + (s22 - s33) ** 2 + (s33 - s11) ** 2
    term2 = 6 * (s12 ** 2 + s13 ** 2 + s23 ** 2)
    return math.sqrt(0.5 * (term1 + term2))

def main(datfile):
    disp, reac, stress = read_blocks(datfile)

    # ---- displacement of node 7 ----
    if disp:
        for row in parse_table(disp.group(1)):
            nid = int(row[0])
            if nid == 7:
                print(f'Node 7 displacement (U1,U2,U3) = {row[1]:.6e}, {row[2]:.6e}, {row[3]:.6e}')
                break

    # ---- reaction forces at node 7 ----
    if reac:
        for row in parse_table(reac.group(1)):
            nid = int(row[0])
            if nid == 7:
                print(f'Node 7 reaction (RF1,RF2,RF3) = {row[1]:.6e}, {row[2]:.6e}, {row[3]:.6e}')
                break

    # ---- average von Mises stress per element (average over Gauss points) ----
    if stress:
        vm_per_el = defaultdict(list)
        for row in parse_table(stress.group(1)):
            el = int(row[0])
            s = row[1:7]                     # six stress components
            vm_per_el[el].append(von_mises(s))

        print('\nAverage von Mises stress per element (MPa):')
        for el in sorted(vm_per_el):
            avg_vm = sum(vm_per_el[el]) / len(vm_per_el[el])
            print(f'  Element {el:4d}: {avg_vm:8.2f}')
    else:
        print('No ELEMENT STRESS block found.')

if __name__ == '__main__':
```

```python
#!/usr/bin/env python3
"""
extract.py  –  pull key results from a CalculiX .dat file

Usage:
    python3 extract.py axiplane.dat

Outputs:
    • displacement of node 7 (U1, U2, U3)
    • reaction forces at node 7 (R1, R2, R3)
    • average von Mises stress per element (printed as a table)
"""

import sys
import re
from collections import defaultdict

def parse_dat(fname):
    with open(fname, 'r') as f:
        lines = f.readlines()

    # ------------------------------------------------------------------
    # 1) Displacements (section starts with "*NODE" then "*DISPLACEMENT")
    # ------------------------------------------------------------------
    disp = {}
    for i, line in enumerate(lines):
        if line.strip().upper().startswith('*DISPLACEMENT'):
            j = i + 1
            while j < len(lines) and not lines[j].strip().startswith('*'):
                parts = lines[j].split()
                nid = int(parts[0])
                ux, uy, uz = map(float, parts[1:4])
                disp[nid] = (ux, uy, uz)
                j += 1
            break

    # ------------------------------------------------------------------
    # 2) Reactions (section "*REACTION")
    # ------------------------------------------------------------------
    reac = {}
    for i, line in enumerate(lines):
        if line.strip().upper().startswith('*REACTION'):
            j = i + 1
            while j < len(lines) and not lines[j].strip().startswith('*'):
                parts = lines[j].split()
                nid = int(parts[0])
                fx, fy, fz = map(float, parts[1:4])
                reac[nid] = (fx, fy, fz)
                j += 1
            break

    # ------------------------------------------------------------------
    # 3) Stresses – we need the von Mises column (usually column 7)
    # ------------------------------------------------------------------
    # The stress block starts with "*STRESS" and ends with a line that
    # begins with "*".  The format is:
    #   el#  ip#  Sxx  Syy  Szz  Sxy  Syz  Szx  vonMises  ...
    vm_per_elem = defaultdict(list)
    for i, line in enumerate(lines):
        if line.strip().upper().startswith('*STRESS'):
            j = i + 1
            while j < len(lines) and not lines[j].strip().startswith('*'):
                parts = lines[j].split()
                if len(parts) < 9:
                    j += 1
                    continue
                el = int(parts[0])
                vm = float(parts[8])          # von Mises is column 9 (0‑based idx 8)
                vm_per_elem[el].append(vm)
                j += 1
            break

    # Compute average von Mises per element
    avg_vm = {el: sum(vms)/len(vms) for el, vms in vm_per_elem.items()}

    return disp, reac, avg_vm

def main():
    if len(sys.argv) != 2:
        print("Usage: python3 extract.py <file.dat>")
        sys.exit(1)

    fname = sys.argv[1]
    disp, reac, avg_vm = parse_dat(fname)

    nid = 7
    print(f"\n=== Node {nid} ===")
    if nid in disp:
        ux, uy, uz = disp[nid]
        print(f"Displacement (mm):   Ux = {ux: .6e}, Uy = {uy: .6e}, Uz = {uz: .6e}")
    else:
        print("Displacement: NOT FOUND")

    if nid in reac:
        fx, fy, fz = reac[nid]
        print(f"Reaction (N):        Fx = {fx: .6e}, Fy = {fy: .6e}, Fz = {fz: .6e}")
    else:
        print("Reaction: NOT FOUND")

    print("\n=== Average von Mises stress per element ===")
    print(f"{'Elem':>6} {'σ_vm (MPa)':>12}")
    for el in sorted(avg_vm):
        print(f"{el:6d} {avg_vm[el]:12.4f}")

if __name__ == "__main__":
    main()
```
