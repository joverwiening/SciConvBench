## 2️⃣  The script – `extract.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
extract.py
~~~~~~~~~~

Parse an Abaqus *.dat* file that contains *NODE PRINT* (U) and *EL PRINT* (S)
sections and write a single CSV file with the columns

    Node Number, U1, U2, U3, Element Number, S11, S22, S33, S12, S13, S23

The script works with the default Abaqus output format (ASCII, space‑separated
columns).  It can be used either from a plain Python interpreter or from the
Abaqus Python interpreter (which does not have pandas).

Author:  ChatGPT – adapted for your specific input file
"""

import sys
import re
import pathlib
from collections import OrderedDict

# ----------------------------------------------------------------------
# Helper: try to import pandas, otherwise fall back to csv module
# ----------------------------------------------------------------------
try:
    import pandas as pd
    _HAS_PANDAS = True
except Exception:               # pragma: no cover
    import csv
    _HAS_PANDAS = False

# ----------------------------------------------------------------------
# Regular expressions that match the two result blocks in the .dat file
# ----------------------------------------------------------------------
# Example of the node‑print block (U):
#   THE FOLLOWING NODAL DISPLACEMENTS ARE PRINTED
#   NODE   U1   U2   U3
#        1  0.00000E+00  0.00000E+00  0.00000E+00
#        2  1.23456E-03  2.34567E-03  3.45678E-03
NODE_BLOCK_RE = re.compile(
    r"THE FOLLOWING NODAL DISPLACEMENTS ARE PRINTED\s+"
    r"NODE\s+U1\s+U2\s+U3\s+(?P<lines>(?:\s*\d+.*\n)+)",
    re.IGNORECASE,
)

# Example of the element‑print block (S):
#   THE FOLLOWING ELEMENT STRESSES ARE PRINTED
#   ELEMENT   S11   S22   S33   S12   S13   S23
#        1  2.10000E+05  1.50000E+05  1.80000E+05  1.20000E+04  8.00000E+03  9.00000E+03
EL_BLOCK_RE = re.compile(
    r"THE FOLLOWING ELEMENT STRESSES ARE PRINTED\s+"
    r"ELEMENT\s+S11\s+S22\s+S33\s+S12\s+S13\s+S23\s+(?P<lines>(?:\s*\d+.*\n)+)",
    re.IGNORECASE,
)

# ----------------------------------------------------------------------
def _parse_block(block_text: str, ncols: int) -> OrderedDict:
    """
    Convert a block of whitespace‑separated numbers into an OrderedDict
    keyed by the first column (node or element number).

    Parameters
    ----------
    block_text : str
        The raw multiline string that contains the data rows.
    ncols : int
        Expected number of columns (including the ID column).

    Returns
    -------
    OrderedDict[int, list[float]]
        Mapping ``ID -> [col1, col2, …]`` preserving the order in which the
        rows appear in the file.
    """
    data = OrderedDict()
    for line in block_text.strip().splitlines():
        # Collapse any number of spaces/tabs into a single space and split
        parts = re.split(r"\s+", line.strip())
        if len(parts) < ncols:
            # Skip malformed lines (e.g. blank lines)
            continue
        try:
            key = int(parts[0])
            values = [float(p) for p in parts[1:ncols]]
            data[key] = values
        except ValueError:
            # If conversion fails we simply ignore the line – it is not data.
            continue
    return data


# ----------------------------------------------------------------------
def extract_from_dat(dat_path: pathlib.Path, csv_path: pathlib.Path) -> None:
    """
    Main driver: read the *.dat* file, pull out the two result blocks,
    and write the combined CSV.

    Parameters
    ----------
    dat_path : pathlib.Path
        Path to the Abaqus *.dat* file.
    csv_path : pathlib.Path
        Destination CSV file.
    """
    txt = dat_path.read_text(encoding="utf-8", errors="ignore")

    # ------------------------------------------------------------------
    # 1) Nodal displacements (U)
    # ------------------------------------------------------------------
    node_match = NODE_BLOCK_RE.search(txt)
    if not node_match:
        sys.exit(f"ERROR – could not find a *NODE PRINT* block in {dat_path}")

    node_block = node_match.group("lines")
    node_data = _parse_block(node_block, ncols=4)   # ID + U1 U2 U3
    print(f"Found {len(node_data)} nodal displacement records.")

    # ------------------------------------------------------------------
    # 2) Element stresses (S)
    # ------------------------------------------------------------------
    el_match = EL_BLOCK_RE.search(txt)
    if not el_match:
        sys.exit(f"ERROR – could not find an *EL PRINT* block in {dat_path}")

    el_block = el_match.group("lines")
    el_data = _parse_block(el_block, ncols=7)       # ID + S11 S22 S33 S12 S13 S23
    print(f"Found {len(el_data)} element stress records.")

    # ------------------------------------------------------------------
    # 3) Build the combined table
    # ------------------------------------------------------------------
    # Sort both dictionaries by their keys so that the i‑th node pairs with
    # the i‑th element.  This is deterministic and works even when the two
    # sets have different lengths – the shorter one is truncated.
    node_items = sorted(node_data.items())
    el_items   = sorted(el_data.items())

    n_pairs = min(len(node_items), len(el_items))
    rows = []
    for i in range(n_pairs):
        nid, u_vals = node_items[i]
        eid, s_vals = el_items[i]
        rows.append(
            [nid, *u_vals, eid, *s_vals]
        )

    # ------------------------------------------------------------------
    # 4) Write CSV
    # ------------------------------------------------------------------
    header = [
        "Node Number", "U1", "U2", "U3",
        "Element Number", "S11", "S22", "S33", "S12", "S13", "S23"
    ]

    if _HAS_PANDAS:                     # -------------------------------------------------
        df = pd.DataFrame(rows, columns=header)
        df.to_csv(csv_path, index=False, float_format="%.6e")
    else:                               # -------------------------------------------------
        with csv_path.open("w", newline="") as f:
            writer = csv.writer(f)
            writer.writerow(header)
            writer.writerows(rows)

    print(f"✅  CSV written to {csv_path} ( {n_pairs} rows )")


# ----------------------------------------------------------------------
if __name__ == "__main__":
    # ------------------------------------------------------------------
    # Command‑line interface
    # ------------------------------------------------------------------
    import argparse

    parser = argparse.ArgumentParser(
        description="Extract nodal U and element S from an Abaqus .dat file "
                    "and write a combined CSV."
    )
    parser.add_argument(
        "dat_file",
        type=pathlib.Path,
        help="Path to the Abaqus *.dat* file (output of the analysis)."
    )
    parser.add_argument(
        "-o", "--output",
        type=pathlib.Path,
        default=None,
        help="Destination CSV file.  If omitted, '<dat_file>.csv' is used."
    )
    args = parser.parse_args()

    dat_path = args.dat_file.resolve()
    if not dat_path.is_file():
        sys.exit(f"ERROR – file not found: {dat_path}")

    csv_path = args.output or dat_path.with_suffix(".csv")
    extract_from_dat(dat_path, csv_path)
```

## 4️⃣  (Optional) Extract directly from an **ODB** file

If you have the `.odb` file (the binary results database) you can avoid the intermediate *.dat* parsing altogether.  
The following short snippet uses the Abaqus **ODB API** to pull the same fields and write the identical CSV:

```python
# --------------------------------------------------------------
# odb_extract.py   –  works only inside the Abaqus Python env
# --------------------------------------------------------------
from odbAccess import openOdb
import csv
import pathlib

def extract_from_odb(odb_path, csv_path):
    odb = openOdb(path=odb_path)

    # 1) Nodal displacements (U) from the last step
    step = odb.steps[odb.steps.keys()[-1]]          # last step
    disp_field = step.frames[-1].fieldOutputs['U']

    # 2) Element stresses (S) from the same step
    stress_field = step.frames[-1].fieldOutputs['S']

    # Build dictionaries keyed by node/element label
    node_disp = {v.nodeLabel: v.data for v in disp_field.values}
    elem_stress = {v.elementLabel: v.data for v in stress_field.values}

    # Pair them (sorted order, truncate to the shorter list)
    node_items = sorted(node_disp.items())
    elem_items = sorted(elem_stress.items())
    n = min(len(node_items), len(elem_items))

    header = ["Node Number","U1","U2","U3",
              "Element Number","S11","S22","S33","S12","S13","S23"]

    with open(csv_path, "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(header)
        for i in range(n):
            nid, u = node_items[i]
            eid, s = elem_items[i]
            writer.writerow([nid, *u, eid, *s])

    odb.close()
    print(f"CSV written to {csv_path}")

# -----------------------------------------------------------------
if __name__ == "__main__":
    import sys, pathlib
    if len(sys.argv) != 2:
        print("Usage: abaqus python odb_extract.py <job.odb>")
        sys.exit(1)

    odb_file = pathlib.Path(sys.argv[1]).resolve()
    csv_file = odb_file.with_suffix(".csv")
    extract_from_odb(odb_file, csv_file)
```

Run it exactly as you would any Abaqus script:

```bash
abaqus python odb_extract.py job1.odb
```

The resulting CSV has **exactly the same layout** as the one produced from the *.dat* file.

```python
    # 1) Nodal displacements (U) from the last step/frame
    # ----------------------------------------------------------
    step = odb.steps[list(odb.steps.keys())[-1]]          # last step
    disp_field = step.frames[-1].fieldOutputs['U']

    # ----------------------------------------------------------
    # 2) Element stresses (S) from the same step/frame
    # ----------------------------------------------------------
    stress_field = step.frames[-1].fieldOutputs['S']

    # Build dictionaries keyed by node/element label
    node_disp = {v.nodeLabel: v.data for v in disp_field.values}
    elem_stress = {v.elementLabel: v.data for v in stress_field.values}

    # ----------------------------------------------------------
    # 3) Pair them (sorted order, truncate to the shorter list)
    # ----------------------------------------------------------
    node_items = sorted(node_disp.items())
    elem_items = sorted(elem_stress.items())
    n = min(len(node_items), len(elem_items))

    header = ["Node Number","U1","U2","U3",
              "Element Number","S11","S22","S33","S12","S13","S23"]

    with csv_path.open("w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(header)
        for i in range(n):
            nid, u = node_items[i]
            eid, s = elem_items[i]
            writer.writerow([nid, *u, eid, *s])

    odb.close()
    print(f"✅  CSV written to {csv_path}")

# --------------------------------------------------------------
if __name__ == "__main__":
    import sys
    if len(sys.argv) != 2:
        print("Usage: abaqus python odb_extract.py <job.odb>")
        sys.exit(1)

    odb_file = pathlib.Path(sys.argv[1]).resolve()
    csv_file = odb_file.with_suffix(".csv")
    extract_from_odb(odb_file, csv_file)
```

## The script

```python
#!/usr/bin/env python3
"""
abaqus_dat_extractor.py

Extract nodal displacements (U1‑U3) for the node set Nall and element stresses
(S11‑S23) for the element set Eall from an Abaqus *.dat* file, pair the i‑th
node with the i‑th element, and write the result as CSV.

Usage
-----
    python abaqus_dat_extractor.py  input.dat  [output.csv]

If the output file is omitted the CSV is written to stdout.
"""

import sys
import csv
import re
from pathlib import Path
from typing import List, Tuple, Dict, Iterable

# ----------------------------------------------------------------------
# Helper data structures
# ----------------------------------------------------------------------
NodeRecord = Tuple[int, float, float, float]          # (node_id, U1, U2, U3)
ElemRecord = Tuple[int, float, float, float,
                   float, float, float]               # (elem_id, S11, S22, S33, S12, S13, S23)

# ----------------------------------------------------------------------
# Parsing utilities
# ----------------------------------------------------------------------
def _strip_comments(line: str) -> str:
    """Remove Abaqus comment characters ('**') and trailing whitespace."""
    return line.split('**', 1)[0].strip()

def _read_block(lines: Iterable[str], start_keyword: str) -> List[str]:
    """
    Return all lines belonging to a block that starts with *start_keyword.
    The block ends when a line beginning with '*' (another keyword) is found
    or when the file ends.
    """
    block: List[str] = []
    in_block = False
    for raw in lines:
        line = _strip_comments(raw)
        if not line:
            continue
        if line.upper().startswith('*'):
            if in_block:
                # a new keyword – block finished
                break
            if line.upper().startswith(start_keyword.upper()):
                in_block = True
                continue          # skip the keyword line itself
        if in_block:
            block.append(line)
    return block

def _parse_node_block(block: List[str]) -> List[NodeRecord]:
    """
    The node‑print block typically looks like:
        1, 0.00123, -0.00045, 0.00000
        2, 0.00125, -0.00044, 0.00001
    i.e. node_id followed by U1, U2, U3 (comma‑separated).
    """
    nodes: List[NodeRecord] = []
    for line in block:
        parts = [p.strip() for p in line.split(',')]
        if len(parts) < 4:
            # some Abaqus versions may separate the values with spaces only
            parts = line.split()
        if len(parts) < 4:
            continue  # malformed line – skip
        try:
            nid = int(parts[0])
            u1, u2, u3 = map(float, parts[1:4])
            nodes.append((nid, u1, u2, u3))
        except ValueError:
            # non‑numeric line – ignore (e.g. header)
            continue
    return nodes

def _parse_elem_block(block: List[str]) -> List[ElemRecord]:
    """
    The element‑print block typically looks like:
        1, 123.4, 56.7, 78.9, 12.3, 45.6, 78.9
        2, 124.0, 57.0, 79.2, 12.5, 45.8, 79.1
    i.e. element_id followed by S11, S22, S33, S12, S13, S23.
    """
    elems: List[ElemRecord] = []
    for line in block:
        parts = [p.strip() for p in line.split(',')]
        if len(parts) < 7:
            parts = line.split()
        if len(parts) < 7:
            continue
        try:
            eid = int(parts[0])
            s_vals = tuple(map(float, parts[1:7]))
            elems.append((eid, *s_vals))
        except ValueError:
            continue
    return elems

# ----------------------------------------------------------------------
# Main extraction routine
# ----------------------------------------------------------------------
def extract_from_dat(dat_path: Path) -> Tuple[List[NodeRecord], List[ElemRecord]]:
    """
    Open *dat_path*, locate the *NODE PRINT* and *EL PRINT* blocks that belong
    to the sets Nall and Eall, and return the parsed data.
    """
    with dat_path.open('r', encoding='utf-8', errors='ignore') as f:
        all_lines = f.readlines()

    # Find the node block – we assume the first *NODE PRINT* that contains
    # the word "Nall" in the preceding comment line belongs to the set we want.
    node_block = []
    elem_block = []

    # Scan line‑by‑line looking for the keywords.  We keep a small look‑behind
    # buffer to see whether the preceding comment mentions Nall/Eall.
    lookbehind: List[str] = []
    for i, raw in enumerate(all_lines):
        line = _strip_comments(raw)
        if not line:
            lookbehind.append('')
            if len(lookbehind) > 5:
                lookbehind.pop(0)
            continue

        # Detect start of a block
        if line.upper().startswith('*NODE PRINT'):
            # Check the last few comment lines for the set name
            comment = ' '.join(l for l in lookbehind if l.startswith('**'))
            if 'NALL' in comment.upper():
                node_block = _read_block(all_lines[i+1:], '*NODE PRINT')
                # No need to continue scanning for another node block
        elif line.upper().startswith('*EL PRINT'):
            comment = ' '.join(l for l in lookbehind if l.startswith('**'))
            if 'EALL' in comment.upper():
                elem_block = _read_block(all_lines[i+1:], '*EL PRINT')

        # keep a short history for the next iteration
        lookbehind.append(line)
        if len(lookbehind) > 5:
            lookbehind.pop(0)

        # break early if both blocks have been found
        if node_block and elem_block:
            break

    if not node_block:
        raise RuntimeError("Could not locate a '*NODE PRINT' block for set NALL.")
    if not elem_block:
        raise RuntimeError("Could not locate a '*EL PRINT' block for set EALL.")

    nodes = _parse_node_block(node_block)
    elems = _parse_elem_block(elem_block)

    if not nodes:
        raise RuntimeError("No nodal data parsed from the '*NODE PRINT' block.")
    if not elems:
        raise RuntimeError("No element data parsed from the '*EL PRINT' block.")

    return nodes, elems

# ----------------------------------------------------------------------
# Pairing & CSV output
# ----------------------------------------------------------------------
def pair_and_write(nodes: List[NodeRecord],
                   elems: List[ElemRecord],
                   csv_path: Path | None = None) -> None:
    """
    Pair the i‑th node with the i‑th element and write a CSV.
    If *csv_path* is None the CSV is written to stdout.
    """
    # Ensure the two lists have the same length – if not, truncate to the
    # shorter one and warn the user.
    n_len = len(nodes)
    e_len = len(elems)
    if n_len != e_len:
        min_len = min(n_len, e_len)
        sys.stderr.write(
            f"Warning: number of nodes ({n_len}) != number of elements ({e_len}). "
            f"Only the first {min_len} pairs will be written.\n"
        )
        nodes = nodes[:min_len]
        elems = elems[:min_len]

    header = [
        'node_id', 'U1', 'U2', 'U3',
        'element_id', 'S11', 'S22', 'S33', 'S12', 'S13', 'S23'
    ]

    out_f = sys.stdout if csv_path is None else csv_path.open('w', newline='')

    with out_f as f:
        writer = csv.writer(f)
        writer.writerow(header)
        for n, e in zip(nodes, elems):
            writer.writerow(list(n) + list(e)[1:])   # skip duplicate element_id in e[0]?
            # Actually e already contains element_id as first entry, so we keep it.
            # The line above writes: node_id, U1, U2, U3, element_id, S11, …, S23

# ----------------------------------------------------------------------
# Command‑line interface
# ----------------------------------------------------------------------
def main(argv: List[str] | None = None) -> None:
    if argv is None:
        argv = sys.argv[1:]

    if not (2 <= len(argv) <= 3):
        sys.stderr.write(
            "Usage: python abaqus_dat_extractor.py  <input.dat>  [output.csv]\n"
        )
        sys.exit(1)

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file '{input_path}' does not exist.\n")
        sys.exit(1)

    output_path = Path(argv[1]) if len(argv) == 2 else None

    try:
        nodes, elems = extract_from_dat(input_path)
        pair_and_write(nodes, elems, output_path)
    except Exception as exc:
        sys.stderr.write(f"Error while processing: {exc}\n")
        sys.exit(1)


if __name__ == '__main__':
    main()
```

### What the script does, step‑by‑step

| Step | What happens | Why it matters |
|------|--------------|----------------|
| **1. Locate the blocks** | Scans the file for `*NODE PRINT` and `*EL PRINT` keywords that are preceded (in the comment lines) by the set names `NALL` and `EALL`. | Guarantees you get the data you asked Abaqus to write for those specific sets. |
| **2. Read the block contents** | Once a block is found, all subsequent lines up to the next `*` keyword are collected. | Abaqus prints the data line‑by‑line; we need the whole block to parse it. |
| **3. Parse nodal data** | Splits each line on commas (or whitespace) and converts the first four fields to `int, float, float, float`. | Gives you `(node_id, U1, U2, U3)`. |
| **4. Parse element data** | Splits each line and converts the first seven fields to `int, float…`. | Gives you `(elem_id, S11, S22, S33, S12, S13, S23)`. |
| **5. Pair i‑th node with i‑th element** | Zips the two lists (truncating to the shorter one if they differ). | Matches the *i‑th* node with the *i‑th* element as requested. |
| **6. Write CSV** | Writes a header line followed by one line per pair. | Easy to import into Excel, pandas, MATLAB, etc. |

#### Customising the script

* **Different output format** – replace the `csv.writer` block with JSON, HDF5, or any other format you prefer.
* **More fields** – if your `*NODE PRINT` or `*EL PRINT` blocks contain additional columns (e.g. reaction forces), just extend the parsing functions (`_parse_node_block`, `_parse_elem_block`) and the CSV header.
* **Multiple sets** – wrap the `extract_from_dat` call in a loop that changes the set name you look for (e.g. `NSET1`, `ESET1`, `NSET2`, …).

```python
NodeRecord = Tuple[int, float, float, float]          # (node_id, U1, U2, U3)
ElemRecord = Tuple[int, float, float, float,
                   float, float, float]               # (elem_id, S11, S22, S33, S12, S13, S23)

# ----------------------------------------------------------------------
```
```python
# Helper utilities
# ----------------------------------------------------------------------
def _strip_comments(line: str) -> str:
    """Remove Abaqus comment markers ('**') and trailing whitespace."""
    return line.split('**', 1)[0].strip()

def _read_block(lines: Iterable[str], start_keyword: str) -> List[str]:
    """
    Return all lines belonging to a block that starts with *start_keyword.
    The block ends when a new keyword (a line beginning with '*') is found
    or when the file ends.
    """
    block: List[str] = []
    in_block = False
    for raw in lines:
        line = _strip_comments(raw)
        if not line:
            continue
        if line.upper().startswith('*'):
            if in_block:
                # a new keyword – block finished
                break
            if line.upper().startswith(start_keyword.upper()):
                in_block = True
                continue          # skip the keyword line itself
        if in_block:
            block.append(line)
    return block

# ----------------------------------------------------------------------
# Parsing the two print sections
# ----------------------------------------------------------------------
def _parse_node_block(block: List[str]) -> List[NodeRecord]:
    """
    Expected format (comma‑ or space‑separated):
        1, 0.00123, -0.00045, 0.00000
        2, 0.00125, -0.00044, 0.00001
    Returns a list of (node_id, U1, U2, U3).
    """
    nodes: List[NodeRecord] = []
    for line in block:
        # Try comma first, fall back to whitespace
        parts = [p.strip() for p in line.split(',')]
        if len(parts) < 4:
            parts = line.split()
        if len(parts) < 4:
            continue
        try:
            nid = int(parts[0])
            u1, u2, u3 = map(float, parts[1:4])
            nodes.append((nid, u1, u2, u3))
        except ValueError:
            continue          # skip non‑numeric lines (e.g. headers)
    return nodes

def _parse_elem_block(block: List[str]) -> List[ElemRecord]:
    """
    Expected format (comma‑ or space‑separated):
        10, 123.4, 56.7, 78.9, 12.3, 45.6, 78.9
    Returns a list of (elem_id, S11, S22, S33, S12, S13, S23).
    """
    elems: List[ElemRecord] = []
    for line in block:
        parts = [p.strip() for p in line.split(',')]
        if len(parts) < 7:
            parts = line.split()
        if len(parts) < 7:
            continue
        try:
            eid = int(parts[0])
            s_vals = tuple(map(float, parts[1:7]))
            elems.append((eid, *s_vals))
        except ValueError:
            continue
    return elems

# ----------------------------------------------------------------------
# Locate the two print blocks that belong to the sets NALL and EALL
# ----------------------------------------------------------------------
def extract_from_dat(dat_path: Path) -> Tuple[List[NodeRecord], List[ElemRecord]]:
    """
    Open *dat_path*, locate the *NODE PRINT* and *EL PRINT* blocks that
    belong to the sets NALL and EALL, and return the parsed data.
    """
    with dat_path.open('r', encoding='utf-8', errors='ignore') as f:
        all_lines = f.readlines()

    node_block: List[str] = []
    elem_block: List[str] = []

    # Short look‑behind buffer (max 5 lines) to see the preceding comment
    lookbehind: List[str] = []
    for i, raw in enumerate(all_lines):
        line = _strip_comments(raw)
        if line.upper().startswith('*NODE PRINT'):
            comment = ' '.join(l for l in lookbehind if l.startswith('**'))
            if 'NALL' in comment.upper():
                node_block = _read_block(all_lines[i+1:], '*NODE PRINT')
        elif line.upper().startswith('*EL PRINT'):
            comment = ' '.join(l for l in lookbehind if l.startswith('**'))
            if 'EALL' in comment.upper():
                elem_block = _read_block(all_lines[i+1:], '*EL PRINT')

        # keep a short history (max 5 lines) for the next iteration
        lookbehind.append(line)
        if len(lookbehind) > 5:
            lookbehind.pop(0)

        if node_block and elem_block:
            break

    if not node_block:
        raise RuntimeError("Could not locate a '*NODE PRINT' block for set NALL.")
    if not elem_block:
        raise RuntimeError("Could not locate a '*EL PRINT' block for set EALL.")

    nodes = _parse_node_block(node_block)
    elems = _parse_elem_block(elem_block)

    if not nodes:
        raise RuntimeError("No nodal data parsed from the '*NODE PRINT' block.")
    if not elems:
        raise RuntimeError("No element data parsed from the '*EL PRINT' block.")

    return nodes, elems

# ----------------------------------------------------------------------
# Pair i‑th node with i‑th element and write CSV
# ----------------------------------------------------------------------
def pair_and_write(nodes: List[NodeRecord],
                   elems: List[ElemRecord],
                   csv_path: Path | None = None) -> None:
    """
    Pair the i‑th node with the i‑th element and write a CSV.
    If *csv_path* is None the CSV is written to stdout.
    """
    # Truncate to the shorter list if the numbers differ
    if len(nodes) != len(elems):
        min_len = min(len(nodes), len(elems))
        sys.stderr.write(
            f"Warning: {len(nodes)} nodes vs {len(elems)} elements – "
            f"only the first {min_len} pairs will be written.\n"
        )
        nodes = nodes[:min_len]
        elems = elems[:min_len]

    header = [
        'node_id', 'U1', 'U2', 'U3',
        'element_id', 'S11', 'S22', 'S33', 'S12', 'S13', 'S23'
    ]

    out_f = sys.stdout if csv_path is None else csv_path.open('w', newline='')

    with out_f as f:
        writer = csv.writer(f)
        writer.writerow(header)
        for n, e in zip(nodes, elems):
            # n = (node_id, U1, U2, U3)
            # e = (elem_id, S11, S22, S33, S12, S13, S23)
            writer.writerow(list(n) + list(e))

# ----------------------------------------------------------------------
# Command‑line interface
# ----------------------------------------------------------------------
def main(argv: List[str] | None = None) -> None:
    if argv is None:
        argv = sys.argv[1:]

    if not (1 <= len(argv) <= 2):
        sys.stderr.write(
            "Usage: python abaqus_dat_extractor.py <input.dat> [output.csv]\n"
        )
        sys.exit(1)

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file '{input_path}' does not exist.\n")
        sys.exit(1)

    output_path = Path(argv[1]) if len(argv) == 2 else None

    try:
        nodes, elems = extract_from_dat(input_path)
        pair_and_write(nodes, elems, output_path)
    except Exception as exc:
        sys.stderr.write(f"Error while processing: {exc}\n")
        sys.exit(1)


if __name__ == '__main__':
    main()
```

## `abaqus_dat_extractor.py`

```python
#!/usr/bin/env python3
"""
abaqus_dat_extractor.py

Extracts nodal displacement (U1‑U3) and elemental stress (S11‑S23) data from an
Abaqus *.dat* file and writes a CSV with one line per node/element pair.

Usage
-----
    python abaqus_dat_extractor.py <input.dat> [output.csv]

If *output.csv* is omitted the CSV is printed to STDOUT.

The script expects the following pattern in the *.dat* file:

    ** NALL
    *NODE PRINT, ...
    <data lines>

    ** EALL
    *EL PRINT, ...
    <data lines>

Only the first block that follows each comment is used.  Data lines may be
comma‑ or space‑separated; any extra columns are ignored.
"""

import sys
import csv
import itertools
from pathlib import Path

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _strip_comments(line: str) -> str:
    """Remove leading “*” or “**” comment markers and surrounding whitespace."""
    return line.lstrip("* ").strip()


def _is_comment_containing(line: str, token: str) -> bool:
    """True if the line is a comment (starts with '*') and contains *token*."""
    return line.lstrip().startswith("*") and token in line


def _read_block(lines_iter, start_keyword):
    """
    Consume lines from *lines_iter* until a line that starts with *start_keyword*
    is found, then return a list with all subsequent data lines (stopping at the
    next line that starts with '*').
    """
    # Find the start line
    for line in lines_iter:
        if line.lstrip().upper().startswith(start_keyword):
            break
    else:
        # never found the keyword
        return []

    # Collect data lines
    block = []
    for line in lines_iter:
        stripped = line.strip()
        if not stripped:                     # blank line → end of block
            continue
        if stripped.startswith("*"):         # next keyword → end of block
            break
        block.append(stripped)
    return block


def _parse_numbers(line):
    """Return a list of floats found in *line* (comma‑ or space‑separated)."""
    # Replace commas with spaces, split, and convert
    parts = line.replace(",", " ").split()
    try:
        return [float(p) for p in parts]
    except ValueError:
        # If conversion fails we simply ignore the line
        return []


def _extract_node_data(block):
    """
    From a *NODE PRINT* block return a list of tuples:
        (node_id, U1, U2, U3)
    The block is assumed to contain at least four columns per line.
    """
    nodes = []
    for line in block:
        nums = _parse_numbers(line)
        if len(nums) >= 4:
            node_id, u1, u2, u3 = nums[:4]
            nodes.append((int(node_id), u1, u2, u3))
    return nodes


def _extract_elem_data(block):
    """
    From an *EL PRINT* block return a list of tuples:
        (elem_id, S11, S22, S33, S12, S13, S23)
    The block is assumed to contain at least seven columns per line.
    """
    elems = []
    for line in block:
        nums = _parse_numbers(line)
        if len(nums) >= 7:
            elem_id, s11, s22, s33, s12, s13, s23 = nums[:7]
            elems.append((int(elem_id), s11, s22, s33, s12, s13, s23))
    return elems


# ----------------------------------------------------------------------
# Core extraction routine
# ----------------------------------------------------------------------
def extract_from_dat(dat_path: Path):
    """
    Parse *dat_path* and return two lists:
        nodes  – [(node_id, U1, U2, U3), ...]
        elems  – [(elem_id, S11, S22, S33, S12, S13, S23), ...]
    Only the first *NODE PRINT* block that follows a comment containing
    “NALL” and the first *EL PRINT* block that follows a comment containing
    “EALL” are used.
    """
    with dat_path.open("r", encoding="utf-8") as f:
        lines = f.readlines()

    # ------------------------------------------------------------------
    # Find the two blocks
    # ------------------------------------------------------------------
    node_block = []
    elem_block = []

    # We walk through the file line‑by‑line keeping a look‑ahead iterator.
    # When we see a comment that contains the marker we start reading the
    # corresponding block.
    line_iter = iter(lines)
    for line in line_iter:
        # ---- NALL ----------------------------------------------------
        if _is_comment_containing(line, "NALL"):
            node_block = _read_block(line_iter, "*NODE PRINT")
            # we only need the first occurrence
            # (break only if both blocks have been found)
        # ---- EALL ----------------------------------------------------
        elif _is_comment_containing(line, "EALL"):
            elem_block = _read_block(line_iter, "*EL PRINT")

        if node_block and elem_block:
            break   # both blocks have been collected

    # ------------------------------------------------------------------
    # Convert raw text blocks to numeric lists
    # ------------------------------------------------------------------
    nodes = _extract_node_data(node_block)
    elems = _extract_elem_data(elem_block)

    return nodes, elems


# ----------------------------------------------------------------------
# CSV writer
# ----------------------------------------------------------------------
def pair_and_write(nodes, elems, out_fh):
    """
    Pair the *i‑th* node with the *i‑th* element (truncating to the shorter list)
    and write a CSV to *out_fh*.
    """
    header = [
        "node_id", "U1", "U2", "U3",
        "element_id", "S11", "S22", "S33", "S12", "S13", "S23"
    ]
    writer = csv.writer(out_fh)
    writer.writerow(header)

    for (nid, u1, u2, u3), (eid, s11, s22, s33, s12, s13, s23) in itertools.zip_longest(
        nodes, elems, fillvalue=None
    ):
        # Stop when either side is None (i.e. one list is shorter)
        if nid is None or eid is None:
            break
        writer.writerow([nid, u1, u2, u3, eid, s11, s22, s33, s12, s13, s23])


# ----------------------------------------------------------------------
# CLI entry point
# ----------------------------------------------------------------------
def main(argv=None):
    argv = argv or sys.argv[1:]

    if not (1 <= len(argv) <= 2):
        sys.stderr.write(
            "Usage: python abaqus_dat_extractor.py <input.dat> [output.csv]\n"
        )
        return 1

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file not found – {input_path}\n")
        return 1

    # ------------------------------------------------------------------
    # Extract data
    # ------------------------------------------------------------------
    nodes, elems = extract_from_dat(input_path)

    if not nodes:
        sys.stderr.write("Warning: no *NODE PRINT* block found after ** NALL\n")
    if not elems:
        sys.stderr.write("Warning: no *EL PRINT* block found after ** EALL\n")

    # ------------------------------------------------------------------
    # Write CSV (to file or stdout)
    # ------------------------------------------------------------------
    if len(argv) == 2:
        out_path = Path(argv[1])
        with out_path.open("w", newline="", encoding="utf-8") as out_fh:
            pair_and_write(nodes, elems, out_fh)
    else:
        # stdout – csv.writer expects a text stream, sys.stdout is fine
        pair_and_write(nodes, elems, sys.stdout)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## How it works (step‑by‑step)

| Step | What the script does |
|------|----------------------|
| **1️⃣  Locate the comment** | Scans the file line‑by‑line. When it sees a line that starts with `*` (or `**`) **and** contains the word `NALL` it knows the next `*NODE PRINT` block belongs to you. The same logic applies for `EALL` → `*EL PRINT`. |
| **2️⃣  Grab the block** | Once the keyword (`*NODE PRINT` or `*EL PRINT`) is found, every subsequent non‑blank line that **does not** start with `*` is treated as data. The block ends when another line beginning with `*` appears (or the file ends). |
| **3️⃣  Parse numbers** | Each data line is split on commas **or** whitespace, converted to `float`. The first column is cast to `int` because node/element IDs are integral. |
| **4️⃣  Keep only the columns you need** | From the node block we keep the first four numbers → `node_id, U1, U2, U3`. From the element block we keep the first seven numbers → `elem_id, S11, S22, S33, S12, S13, S23`. |
| **5️⃣  Pair & truncate** | `itertools.zip_longest` is used to walk the two lists in parallel; as soon as one list runs out the loop breaks, effectively truncating to the shorter list. |
| **6️⃣  Write CSV** | The header is written first, then each paired row. If an output filename is supplied the CSV is written there; otherwise it goes to `stdout`. |

```python
#!/usr/bin/env python3
"""
abaqus_dat_extractor.py

Extracts the first 4 columns of the *NODE PRINT block that follows a comment
containing "NALL" and the first 7 columns of the *EL PRINT block that follows a
comment containing "EALL".  The i‑th node row is paired with the i‑th element
row (truncated to the shorter list) and written as CSV.

Usage
-----
    python abaqus_dat_extractor.py <input>.dat [<output>.csv]

If <output>.csv is omitted the CSV is printed to STDOUT.

The script works with both comma‑ and space‑separated numeric fields and ignores
any trailing text after the required columns.

Author:  ChatGPT (2024‑06)
"""

import sys
import csv
import itertools
from pathlib import Path

# --------------------------------------------------------------------------- #
# Helper functions
# --------------------------------------------------------------------------- #
def _is_comment_containing(line: str, token: str) -> bool:
    """
    Return True if the line is a comment (starts with '*') and contains the
    given token (case‑insensitive).  Abaqus comments start with '**' but we
    accept any leading '*'.
    """
    return line.lstrip().startswith("*") and token.lower() in line.lower()


def _strip_comments(line: str) -> str:
    """Remove any trailing comment that starts with '**' or '*'.  Used for safety."""
    for marker in ("**", "*"):
        idx = line.find(marker)
        if idx != -1:
            return line[:idx].strip()
    return line.strip()


def _parse_numbers(line: str) -> list[float]:
    """
    Split a line on commas or whitespace, discard empty fields and convert the
    first few entries to float.  Returns a list of floats.
    """
    # Remove any inline comment first
    line = _strip_comments(line)

    # Split on commas first; if no commas, split on whitespace
    if "," in line:
        parts = [p.strip() for p in line.split(",")]
    else:
        parts = line.split()

    # Convert to float, ignoring conversion errors (they will be filtered later)
    nums = []
    for p in parts:
        if not p:
            continue
        try:
            nums.append(float(p))
        except ValueError:
            # Stop at the first non‑numeric token – typical for Abaqus extra text
            break
    return nums


def _read_block(lines_iter, start_keyword: str) -> list[list[float]]:
    """
    Read a block that starts with a line that begins with *<start_keyword>.
    The block ends at the first blank line or at the next line that starts
    with '*'.  Returns a list of rows, each row being a list of floats.
    """
    block = []
    for line in lines_iter:
        stripped = line.strip()
        # End of block?
        if not stripped or stripped.startswith("*"):
            # Put the line back for the outer loop to see (if it starts with '*')
            # by using a generator trick – we simply break and let the caller
            # continue from the current line.
            break
        nums = _parse_numbers(stripped)
        if nums:                     # ignore completely empty lines
            block.append(nums)
    return block


def extract_from_dat(dat_path: Path):
    """
    Scan the .dat file and return two lists:
        nodes  -> list of rows (first 4 columns of *NODE PRINT)
        elems  -> list of rows (first 7 columns of *EL PRINT)
    """
    nodes = []
    elems = []

    with dat_path.open("r", encoding="utf-8", errors="ignore") as f:
        # Turn the file into an iterator so we can “peek” ahead
        lines = iter(f)

        for raw_line in lines:
            line = raw_line.rstrip("\n")

            # -------------------------------------------------------------
            # Look for the NALL comment → then the *NODE PRINT block
            # -------------------------------------------------------------
            if _is_comment_containing(line, "NALL"):
                # Skip any intervening blank lines/comments until we hit *NODE
                for nxt in lines:
                    if nxt.strip().upper().startswith("*NODE"):
                        # We have the *NODE PRINT line – the actual data starts on the next line
                        nodes = _read_block(lines, "*NODE")
                        break
                # Continue scanning for the next section (EALL)
                continue

            # -------------------------------------------------------------
            # Look for the EALL comment → then the *EL PRINT block
            # -------------------------------------------------------------
            if _is_comment_containing(line, "EALL"):
                for nxt in lines:
                    if nxt.strip().upper().startswith("*EL"):
                        elems = _read_block(lines, "*EL")
                        break
                continue

    # Truncate / pad to the shorter length (as requested)
    min_len = min(len(nodes), len(elems))
    nodes = nodes[:min_len]
    elems = elems[:min_len]

    # Keep only the required columns
    nodes = [row[:4] for row in nodes]   # node_id, U1, U2, U3
    elems = [row[:7] for row in elems]   # elem_id, S11, S22, S33, S12, S13, S23

    return nodes, elems


def pair_and_write(nodes, elems, out_fh):
    """
    Write the paired data as CSV.  Header is:
        node_id,U1,U2,U3,element_id,S11,S22,S33,S12,S13,S23
    """
    writer = csv.writer(out_fh)
    header = [
        "node_id",
        "U1",
        "U2",
        "U3",
        "element_id",
        "S11",
        "S22",
        "S33",
        "S12",
        "S13",
        "S23",
    ]
    writer.writerow(header)

    for nrow, erow in zip(nodes, elems):
        # Convert all numbers back to string (preserve original formatting as much as possible)
        writer.writerow([*nrow, *erow])


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def main(argv=None):
    if argv is None:
        argv = sys.argv[1:]

    if not (1 <= len(argv) <= 2):
        sys.stderr.write(
            "Usage: python abaqus_dat_extractor.py <input>.dat [<output>.csv]\n"
        )
        return 1

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file not found – {input_path}\n")
        return 1

    output_path = Path(argv[1]) if len(argv) == 2 else None

    # ------------------------------------------------------------------- #
    # Extraction
    # ------------------------------------------------------------------- #
    nodes, elems = extract_from_dat(input_path)

    if not nodes:
        sys.stderr.write("Warning: No node data found.\n")
    if not elems:
        sys.stderr.write("Warning: No element data found.\n")
    if not nodes or not elems:
        sys.stderr.write("No paired data to write – exiting.\n")
        return 1

    # ------------------------------------------------------------------- #
    # Write CSV
    # ------------------------------------------------------------------- #
    if output_path:
        with output_path.open("w", newline="", encoding="utf-8") as out_fh:
            pair_and_write(nodes, elems, out_fh)
    else:
        # Write to STDOUT
        pair_and_write(nodes, elems, sys.stdout)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
        nodes  -> list of rows (first 4 columns of *NODE PRINT)
        elems  -> list of rows (first 7 columns of *EL PRINT)
    """
    nodes = []
    elems = []

    with dat_path.open("r", encoding="utf-8", errors="ignore") as f:
        lines = iter(f)                     # iterator so we can “peek” ahead

        for raw_line in lines:
            line = raw_line.rstrip("\n")

            # -------------------------------------------------------------
            # Look for the NALL comment → then the *NODE PRINT block
            # -------------------------------------------------------------
            if _is_comment_containing(line, "NALL"):
                for nxt in lines:
                    if nxt.strip().upper().startswith("*NODE"):
                        # Data starts on the line *after* the *NODE line
                        nodes = _read_block(lines)
                        break
                continue

            # -------------------------------------------------------------
            # Look for the EALL comment → then the *EL PRINT block
            # -------------------------------------------------------------
            if _is_comment_containing(line, "EALL"):
                for nxt in lines:
                    if nxt.strip().upper().startswith("*EL"):
                        elems = _read_block(lines)
                        break
                continue

    # Truncate to the shorter length (as requested)
    min_len = min(len(nodes), len(elems))
    nodes = nodes[:min_len]
    elems = elems[:min_len]

    # Keep only the required columns
    nodes = [row[:4] for row in nodes]   # node_id, U1, U2, U3
    elems = [row[:7] for row in elems]   # elem_id, S11, S22, S33, S12, S13, S23

    return nodes, elems


def pair_and_write(nodes, elems, out_fh):
    """
    Write the paired data as CSV.  Header is:
        node_id,U1,U2,U3,element_id,S11,S22,S33,S12,S13,S23
    """
    writer = csv.writer(out_fh)
    header = [
        "node_id", "U1", "U2", "U3",
        "element_id", "S11", "S22", "S33", "S12", "S13", "S23"
    ]
    writer.writerow(header)

    for nrow, erow in zip(nodes, elems):
        writer.writerow([*nrow, *erow])


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def main(argv=None):
    if argv is None:
        argv = sys.argv[1:]

    if not (1 <= len(argv) <= 2):
        sys.stderr.write(
            "Usage: python abaqus_dat_extractor.py <input>.dat [<output>.csv]\n"
        )
        return 1

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file not found – {input_path}\n")
        return 1

    output_path = Path(argv[1]) if len(argv) == 2 else None

    # ------------------------------------------------------------------- #
    # Extraction
    # ------------------------------------------------------------------- #
    nodes, elems = extract_from_dat(input_path)

    if not nodes:
        sys.stderr.write("Warning: No node data found.\n")
    if not elems:
        sys.stderr.write("Warning: No element data found.\n")
    if not nodes or not elems:
        sys.stderr.write("No paired data to write – exiting.\n")
        return 1

    # ------------------------------------------------------------------- #
    # Write CSV
    # ------------------------------------------------------------------- #
    if output_path:
        with output_path.open("w", newline="", encoding="utf-8") as out_fh:
            pair_and_write(nodes, elems, out_fh)
    else:
        # Write to STDOUT
        pair_and_write(nodes, elems, sys.stdout)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## `abaqus_dat_extractor.py`

```python
#!/usr/bin/env python3
"""
abaqus_dat_extractor.py

Extract the *NODE PRINT* and *EL PRINT* sections that follow the
comment lines containing “NALL” and “EALL” from an Abaqus *.dat* file,
pair the i‑th node row with the i‑th element row and write a CSV.

Usage
-----
    ./abaqus_dat_extractor.py <input>.dat [<output>.csv]

If <output>.csv is omitted the CSV is written to STDOUT.

The script is deliberately tiny (≈150 LOC) so that the one‑line tweaks
listed in the original description can be applied without hunting
through a huge code base.
"""

import sys
import csv
from pathlib import Path
from typing import List, Tuple, Iterable

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _is_comment_containing(line: str, token: str) -> bool:
    """
    Return True if *line* is a comment (starts with '*') and contains *token*.
    The check is case‑insensitive and ignores leading/trailing whitespace.
    """
    return line.lstrip().startswith("*") and token.upper() in line.upper()


def _read_block(it: Iterable[str], start_keyword: str) -> List[List[str]]:
    """
    Consume lines from *it* (an iterator over the file) until a line that
    starts with *start_keyword* (e.g. "*NODE") is found.  Then read the
    comma‑separated data rows that follow, stopping when a blank line or a
    line that starts with '*' (the next Abaqus keyword) is encountered.

    Returns a list of rows, each row being a list of stripped strings.
    """
    # Find the first line that begins the block
    for line in it:
        if line.strip().upper().startswith(start_keyword):
            break
    else:
        # No block found – return empty list
        return []

    # Now collect the data rows
    rows: List[List[str]] = []
    for line in it:
        stripped = line.strip()
        if not stripped:                     # blank line → end of block
            break
        if stripped.startswith("*"):         # next Abaqus keyword → end of block
            # Put the line back for the outer loop to see it again
            it = _prepend(line, it)
            break
        rows.append([c.strip() for c in stripped.split(",")])
    return rows


def _prepend(value, iterator):
    """
    Return a new iterator that yields *value* first, then everything from
    *iterator*.  Used to “push back” a line that we have read too far.
    """
    yield value
    yield from iterator


# ----------------------------------------------------------------------
# Core extraction logic
# ----------------------------------------------------------------------
def extract_from_dat(dat_path: Path) -> Tuple[List[List[str]], List[List[str]]]:
    """
    Parse *dat_path* and return two lists:

        nodes  – rows from the *NODE PRINT* block that follows a comment
                 containing “NALL”.
        elems  – rows from the *EL PRINT* block that follows a comment
                 containing “EALL”.

    Only the first occurrence of each comment/block pair is used.
    """
    nodes: List[List[str]] = []
    elems: List[List[str]] = []

    with dat_path.open("r", encoding="utf-8") as f:
        lines = iter(f)                     # make an iterator we can advance

        for line in lines:
            # --------------------------------------------------------------
            # 1️⃣  Look for the NALL comment → grab the following *NODE* block
            # --------------------------------------------------------------
            if _is_comment_containing(line, "NALL"):
                nodes = _read_block(lines, "*NODE")
                # If you also want to accept "*NODE FILE" you can change the
                # line above to:  "*NODE", "*NODE FILE"
                continue

            # --------------------------------------------------------------
            # 2️⃣  Look for the EALL comment → grab the following *EL* block
            # --------------------------------------------------------------
            if _is_comment_containing(line, "EALL"):
                elems = _read_block(lines, "*EL")
                continue

    return nodes, elems


# ----------------------------------------------------------------------
# Pairing & CSV output
# ----------------------------------------------------------------------
def pair_and_write(
    nodes: List[List[str]],
    elems: List[List[str]],
    out_fh,
    delimiter: str = ",",
) -> None:
    """
    Pair the i‑th node row with the i‑th element row (truncating to the
    shorter list) and write a CSV to *out_fh*.

    The CSV header is:

        node_id,U1,U2,U3,element_id,S11,S22,S33,S12,S13,S23
    """
    # Keep only the columns we need
    node_cols = [row[:4] for row in nodes]   # 0‑3 → id, U1, U2, U3
    elem_cols = [row[:7] for row in elems]   # 0‑6 → id, S11 … S23

    # Truncate to the shorter of the two lists (as required)
    pair_count = min(len(node_cols), len(elem_cols))

    header = [
        "node_id",
        "U1",
        "U2",
        "U3",
        "element_id",
        "S11",
        "S22",
        "S33",
        "S12",
        "S13",
        "S23",
    ]

    writer = csv.writer(out_fh, delimiter=delimiter, lineterminator="\n")
    writer.writerow(header)

    for i in range(pair_count):
        writer.writerow(node_cols[i] + elem_cols[i])


# ----------------------------------------------------------------------
# CLI entry point
# ----------------------------------------------------------------------
def main(argv: List[str] = None) -> int:
    argv = argv if argv is not None else sys.argv[1:]

    if not (1 <= len(argv) <= 2):
        sys.stderr.write(
            "Usage: {} <input>.dat [<output>.csv]\n".format(Path(sys.argv[0]).name)
        )
        return 1

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file not found – {input_path}\n")
        return 1

    # ------------------------------------------------------------------
    # 1️⃣  Extract the two blocks
    # ------------------------------------------------------------------
    nodes, elems = extract_from_dat(input_path)

    if not nodes:
        sys.stderr.write("Warning: No *NODE PRINT* block found after a NALL comment.\n")
    if not elems:
        sys.stderr.write("Warning: No *EL PRINT* block found after an EALL comment.\n")

    # ------------------------------------------------------------------
    # 2️⃣  Write CSV (to file or STDOUT)
    # ------------------------------------------------------------------
    if len(argv) == 2:
        out_path = Path(argv[1])
        with out_path.open("w", newline="", encoding="utf-8") as out_fh:
            pair_and_write(nodes, elems, out_fh)
    else:
        # STDOUT – use sys.stdout.buffer for proper handling of newlines on Windows
        pair_and_write(nodes, elems, sys.stdout)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## 1️⃣ The script – `abaqus_dat_extractor.py`

```python
#!/usr/bin/env python3
"""
abaqus_dat_extractor.py

Extract the *NODE PRINT* and *EL PRINT* sections that follow the
comment lines containing “NALL” and “EALL” from an Abaqus *.dat* file,
pair the i‑th node row with the i‑th element row and write a CSV.

Usage
-----
    ./abaqus_dat_extractor.py <input>.dat [<output>.csv]

If <output>.csv is omitted the CSV is written to STDOUT.

The script is deliberately tiny (≈150 LOC) so that the one‑line tweaks
listed in the original description can be applied without hunting
through a huge code base.
"""

import sys
import csv
from pathlib import Path
from typing import List, Tuple, Iterable

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _is_comment_containing(line: str, token: str) -> bool:
    """True if *line* is a comment (starts with '*') and contains *token*."""
    return line.lstrip().startswith("*") and token.upper() in line.upper()


def _prepend(value, iterator):
    """Yield *value* first, then everything from *iterator*."""
    yield value
    yield from iterator


def _read_block(it: Iterable[str], start_keyword: str) -> List[List[str]]:
    """
    Consume lines from *it* until a line that starts with *start_keyword*
    (e.g. "*NODE") is found. Then read the comma‑separated data rows that
    follow, stopping when a blank line or a line that starts with '*'
    (the next Abaqus keyword) is encountered.

    Returns a list of rows, each row being a list of stripped strings.
    """
    # Find the first line that begins the block
    for line in it:
        if line.strip().upper().startswith(start_keyword):
            break
    else:                     # No block found → empty list
        return []

    # Collect the data rows
    rows: List[List[str]] = []
    for line in it:
        stripped = line.strip()
        if not stripped:                     # blank line → end of block
            break
        if stripped.startswith("*"):         # next Abaqus keyword → end of block
            it = _prepend(line, it)          # push the line back for outer loop
            break
        rows.append([c.strip() for c in stripped.split(",")])
    return rows


# ----------------------------------------------------------------------
# Core extraction logic
# ----------------------------------------------------------------------
def extract_from_dat(dat_path: Path) -> Tuple[List[List[str]], List[List[str]]]:
    """
    Parse *dat_path* and return two lists:

        nodes – rows from the *NODE PRINT* block that follows a comment
                containing “NALL”.
        elems – rows from the *EL PRINT* block that follows a comment
                containing “EALL”.

    Only the first occurrence of each comment/block pair is used.
    """
    nodes: List[List[str]] = []
    elems: List[List[str]] = []

    with dat_path.open("r", encoding="utf-8") as f:
        lines = iter(f)                     # iterator we can advance

        for line in lines:
            # 1️⃣ NALL comment → grab the following *NODE* block
            if _is_comment_containing(line, "NALL"):
                nodes = _read_block(lines, "*NODE")
                continue

            # 2️⃣ EALL comment → grab the following *EL* block
            if _is_comment_containing(line, "EALL"):
                elems = _read_block(lines, "*EL")
                continue

    return nodes, elems


# ----------------------------------------------------------------------
# Pairing & CSV output
# ----------------------------------------------------------------------
def pair_and_write(
    nodes: List[List[str]],
    elems: List[List[str]],
    out_fh,
    delimiter: str = ",",
) -> None:
    """
    Pair the i‑th node row with the i‑th element row (truncating to the
    shorter list) and write a CSV to *out_fh*.

    Header:
        node_id,U1,U2,U3,element_id,S11,S22,S33,S12,S13,S23
    """
    # Keep only the columns we need
    node_cols = [row[:4] for row in nodes]   # id, U1, U2, U3
    elem_cols = [row[:7] for row in elems]   # id, S11 … S23

    pair_count = min(len(node_cols), len(elem_cols))

    header = [
        "node_id", "U1", "U2", "U3",
        "element_id", "S11", "S22", "S33", "S12", "S13", "S23",
    ]

    writer = csv.writer(out_fh, delimiter=delimiter, lineterminator="\n")
    writer.writerow(header)

    for i in range(pair_count):
        writer.writerow(node_cols[i] + elem_cols[i])


# ----------------------------------------------------------------------
# CLI entry point
# ----------------------------------------------------------------------
def main(argv: List[str] = None) -> int:
    argv = argv if argv is not None else sys.argv[1:]

    if not (1 <= len(argv) <= 2):
        sys.stderr.write(
            f"Usage: {Path(sys.argv[0]).name} <input>.dat [<output>.csv]\n"
        )
        return 1

    input_path = Path(argv[0])
    if not input_path.is_file():
        sys.stderr.write(f"Error: file not found – {input_path}\n")
        return 1

    # 1️⃣ Extract the two blocks
    nodes, elems = extract_from_dat(input_path)

    if not nodes:
        sys.stderr.write(
            "Warning: No *NODE PRINT* block found after a NALL comment.\n"
        )
    if not elems:
        sys.stderr.write(
            "Warning: No *EL PRINT* block found after an EALL comment.\n"
        )

    # 2️⃣ Write CSV (to file or STDOUT)
    if len(argv) == 2:
        out_path = Path(argv[1])
        with out_path.open("w", newline="", encoding="utf-8") as out_fh:
            pair_and_write(nodes, elems, out_fh)
    else:
        # STDOUT – use sys.stdout (text mode) for proper newline handling
        pair_and_write(nodes, elems, sys.stdout)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
    "element_id", "S11", "S22", "S33",   # element columns
    "S12", "S13", "S23"
]  # ← CUSTOMISE: add "U4" here if you need a fourth displacement column

#    (the raw node line is: id, U1, U2, U3, …)

    level=logging.INFO,
    format="%(levelname)s: %(message)s",
    stream=sys.stderr,
)

# ----------------------------------------------------------------------
#  INTERNAL HELPERS
# ----------------------------------------------------------------------
def _is_comment_containing(line: str, token: str) -> bool:
    """
    Returns True if *line* is a comment (starts with '*') and contains *token*.
    The check is case‑insensitive and ignores leading/trailing whitespace.
    """
    return line.lstrip().startswith("*") and token.lower() in line.lower()


def _read_block(lines: Iterable[str], start_marker: str) -> List[List[str]]:
    """
    Reads a block that starts with *start_marker* (e.g. "*NODE PRINT")
    and continues until the next line that starts with "*".
    Returns a list of rows, each row being a list of stripped strings.
    """
    block: List[List[str]] = []
    in_block = False
    for raw in lines:
        line = raw.strip()
        if not line:
            continue

        if in_block:
            if line.startswith("*"):          # end of block
                break
            # split on commas (Abaqus .dat files are comma‑separated)
            block.append([c.strip() for c in line.split(",")])
        elif line.upper().startswith(start_marker.upper()):
            in_block = True
    return block


def _extract_section(
    lines: List[str],
    comment_token: str,
    block_marker: str,
    cols_to_keep: int,
) -> List[List[str]]:
    """
    Finds the first comment that contains *comment_token*,
    then reads the block that starts with *block_marker*.
    Returns the block truncated to *cols_to_keep* columns per row.
    """
    for i, line in enumerate(lines):
        if _is_comment_containing(line, comment_token):
            # The block starts somewhere after this comment.
            # We slice the original list to start at the next line.
            block = _read_block(lines[i + 1 :], block_marker)

    # ------------------------------------------------------------------
    # 1️⃣  Extract node rows
    # ------------------------------------------------------------------
    nodes = _extract_section(
        lines,
        comment_token="NALL",
        block_marker="*NODE PRINT",
        cols_to_keep=NODE_COLS,

    # ------------------------------------------------------------------
    # 2️⃣  Extract element rows
    # ------------------------------------------------------------------
    elems = _extract_section(
        lines,
        comment_token="EALL",
        block_marker="*EL PRINT",
        cols_to_keep=EL_COLS,
    )
    return nodes, elems

    nodes: List[List[str]], elems: List[List[str]]
) -> List[List[str]]:
    """
    Pair the i‑th node row with the i‑th element row.
    If the two lists differ in length we truncate to the shorter one
    (or raise an error if RAISE_ON_MISMATCH is True).
    """
    if RAISE_ON_MISMATCH and len(nodes) != len(elems):
        raise ValueError(
            f"Node count ({len(nodes)}) != element count ({len(elems)})"
        )
    length = min(len(nodes), len(elems))
    paired = [nodes[i] + elems[i] for i in range(length)]
    if len(nodes) != len(elems):
        logging.info(
            f"Truncated to {length} rows (nodes={len(nodes)}, elems={len(elems)})"
        )
    return paired

    using the global HEADER and DELIMITER.
    """

    if len(HEADER) != len(rows[0]) if rows else 0:

            f"Header length ({len(HEADER)}) does not match data width ({len(rows[0]) if rows else 0})."
        )
    # Choose the output stream
    out_fh = out_path.open("w", newline="", encoding="utf-8") if out_path else sys.stdout
    writer = csv.writer(out_fh, delimiter=DELIMITER)
    writer.writerow(HEADER)
    writer.writerows(rows)
    if out_path:
        out_fh.close()
        logging.info(f"Wrote CSV to {out_path}")


# ----------------------------------------------------------------------
#  MAIN ENTRY POINT
# ----------------------------------------------------------------------
def main(argv: List[str] | None = None) -> int:

    CLI entry point.
    Usage:
        python abaqus_dat_extractor.py <input>.dat [<output>.csv]

    # ------------------------------------------------------------------
    paired = pair_rows(nodes, elems)
    write_csv(paired, output_path)
```
