## 2️⃣  Python parser → `summary.csv`

Save the following script as **`parse_beamtor.py`** in the same directory as `beamtor.dat` (and `results.json`).  
It

* reads `results.json` (just to confirm the job succeeded),
* parses `beamtor.dat` for **all nodes** (displacements) and **all elements** (von Mises stress, pressure if any),
* writes a CSV called `summary.csv`,
* prints the three global quantities you asked for.

```python
#!/usr/bin/env python3
# --------------------------------------------------------------
# parse_beamtor.py
#   - reads results.json (only to check status)
#   - parses beamtor.dat for:
#        * nodal displacements Ux, Uy, Uz
#        * element von Mises stress (SVM)
#        * element pressure (if a pressure load was defined)
#   - writes summary.csv
#   - prints:
#        max |U|  and node,
#        max von Mises and element,
#        reaction forces at the fixed support
# --------------------------------------------------------------

import json
import csv
import re
import sys
from pathlib import Path
import math

# ------------------------------------------------------------------
# 1.  Load results.json ------------------------------------------------
json_file = Path("results.json")
if not json_file.is_file():
    sys.exit("ERROR: results.json not found")
with json_file.open() as f:
    jdata = json.load(f)

if jdata.get("status") != "success":
    sys.exit("ERROR: job did not finish successfully")

print(f"Solver: {jdata['solver']}  –  Version: {jdata['ccx_version']}")
print("-" * 60)

# ------------------------------------------------------------------
# 2.  Parse the .dat file --------------------------------------------
dat_file = Path("beamtor.dat")
if not dat_file.is_file():
    sys.exit("ERROR: beamtor.dat not found – run the job first")

# Containers
node_disp = {}          # node_id -> (Ux, Uy, Uz)
elem_vm   = {}          # elem_id -> vonMises
reaction  = [0.0, 0.0, 0.0]   # total reaction forces (X,Y,Z)

# Helper regexes (they work for the default CalculiX output format)
re_node_disp = re.compile(r"^\s*U\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")
re_elem_stress = re.compile(r"^\s*S\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")
# The stress line contains: Sxx, Syy, Szz, Sxy, Syz, Sxz, SVM, PRESS, … (order may vary)
# We will capture the 7th column (von Mises) and the 8th column (pressure)

re_reaction = re.compile(r"^\s*RF\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")

# ------------------------------------------------------------------
# 3.  Scan the file ---------------------------------------------------
with dat_file.open() as f:
    for line in f:
        # ---- nodal displacements ------------------------------------
        m = re_node_disp.match(line)
        if m:
            nid = int(m.group(1))
            ux, uy, uz = map(float, m.groups()[1:])
            node_disp[nid] = (ux, uy, uz)
            continue

        # ---- element stresses (von Mises & pressure) ----------------
        m = re_elem_stress.match(line)
        if m:
            eid = int(m.group(1))
            # column 7 = von Mises, column 8 = pressure (if present)
            # The regex captures 10 numbers; we pick the right ones:
            cols = list(map(float, m.groups()[1:]))
            # In the default output order:
            # 0:Sxx 1:Syy 2:Szz 3:Sxy 4:Syz 5:Sxz 6:SVM 7:PRESS 8:… 9:…
            vm = cols[6]
            press = cols[7] if len(cols) > 7 else 0.0
            elem_vm[eid] = (vm, press)
            continue

        # ---- reaction forces ----------------------------------------
        m = re_reaction.match(line)
        if m:
            # The node number after RF is the constrained node (here node 1)
            # The three numbers are the reaction components in X,Y,Z
            rx, ry, rz = map(float, m.groups()[1:])
            reaction[0] += rx
            reaction[1] += ry
            reaction[2] += rz
            continue

# ------------------------------------------------------------------
# 4.  Determine the requested global quantities -----------------------
#    (maximum displacement magnitude, peak von Mises, reaction forces)
max_disp = -1.0
max_disp_node = None
for nid, (ux, uy, uz) in node_disp.items():
    mag = math.sqrt(ux*ux + uy*uy + uz*uz)
    if mag > max_disp:
        max_disp = mag
        max_disp_node = nid

max_vm = -1.0
max_vm_elem = None
for eid, (vm, _) in elem_vm.items():
    if vm > max_vm:
        max_vm = vm
        max_vm_elem = eid

# ------------------------------------------------------------------
# 5.  Print the three key results ------------------------------------
print("\n=== KEY RESULTS ===")
print(f"Maximum displacement magnitude : {max_disp:.6e} m  (Node {max_disp_node})")
print(f"Peak von Mises stress          : {max_vm:.3f} MPa  (Element {max_vm_elem})")
print(f"Reaction forces at support    : Rx = {reaction[0]:.3e} N, "
      f"Ry = {reaction[1]:.3e} N, Rz = {reaction[2]:.3e} N")
print("=" * 60)

# ------------------------------------------------------------------
# 6.  Write the CSV file ---------------------------------------------
csv_file = Path("summary.csv")
with csv_file.open("w", newline="") as csvf:
    writer = csv.writer(csvf)
    writer.writerow(["Node/Element ID", "Ux (m)", "Uy (m)", "Uz (m)",
                     "VonMises (MPa)", "Pressure (Pa)"])

    # Nodes – write displacement, leave stress columns empty (0)
    for nid, (ux, uy, uz) in sorted(node_disp.items()):
        writer.writerow([nid, ux, uy, uz, 0.0, 0.0])

    # Elements – write von Mises and pressure (if any)
    for eid, (vm, press) in sorted(elem_vm.items()):
        writer.writerow([eid, 0.0, 0.0, 0.0, vm, press])

print(f"\nCSV summary written to: {csv_file.resolve()}")
```

## 1️⃣  Python parser – `parse_beamtor.py`

Save this file in the same folder as `beamtor.dat` and `results.json`.

```python
#!/usr/bin/env python3
# --------------------------------------------------------------
# parse_beamtor.py
#   • checks results.json (job status)
#   • parses beamtor.dat for:
#        – nodal displacements (Ux, Uy, Uz)
#        – element von Mises stress (SVM) and pressure (P)
#        – reaction forces at the constrained support
#   • prints the three global quantities you asked for
#   • writes summary.csv (one line per node, one line per element)
# --------------------------------------------------------------

import json, csv, re, sys, math
from pathlib import Path

# ------------------------------------------------------------------
# 1.  Verify job status
# ------------------------------------------------------------------
json_file = Path("results.json")
if not json_file.is_file():
    sys.exit("ERROR: results.json not found")
with json_file.open() as f:
    jdata = json.load(f)

if jdata.get("status") != "success":
    sys.exit("ERROR: job did not finish successfully")

print(f"Solver: {jdata['solver']} – version {jdata['ccx_version']}")
print("-" * 60)

# ------------------------------------------------------------------
# 2.  Parse beamtor.dat
# ------------------------------------------------------------------
dat_file = Path("beamtor.dat")
if not dat_file.is_file():
    sys.exit("ERROR: beamtor.dat not found – run the job first")

# Containers
node_disp = {}          # node_id → (Ux, Uy, Uz)
elem_vm   = {}          # elem_id → (vonMises, pressure)
reaction  = [0.0, 0.0, 0.0]   # total Rx,Ry,Rz

# Regular expressions (default CalculiX output)
re_node = re.compile(r"^\s*U\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")
# Stress line contains: Sxx Syy Szz Sxy Syz Sxz SVM PRESS … (10 numbers)
re_elem = re.compile(r"^\s*S\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")
re_reac = re.compile(r"^\s*RF\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")

with dat_file.open() as f:
    for line in f:
        m = re_node.match(line)
        if m:
            nid = int(m.group(1))
            ux, uy, uz = map(float, m.groups()[1:])
            node_disp[nid] = (ux, uy, uz)
            continue

        m = re_elem.match(line)
        if m:
            eid = int(m.group(1))
            cols = list(map(float, m.groups()[1:]))   # 10 numbers
            vm   = cols[6]          # 7th column = von Mises
            press = cols[7] if len(cols) > 7 else 0.0   # 8th column = pressure (if printed)
            elem_vm[eid] = (vm, press)
            continue

        m = re_reac.match(line)
        if m:
            rx, ry, rz = map(float, m.groups()[1:])
            reaction[0] += rx
            reaction[1] += ry
            reaction[2] += rz
            continue

# ------------------------------------------------------------------
# 3.  Global quantities
# ------------------------------------------------------------------
max_disp = -1.0
max_disp_node = None
for nid, (ux, uy, uz) in node_disp.items():
    mag = math.sqrt(ux*ux + uy*uy + uz*uz)
    if mag > max_disp:
        max_disp = mag
        max_disp_node = nid

max_vm = -1.0
max_vm_elem = None
for eid, (vm, _) in elem_vm.items():
    if vm > max_vm:
        max_vm = vm
        max_vm_elem = eid

# ------------------------------------------------------------------
# 4.  Print results
# ------------------------------------------------------------------
print("\n=== KEY RESULTS ===")
print(f"Maximum displacement magnitude : {max_disp:.6e} m  (Node {max_disp_node})")
print(f"Peak von Mises stress          : {max_vm:.3f} MPa  (Element {max_vm_elem})")
print(f"Reaction forces at support    : Rx = {reaction[0]:.3e} N, "
      f"Ry = {reaction[1]:.3e} N, Rz = {reaction[2]:.3e} N")
print("=" * 60)

# ------------------------------------------------------------------
# 5.  Write summary.csv
# ------------------------------------------------------------------
csv_file = Path("summary.csv")
with csv_file.open("w", newline="") as cf:
    w = csv.writer(cf)
    w.writerow(["ID", "Ux (m)", "Uy (m)", "Uz (m)", "VonMises (MPa)", "Pressure (Pa)"])

    # Nodes – displacement only
    for nid, (ux, uy, uz) in sorted(node_disp.items()):
        w.writerow([nid, ux, uy, uz, 0.0, 0.0])

    # Elements – stress & pressure only
    for eid, (vm, press) in sorted(elem_vm.items()):
        w.writerow([eid, 0.0, 0.0, 0.0, vm, press])

print(f"\nCSV summary written to: {csv_file.resolve()}")
```

## 1️⃣  Python parser – `parse_beamtor.py`

Save the following file **exactly** as `parse_beamtor.py` in the folder that contains `beamtor.dat` and `results.json`.

```python
#!/usr/bin/env python3
# --------------------------------------------------------------
# parse_beamtor.py
#   • checks results.json (job status)
#   • parses beamtor.dat for:
#        – nodal displacements (Ux, Uy, Uz)
#        – element von Mises stress (SVM) and pressure (P)
#        – reaction forces at the constrained support
#   • prints the three global quantities you asked for
#   • writes summary.csv (one line per node, one line per element)
# --------------------------------------------------------------

import json, csv, re, sys, math
from pathlib import Path

# --------------------------------------------------------------
# 1. Verify job status (results.json)
# --------------------------------------------------------------
json_file = Path("results.json")
if not json_file.is_file():
    sys.exit("ERROR: results.json not found")
with json_file.open() as f:
    jdata = json.load(f)

if jdata.get("status") != "success":
    sys.exit("ERROR: job did not finish successfully")

print(f"Solver: {jdata['solver']} – version {jdata['ccx_version']}")
print("-" * 60)

# --------------------------------------------------------------
# 2. Parse beamtor.dat
# --------------------------------------------------------------
dat_file = Path("beamtor.dat")
if not dat_file.is_file():
    sys.exit("ERROR: beamtor.dat not found – run the job first")

node_disp = {}          # node_id → (Ux, Uy, Uz)
elem_vm   = {}          # elem_id → (vonMises, pressure)
reaction  = [0.0, 0.0, 0.0]   # total Rx,Ry,Rz

# Regular expressions (default CalculiX output)
re_node = re.compile(r"^\s*U\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")
# Stress line: Sxx Syy Szz Sxy Syz Sxz SVM PRESS … (10 numbers)
re_elem = re.compile(r"^\s*S\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")
re_reac = re.compile(r"^\s*RF\s+(\d+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)\s+([-\d.Ee+]+)")

with dat_file.open() as f:
    for line in f:
        m = re_node.match(line)
        if m:
            nid = int(m.group(1))
            ux, uy, uz = map(float, m.groups()[1:])
            node_disp[nid] = (ux, uy, uz)
            continue

        m = re_elem.match(line)
        if m:
            eid = int(m.group(1))
            cols = list(map(float, m.groups()[1:]))   # 10 numbers
            vm   = cols[6]          # 7th column = von Mises
            press = cols[7] if len(cols) > 7 else 0.0   # 8th column = pressure (if printed)
            elem_vm[eid] = (vm, press)
            continue

        m = re_reac.match(line)
        if m:
            rx, ry, rz = map(float, m.groups()[1:])
            reaction[0] += rx
            reaction[1] += ry
            reaction[2] += rz
            continue

# --------------------------------------------------------------
# 3. Global quantities (taken from the **last** step in the .dat file)
# --------------------------------------------------------------
max_disp = -1.0
max_disp_node = None
for nid, (ux, uy, uz) in node_disp.items():
    mag = math.sqrt(ux*ux + uy*uy + uz*uz)
    if mag > max_disp:
        max_disp = mag
        max_disp_node = nid

max_vm = -1.0
max_vm_elem = None
for eid, (vm, _) in elem_vm.items():
    if vm > max_vm:
        max_vm = vm
        max_vm_elem = eid

# --------------------------------------------------------------
# 4. Print results
# --------------------------------------------------------------
print("\n=== KEY RESULTS ===")
print(f"Maximum displacement magnitude : {max_disp:.6e} m  (Node {max_disp_node})")
print(f"Peak von Mises stress          : {max_vm:.3f} MPa  (Element {max_vm_elem})")
print(f"Reaction forces at support    : Rx = {reaction[0]:.3e} N, "
      f"Ry = {reaction[1]:.3e} N, Rz = {reaction[2]:.3e} N")
print("=" * 60)

# --------------------------------------------------------------
# 5. Write summary.csv
# --------------------------------------------------------------
csv_file = Path("summary.csv")
with csv_file.open("w", newline="") as cf:
    w = csv.writer(cf)
    w.writerow(["ID", "Ux (m)", "Uy (m)", "Uz (m)", "VonMises (MPa)", "Pressure (Pa)"])

    # Nodes – displacement only
    for nid, (ux, uy, uz) in sorted(node_disp.items()):
        w.writerow([nid, ux, uy, uz, 0.0, 0.0])

    # Elements – stress & pressure only
    for eid, (vm, press) in sorted(elem_vm.items()):
        w.writerow([eid, 0.0, 0.0, 0.0, vm, press])

print(f"\nCSV summary written to: {csv_file.resolve()}")
```

## 3. `parse_beamtor.py` – JSON parser & CSV writer

```python
#!/usr/bin/env python3
"""
parse_beamtor.py
----------------
* Reads the CalculiX JSON export (results.json)
* Extracts:
    - Global max displacement (magnitude)
    - Global reaction force (sum of RF on the constrained nodes)
    - Global max von Mises stress
* Writes a CSV (summary.csv) with:
    - NodeID, X, Y, Z, Ux, Uy, Uz
    - ElemID, VonMises, Pressure (if present)
"""

import json
import csv
import math
import sys
from pathlib import Path

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def vec_norm(v):
    """Euclidean norm of a 3‑component vector."""
    return math.sqrt(v[0]**2 + v[1]**2 + v[2]**2)

def load_json(filepath):
    """Load the CalculiX JSON file."""
    try:
        with open(filepath, "r") as f:
            return json.load(f)
    except FileNotFoundError:
        sys.exit(f"❌  File not found: {filepath}")
    except json.JSONDecodeError as e:
        sys.exit(f"❌  JSON decode error: {e}")

def find_last_step(data):
    """CalculiX writes a list of steps; the last one is the one we need."""
    if not isinstance(data, list) or not data:
        sys.exit("❌  Unexpected JSON structure – expected a non‑empty list.")
    return data[-1]   # last step

def extract_global_quantities(step):
    """Return (max_disp, total_RF, max_vM)."""
    # ---- Displacements -------------------------------------------------
    max_disp = 0.0
    for node in step.get("U", []):          # U = nodal displacement vectors
        disp = vec_norm(node["value"])
        if disp > max_disp:
            max_disp = disp

    # ---- Reaction forces ------------------------------------------------
    total_rf = 0.0
    for node in step.get("RF", []):         # RF = reaction forces
        total_rf += sum(node["value"])      # sum of Fx+Fy+Fz (signs already included)

    # ---- Stresses -------------------------------------------------------
    max_vm = 0.0
    for elem in step.get("S", []):          # S = element stresses (von Mises first)
        vm = elem["value"][0]               # first entry is von Mises
        if vm > max_vm:
            max_vm = vm

    return max_disp, total_rf, max_vm

def write_summary_csv(step, csv_path):
    """Create summary.csv with nodal displacements + element stresses."""
    header = [
        "NodeID", "X", "Y", "Z", "Ux", "Uy", "Uz",
        "ElemID", "VonMises", "Pressure"
    ]

    # Build dictionaries for fast lookup
    disp_dict = {n["node"]: n["value"] for n in step.get("U", [])}
    coord_dict = {n["node"]: n["coord"] for n in step.get("N", [])}
    stress_dict = {}
    for e in step.get("S", []):
        # e["value"] = [vonMises, s11, s22, s33, s12, s13, s23, ...]
        stress_dict[e["elem"]] = {
            "vonMises": e["value"][0],
            "pressure": e["value"][1] if len(e["value"]) > 1 else None
        }

    # Write rows – one row per node (missing element data will be left blank)
    with open(csv_path, "w", newline="") as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(header)

        for nid, coord in coord_dict.items():
            ux, uy, uz = disp_dict.get(nid, (0.0, 0.0, 0.0))
            # Find an element that contains this node (optional – we just pick the first)
            elem_id = ""
            vm = ""
            press = ""
            for eid, vals in stress_dict.items():
                # In a real model you would map nodes→elements; here we just dump the first
                elem_id = eid
                vm = vals["vonMises"]
                press = vals["pressure"] if vals["pressure"] is not None else ""
                break
            writer.writerow([nid, *coord, ux, uy, uz, elem_id, vm, press])

# ------------------------------------------------------------
# Main
# ------------------------------------------------------------
def main():
    json_file = Path("results.json")
    if not json_file.is_file():
        sys.exit("❌  results.json not found – run CalculiX with the *OUTPUT, JSON option.")

    data = load_json(json_file)
    last_step = find_last_step(data)

    # 1️⃣ Global quantities
    max_disp, total_rf, max_vm = extract_global_quantities(last_step)
    print("\n=== Global results (last step) ===")
    print(f"Maximum nodal displacement   : {max_disp:.6e}  m")
    print(f"Total reaction force (sum)   : {total_rf:.6e}  N")
    print(f"Maximum von Mises stress     : {max_vm:.6e}  Pa\n")

    # 2️⃣ CSV summary
    csv_path = Path("summary.csv")
    write_summary_csv(last_step, csv_path)
    print(f"✅  summary.csv written to {csv_path.resolve()}\n")

if __name__ == "__main__":
    main()
```

### How it works
| Step | What the script does |
|------|----------------------|
| **Load** | Reads `results.json` (the JSON export you obtain with `*OUTPUT, JSON`). |
| **Select step** | Takes the **last** step in the list – this corresponds to the 2 MPa pressure step. |
| **Global quantities** | *Maximum displacement* = Euclidean norm of the largest nodal `U` vector.<br>*Total reaction force* = Σ RF components (signs already accounted for).<br>*Maximum von Mises* = largest `S` entry (first value in each stress vector). |
| **CSV** | Writes one line per node: coordinates, displacement components, and (for the first element that contains the node) its von Mises and pressure. The format is deliberately simple so you can post‑process it with Excel, pandas, etc. |

> **Tip** – If you ever need the *first* step (the 1 MPa case) simply replace `find_last_step(data)` with `data[0]`.

| **I need the 1 MPa step instead of the 2 MPa step.** | Edit `parse_beamtor.py`: replace `last_step = find_last_step(data)` with `last_step = data[0]`. The same change works for the ParaView script – set `result = JSONReader(..., Step=0)` (or use the `*STEP` selector in ParaView). |

| **I get “KeyError: 'U'”** | The JSON export was probably generated without the `U` field. Re‑run CalculiX with `*OUTPUT, FIELD, VARIABLE=U, RF, S, N`. |
| **I want the script to run automatically after each CalculiX run.** | Add a post‑process line to the end of `beamtor.inp` using the `*EL FILE` or a shell wrapper; the `run_all.sh` script already does that. |

## 2. `parse_beamtor.py` – JSON parser & CSV writer

```python
#!/usr/bin/env python3
"""
parse_beamtor.py
----------------
* Reads the CalculiX JSON export (results.json)
* Extracts:
    - Global max displacement (magnitude)
    - Global reaction force (sum of RF on the constrained nodes)
    - Global max von Mises stress
* Writes a CSV (summary.csv) with:
    - NodeID, X, Y, Z, Ux, Uy, Uz
    - ElemID, VonMises, Pressure (if present)
"""

import json, csv, math, sys
from pathlib import Path

def vec_norm(v):
    return math.sqrt(v[0]**2 + v[1]**2 + v[2]**2)

def load_json(fp):
    try:
        with open(fp, "r") as f:
            return json.load(f)
    except Exception as e:
        sys.exit(f"❌  Cannot read {fp}: {e}")

def find_last_step(data):
    if not isinstance(data, list) or not data:
        sys.exit("❌  Unexpected JSON structure.")
    return data[-1]          # last step = 2 MPa case

def extract_globals(step):
    # max displacement
    max_disp = max(vec_norm(n["value"]) for n in step.get("U", []))
    # total reaction force (sum of all components)
    total_rf = sum(sum(node["value"]) for node in step.get("RF", []))
    # max von Mises stress (first entry of each stress vector)
    max_vm = max(elem["value"][0] for elem in step.get("S", []))
    return max_disp, total_rf, max_vm

def write_summary(step, csv_path):
    header = ["NodeID","X","Y","Z","Ux","Uy","Uz","ElemID","VonMises","Pressure"]
    # build lookup tables
    disp = {n["node"]: n["value"] for n in step.get("U", [])}
    coord = {n["node"]: n["coord"] for n in step.get("N", [])}
    stress = {}
    for e in step.get("S", []):
        stress[e["elem"]] = {
            "vonMises": e["value"][0],
            "pressure": e["value"][1] if len(e["value"])>1 else ""
        }

    with open(csv_path, "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(header)
        for nid, xyz in coord.items():
            ux, uy, uz = disp.get(nid, (0.0,0.0,0.0))
            # pick the first element (just to have something in the row)
            eid, vm, press = "", "", ""
            for eid_key, val in stress.items():
                eid, vm, press = eid_key, val["vonMises"], val["pressure"]
                break
            w.writerow([nid, *xyz, ux, uy, uz, eid, vm, press])

def main():
    json_file = Path("results.json")
    if not json_file.is_file():
        sys.exit("❌  results.json not found – run CalculiX with *OUTPUT, JSON.")

    data = load_json(json_file)
    step = find_last_step(data)

    max_disp, total_rf, max_vm = extract_globals(step)
    print("\n=== Global results (last step) ===")
    print(f"Maximum nodal displacement   : {max_disp:.6e}  m")
    print(f"Total reaction force (sum)   : {total_rf:.6e}  N")
    print(f"Maximum von Mises stress     : {max_vm:.6e}  Pa\n")

    csv_path = Path("summary.csv")
    write_summary(step, csv_path)
    print(f"✅  summary.csv written to {csv_path.resolve()}\n")

if __name__ == "__main__":
    main()
```

**What it does**

1. Loads `results.json`.  
2. Takes the **last** step (the 2 MPa pressure case).  
3. Computes the three global quantities and prints them.  
4. Writes `summary.csv` with one row per node (coordinates + displacements) and the von Mises/pressure of the first element that contains the node.  

> **If you need the 1 MPa step instead**, replace  
> `step = find_last_step(data)`  
> with `step = data[0]`.

```python
#!/usr/bin/env python3
"""
parse_beamtor.py
----------------
* Reads the CalculiX JSON export (results.json)
* Extracts:
    - Global max displacement (magnitude)
    - Global reaction force (sum of RF on the constrained nodes)
    - Global max von Mises stress
* Writes a CSV (summary.csv) with:
    - NodeID, X, Y, Z, Ux, Uy, Uz
    - ElemID, VonMises, Pressure (if present)
"""

import json
import csv
import math
import sys
from pathlib import Path

# ----------------------------------------------------------------------
def vec_norm(v):
    """Euclidean norm of a 3‑component vector."""
    return math.sqrt(v[0]**2 + v[1]**2 + v[2]**2)

# ----------------------------------------------------------------------
def load_json(fp: Path):
    """Read the JSON file produced by CalculiX."""
    try:
        with fp.open("r") as f:
            return json.load(f)
    except Exception as exc:
        sys.exit(f"❌  Cannot read {fp}: {exc}")

# ----------------------------------------------------------------------
def find_last_step(data):
    """
    CalculiX writes a *list* – one entry per step.
    The last entry corresponds to the 2 MPa pressure step.
    """
    if not isinstance(data, list) or not data:
        sys.exit("❌  Unexpected JSON structure – expected a non‑empty list.")
    return data[-1]

# ----------------------------------------------------------------------
def extract_globals(step):
    """
    Returns three scalars:
        max_disp   – max nodal displacement magnitude (m)
        total_rf   – sum of all reaction‑force components (N)
        max_vm     – max von Mises stress (Pa)
    """
    # ---- max displacement ------------------------------------------------
    max_disp = max(vec_norm(node["value"]) for node in step.get("U", []))

    # ---- total reaction force --------------------------------------------
    total_rf = sum(sum(node["value"]) for node in step.get("RF", []))

    # ---- max von Mises stress ---------------------------------------------
    # In the JSON export the stress vector is [vonMises, pressure, ...]
    max_vm = max(elem["value"][0] for elem in step.get("S", []))

    return max_disp, total_rf, max_vm

# ----------------------------------------------------------------------
def write_summary(step, csv_path: Path):
    """
    Creates `summary.csv` with the columns:
        NodeID, X, Y, Z, Ux, Uy, Uz, ElemID, VonMises, Pressure
    """
    header = ["NodeID", "X", "Y", "Z", "Ux", "Uy", "Uz",
              "ElemID", "VonMises", "Pressure"]

    # Build lookup tables for fast access
    disp   = {n["node"]: n["value"] for n in step.get("U", [])}
    coord  = {n["node"]: n["coord"] for n in step.get("N", [])}
    stress = {}
    for e in step.get("S", []):
        # e["value"] = [vonMises, pressure, ...] – keep both if present
        stress[e["elem"]] = {
            "vonMises": e["value"][0],
            "pressure": e["value"][1] if len(e["value"]) > 1 else ""
        }

    # Write CSV
    with csv_path.open("w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(header)

        # One row per node – we also attach the *first* element that contains a stress
        # (just to have a non‑empty ElemID column). If you need a strict node‑to‑element
        # mapping you can build it from the element connectivity.
        first_elem_id = next(iter(stress), "")

        for nid, xyz in coord.items():
            ux, uy, uz = disp.get(nid, (0.0, 0.0, 0.0))
            vm   = stress.get(first_elem_id, {}).get("vonMises", "")
            pres = stress.get(first_elem_id, {}).get("pressure", "")
            writer.writerow([nid, *xyz, ux, uy, uz,
                             first_elem_id, vm, pres])

# ----------------------------------------------------------------------
def main():
    json_file = Path("results.json")
    if not json_file.is_file():
        sys.exit("❌  results.json not found – run CalculiX with *OUTPUT, JSON.")

    data = load_json(json_file)
    step = find_last_step(data)

    max_disp, total_rf, max_vm = extract_globals(step)

    print("\n=== Global results (last step) ===")
    print(f"Maximum nodal displacement   : {max_disp:.6e}  m")
    print(f"Total reaction force (sum)   : {total_rf:.6e}  N")
    print(f"Maximum von Mises stress     : {max_vm:.6e}  Pa\n")

    csv_path = Path("summary.csv")
    write_summary(step, csv_path)
    print(f"✅  summary.csv written to {csv_path.resolve()}\n")

if __name__ == "__main__":
    main()
```

## 2️⃣ `parse_beamtor.py` – JSON parser & CSV writer

```python
#!/usr/bin/env python3
"""
parse_beamtor.py
----------------
* Reads the CalculiX JSON export (results.json)
* Extracts:
    - Global max displacement (magnitude)
    - Global reaction force (sum of RF on the constrained nodes)
    - Global max von Mises stress
* Writes a CSV (summary.csv) with:
    - NodeID, X, Y, Z, Ux, Uy, Uz
    - ElemID, VonMises, Pressure (if present)
"""

import json
import csv
import math
import sys
from pathlib import Path

# --------------------------------------------------------------
def vec_norm(v):
    """Euclidean norm of a 3‑component vector."""
    return math.sqrt(v[0]**2 + v[1]**2 + v[2]**2)

# --------------------------------------------------------------
def load_json(fp: Path):
    """Read the JSON file produced by CalculiX."""
    try:
        with fp.open("r") as f:
            return json.load(f)
    except Exception as exc:
        sys.exit(f"❌  Cannot read {fp}: {exc}")

# --------------------------------------------------------------
def find_last_step(data):
    """
    CalculiX writes a *list* – one entry per step.
    The last entry corresponds to the 2 MPa pressure step.
    """
    if not isinstance(data, list) or not data:
        sys.exit("❌  Unexpected JSON structure – expected a non‑empty list.")
    return data[-1]

# --------------------------------------------------------------
def extract_globals(step):
    """
    Returns three scalars:
        max_disp   – max nodal displacement magnitude (m)
        total_rf   – sum of all reaction‑force components (N)
        max_vm     – max von Mises stress (Pa)
    """
    max_disp = max(vec_norm(node["value"]) for node in step.get("U", []))
    total_rf = sum(sum(node["value"]) for node in step.get("RF", []))
    max_vm   = max(elem["value"][0] for elem in step.get("S", []))
    return max_disp, total_rf, max_vm

# --------------------------------------------------------------
def write_summary(step, csv_path: Path):
    """
    Creates `summary.csv` with the columns:
        NodeID, X, Y, Z, Ux, Uy, Uz, ElemID, VonMises, Pressure
    """
    header = ["NodeID", "X", "Y", "Z", "Ux", "Uy", "Uz",
              "ElemID", "VonMises", "Pressure"]

    # Build lookup tables for fast access
    disp   = {n["node"]: n["value"] for n in step.get("U", [])}
    coord  = {n["node"]: n["coord"] for n in step.get("N", [])}
    stress = {}
    for e in step.get("S", []):
        stress[e["elem"]] = {
            "vonMises": e["value"][0],
            "pressure": e["value"][1] if len(e["value"]) > 1 else ""
        }

    # Write CSV
    with csv_path.open("w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(header)

        # Attach the *first* element that has a stress entry (just to have a value)
        first_elem_id = next(iter(stress), "")

        for nid, xyz in coord.items():
            ux, uy, uz = disp.get(nid, (0.0, 0.0, 0.0))
            vm   = stress.get(first_elem_id, {}).get("vonMises", "")
            pres = stress.get(first_elem_id, {}).get("pressure", "")
            writer.writerow([nid, *xyz, ux, uy, uz,
                             first_elem_id, vm, pres])

# --------------------------------------------------------------
def main():
    json_file = Path("results.json")
    if not json_file.is_file():
        sys.exit("❌  results.json not found – run CalculiX with *OUTPUT, JSON.")

    data = load_json(json_file)
    step = find_last_step(data)

    max_disp, total_rf, max_vm = extract_globals(step)

    print("\n=== Global results (last step) ===")
    print(f"Maximum nodal displacement   : {max_disp:.6e}  m")
    print(f"Total reaction force (sum)   : {total_rf:.6e}  N")
    print(f"Maximum von Mises stress     : {max_vm:.6e}  Pa\n")

    csv_path = Path("summary.csv")
    write_summary(step, csv_path)
    print(f"✅  summary.csv written to {csv_path.resolve()}\n")

if __name__ == "__main__":
    main()
```

### How it works

| Step | What the code does |
|------|--------------------|
| **Load** | Reads `results.json`. |
| **Select step** | Takes the **last** step (the 2 MPa pressure case). |
| **Global quantities** | <ul><li>Maximum nodal displacement (vector magnitude).</li><li>Sum of all reaction‑force components (`RF`).</li><li>Maximum von Mises stress (`S[0]`).</li></ul> |
| **CSV** | Writes `summary.csv` with every node’s coordinates & displacement and the von Mises/pressure of the *first* element (just to have a value in the row). You can adapt the script to map each node to the elements that contain it if you need a stricter association. |

> **If you ever need the *first* step (the 1 MPa case)**, replace  
> `step = find_last_step(data)`  
> with `step = data[0]`.

## 3️⃣ `paraview_snapshot.py` – One‑click ParaView visualisation

```python
# paraview_snapshot.py
# -------------------------------------------------
# Loads results.json, warps the mesh by the displacement field U,
# colours the model by von Mises stress (S), and saves a PNG.
# -------------------------------------------------
from paraview.simple import *

# -------------------------------------------------
# 1. Load the JSON results (CalculiX writes a very simple JSON format)
# -------------------------------------------------
json_reader = JSONReader(FileName=['results.json'])

# -------------------------------------------------
# 2. Warp the geometry by the nodal displacement vector U
# -------------------------------------------------
warp = WarpByVector(Input=json_reader)
warp.Vectors = ['POINTS', 'U']      # name of the displacement array
warp.ScaleFactor = 1.0             # >1 for exaggerated deformation if you like

# -------------------------------------------------
# 3. Colour by von Mises stress (first component of the S array)
# -------------------------------------------------
display = Show(warp, GetActiveView())
display.ColorArrayName = ['POINTS', 'S']   # S = [vonMises, pressure, …]

# Nice colour map
lut = GetColorTransferFunction('S')
lut.ApplyPreset('Cool to Warm', True)

# Show colour legend
display.SetScalarBarVisibility(GetActiveView(), True)

# -------------------------------------------------
# 4. Camera / view settings (feel free to tweak)
# -------------------------------------------------
view = GetActiveView()
view.ViewSize = [1920, 1080]               # HD image
view.CameraPosition = [0.2, -0.3, 0.15]
view.CameraFocalPoint = [0.0, 0.0, 0.0]
view.CameraViewUp = [0.0, 0.0, 1.0]

# -------------------------------------------------
# 5. Render and write PNG
# -------------------------------------------------
Render()
SaveScreenshot('result.png', view, ImageResolution=[1920, 1080])
print("✅  Screenshot saved as result.png")
```
