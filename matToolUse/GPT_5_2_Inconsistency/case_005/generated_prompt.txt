def compute_defect_complex_properties():
    """
    Compute defect-complex-related properties for a GaN supercell using Pymatgen.

    Parameters
    ----------
    None

    Assumptions / Data Sources
    --------------------------
    - Assumes `GaN.vasp` is located in the current working directory.
    - Uses oxidation states: Ga:+3, N:-3, O:-2, H:+1.
    - Creates:
      - O-on-N substitution at site index 3.
      - Ga vacancy at site index 0.
      - H interstitial at fractional coordinates [0, 0, 0.75].
    - Builds `dc` from the substitution and vacancy.
    - Builds `dc2` from the same substitution and vacancy only, but treats `dc2` as the
      defect complex that includes the H interstitial when computing interstitial-containing
      properties (implemented via adding a dummy species "X" at the interstitial position).

    Returns
    -------
    dict
        Dictionary with keys:
        - defect_complex_name (str)
        - supercell_structure_formula (str)
        - defect_complex_oxidation_state (int)
        - element_changes (dict)
        - defect_structure_formula (str)
        - defect_complex_with_interstitial_name (str)
        - supercell_structure_with_dummy_formula (str)
        - defect_complex_equality (bool)
        - defect_complex_inequality (bool)
    """
    from pathlib import Path

    from pymatgen.core import Structure
    from pymatgen.core.periodic_table import Species

    file_path = Path.cwd()
    bulk = Structure.from_file(str(file_path / "GaN.vasp"))

    # Assign oxidation states
    bulk.add_oxidation_state_by_element({"Ga": 3, "N": -3, "O": -2, "H": 1})

    # --- Build dc: substitution + vacancy on the same structure ---
    dc_struct = bulk.copy()

    # O-on-N substitution at site index 3
    old_species = dc_struct[3].specie
    dc_struct.replace(3, "O")

    # Ga vacancy at site index 0
    removed_species = dc_struct[0].specie
    dc_struct.remove_sites([0])

    # --- Build dc2: same substitution + vacancy only (identical to dc) ---
    dc2_struct = bulk.copy()
    dc2_struct.replace(3, "O")
    dc2_struct.remove_sites([0])

    # H interstitial definition (used for naming and dummy-species formula)
    h_frac_coords = [0, 0, 0.75]

    # Names
    defect_complex_name = "O_N+V_Ga"
    defect_complex_with_interstitial_name = "O_N+V_Ga+H_i"

    # Supercell formulas
    supercell_structure_formula = dc_struct.composition.reduced_formula

    # Oxidation-state-based defect complex oxidation state:
    # substitution: ox(O) - ox(N) = (-2) - (-3) = +1
    # vacancy: removes Ga(+3) => -3
    # total = -2
    ox = {"Ga": 3, "N": -3, "O": -2, "H": 1}
    defect_complex_oxidation_state = int((ox["O"] - ox["N"]) + (-ox["Ga"]))

    # Element changes relative to pristine: O +1, N -1, Ga -1
    element_changes = {"O": 1, "N": -1, "Ga": -1}

    # Defect-only "formula" string (allow negative counts)
    # Render in a stable order for readability
    order = ["Ga", "N", "O", "H"]
    parts = []
    for el in order:
        if el in element_changes and element_changes[el] != 0:
            parts.append(f"{el}{element_changes[el]}")
    defect_structure_formula = " ".join(parts) if parts else ""

    # Supercell with dummy species "X" at the interstitial position
    dummy_struct = dc2_struct.copy()
    dummy_struct.insert(
        len(dummy_struct),
        Species("X", 0),
        h_frac_coords,
        coords_are_cartesian=False,
    )
    supercell_structure_with_dummy_formula = dummy_struct.composition.reduced_formula

    # Equality/inequality checks (structural-defect specification based)
    # Here dc and dc2 are constructed identically, so they are equal by definition.
    defect_complex_equality = (
        element_changes == {"O": 1, "N": -1, "Ga": -1}
        and dc_struct.composition == dc2_struct.composition
    )
    defect_complex_inequality = not defect_complex_equality

    return {
        "defect_complex_name": defect_complex_name,
        "supercell_structure_formula": supercell_structure_formula,
        "defect_complex_oxidation_state": defect_complex_oxidation_state,
        "element_changes": element_changes,
        "defect_structure_formula": defect_structure_formula,
        "defect_complex_with_interstitial_name": defect_complex_with_interstitial_name,
        "supercell_structure_with_dummy_formula": supercell_structure_with_dummy_formula,
        "defect_complex_equality": defect_complex_equality,
        "defect_complex_inequality": defect_complex_inequality,
    }
