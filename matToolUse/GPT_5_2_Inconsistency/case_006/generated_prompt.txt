from __future__ import annotations

from pathlib import Path
from typing import Any, Dict

from pymatgen.core import Structure
from pymatgen.analysis.defects.core import NamedDefect


def compute_named_defect_properties() -> Dict[str, Any]:
    """
    Compute NamedDefect-related properties using pymatgen and pymatgen-analysis-defects.

    Parameters
    ----------
    None

    Reads
    -----
    Bulk structure (gzipped VASP CONTCAR):
        Path.cwd() / "Mg_Ga/bulk_sc/CONTCAR.gz"
    Defect structure (gzipped VASP CONTCAR):
        Path.cwd() / "Mg_Ga/q=0/CONTCAR.gz"

    Returns
    -------
    Dict[str, Any]
        A dictionary with the following keys:
        - "element_changes" (dict): Change in the number of atoms of each element due to the defect.
        - "defect_string_representation" (str): String representation of the NamedDefect.
        - "defect_equality" (bool): Whether the structure-derived defect equals the reference Ga vacancy.
        - "defect_inequality" (bool): Whether the structure-derived defect is not equal to the reference Ga vacancy.
    """
    file_path = Path.cwd()
    bulk_path = file_path / "Mg_Ga/bulk_sc/CONTCAR.gz"
    defect_path = file_path / "Mg_Ga/q=0/CONTCAR.gz"

    bulk = Structure.from_file(bulk_path)
    defect = Structure.from_file(defect_path)

    # Structure-derived NamedDefect for the Mg_Ga defect structure
    nd0 = NamedDefect(structure=defect, bulk_structure=bulk, name="Mg_Ga")

    # Manual references with stated element changes
    bulk_formula = bulk.composition.reduced_formula
    v_Ga = NamedDefect(name="v_Ga", bulk_formula=bulk_formula, element_changes={"Ga": -1})
    Mg_Ga = NamedDefect(name="Mg_Ga", bulk_formula=bulk_formula, element_changes={"Mg": +1, "Ga": -1})

    results: Dict[str, Any] = {
        "element_changes": nd0.element_changes,
        "defect_string_representation": str(nd0),
        "defect_equality": (nd0 == v_Ga),
        "defect_inequality": (nd0 != v_Ga),
    }

    # Keep references available for debugging/inspection if desired (not required by output spec)
    _ = Mg_Ga

    return results
