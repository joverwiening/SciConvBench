Normalize the correction result to a float correction energy in eV, supporting return types of float, dict with common energy keys, or objects with an energy attribute.

def compute_freysoldt_correction_energy():
    """
    Compute the Freysoldt correction energy for the Mg_Ga defect set using Pymatgen.

    The function scans `<file_path>/Mg_Ga/` for immediate subdirectories, each expected to
    contain `vasprun.xml.gz` and `LOCPOT.gz`. It loads all such subfolders into a dictionary
    keyed by subfolder name.

    Because the requirement does not specify how to choose the bulk/pristine reference or
    which defect subfolder to correct, the function uses best-practice assumptions:
    - Bulk/pristine reference is identified by subfolder name containing (case-insensitive)
      one of: "bulk", "pristine", "host", "perfect". If none or multiple match, an error is raised.
    - The defect to correct is chosen as the first non-bulk subfolder in sorted order.

    Parameters
    ----------
    None
        This function takes no required input arguments.

    External context required
    -------------------------
    file_path : str or pathlib.Path
        A global variable named `file_path` must exist in the caller's scope and point to the
        base directory containing the `Mg_Ga/` folder.

    Returns
    -------
    dict
        Dictionary with:
        - "freysoldt_correction_energy" (float): Freysoldt correction energy in eV.

    Raises
    ------
    RuntimeError
        If `file_path` is not defined, if bulk cannot be uniquely identified, if no defect
        folder exists, or if the correction cannot be computed with the available inputs.
    FileNotFoundError
        If expected directories or required files are missing.
    """
    from pathlib import Path

    from pymatgen.io.vasp import Locpot, Vasprun
    from pymatgen.analysis.defects.corrections.freysoldt import get_freysoldt_correction

    # Resolve base paths
    try:
        base = Path(file_path)  # noqa: F821 (expected to exist in caller scope)
    except NameError as exc:
        raise RuntimeError(
            "Expected a global variable `file_path` pointing to the base directory, "
            "but it is not defined."
        ) from exc

    mg_ga_dir = base / "Mg_Ga"
    if not mg_ga_dir.is_dir():
        raise FileNotFoundError(f"Expected directory not found: {mg_ga_dir}")

    # Discover immediate subdirectories
    subdirs = sorted([p for p in mg_ga_dir.iterdir() if p.is_dir()])
    if not subdirs:
        raise RuntimeError(f"No subdirectories found under: {mg_ga_dir}")

    # Load all calculations into a dict keyed by subfolder name
    calcs = {}
    for d in subdirs:
        vr_path = d / "vasprun.xml.gz"
        lp_path = d / "LOCPOT.gz"
        if not vr_path.is_file():
            raise FileNotFoundError(f"Missing required file: {vr_path}")
        if not lp_path.is_file():
            raise FileNotFoundError(f"Missing required file: {lp_path}")

        vasprun = Vasprun(str(vr_path), parse_potcar_file=False)
        locpot = Locpot.from_file(str(lp_path))

        calcs[d.name] = {
            "dir": d,
            "vasprun": vasprun,
            "locpot": locpot,
            "structure": vasprun.final_structure,
        }

    # Identify bulk/pristine reference by best-practice name matching
    bulk_tokens = ("bulk", "pristine", "host", "perfect")
    bulk_candidates = [
        name for name in calcs.keys()
        if any(tok in name.lower() for tok in bulk_tokens)
    ]
    if len(bulk_candidates) != 1:
        raise RuntimeError(
            "Could not uniquely identify the bulk/pristine reference folder under "
            f"{mg_ga_dir}.\n"
            f"Bulk candidates found: {bulk_candidates}\n"
            "Please ensure exactly one subfolder name contains one of: "
            f"{bulk_tokens} (case-insensitive)."
        )

    bulk_name = bulk_candidates[0]
    bulk = calcs[bulk_name]

    # Choose a defect calculation to correct (best-practice assumption)
    defect_names = sorted([n for n in calcs.keys() if n != bulk_name])
    if not defect_names:
        raise RuntimeError(
            f"Only bulk folder '{bulk_name}' found; no defect subfolders available to correct."
        )
    defect_name = defect_names[0]
    defect = calcs[defect_name]

    # Compute Freysoldt correction (handle API differences across pymatgen versions)
    try:
        corr = get_freysoldt_correction(
            defect_locpot=defect["locpot"],
            bulk_locpot=bulk["locpot"],
            defect_structure=defect["structure"],
            bulk_structure=bulk["structure"],
        )
    except TypeError:
        try:
            corr = get_freysoldt_correction(
                defect["locpot"],
                bulk["locpot"],
                defect["structure"],
                bulk["structure"],
            )
        except Exception as exc:
            raise RuntimeError(
                "Failed to compute Freysoldt correction with available inputs. "
                "Your pymatgen version may require additional parameters (e.g., dielectric, "
                "defect charge, defect site)."
            ) from exc
    except Exception as exc:
        raise RuntimeError(
            "Failed to compute Freysoldt correction. Ensure LOCPOTs are compatible and "
            "the bulk/defect cells are commensurate."
        ) from exc

    # Normalize return to float energy
    if isinstance(corr, (int, float)):
        energy = float(corr)
    elif isinstance(corr, dict):
        for k in ("energy", "correction_energy", "freysoldt_correction_energy", "E_corr"):
            if k in corr and isinstance(corr[k], (int, float)):
                energy = float(corr[k])
                break
        else:
            raise RuntimeError(f"Unrecognized correction dict format: keys={list(corr.keys())}")
    else:
        energy = getattr(corr, "energy", None)
        if energy is None:
            raise RuntimeError(f"Unrecognized correction return type: {type(corr)}")
        energy = float(energy)

    return {"freysoldt_correction_energy": energy}
