def compute_defect_formation_diagram_properties():
    """
    Compute basic properties derived from a formation energy (phase) diagram setup.

    Parameters
    ----------
    None
        This function takes no required input arguments.
        It assumes the input files are located in the current working directory (Path.cwd()):
        - GaN.vasp
        - stable_entries_Mg_Ga_N.json

    Returns
    -------
    dict
        A dictionary with the following keys:
        - "chempot_limits_count" (int): Number of chemical potential limits for the Ga-N system.
        - "defect_chemsys" (str): Chemical system string (alphabetical element symbols joined by hyphens).
        - "bulk_formula" (str): Reduced formula of the bulk structure.
    """
    from pathlib import Path

    from monty.serialization import loadfn
    from pymatgen.analysis.phase_diagram import PhaseDiagram
    from pymatgen.core import Structure

    file_path = Path.cwd()
    structure_path = file_path / "GaN.vasp"
    stable_entries_path = file_path / "stable_entries_Mg_Ga_N.json"

    if not structure_path.exists():
        raise FileNotFoundError(f"Missing structure file: {structure_path}")
    if not stable_entries_path.exists():
        raise FileNotFoundError(f"Missing stable entries file: {stable_entries_path}")

    bulk_structure = Structure.from_file(structure_path)
    stable_entries = loadfn(stable_entries_path)

    # Build phase diagram from provided stable entries.
    pd = PhaseDiagram(stable_entries)

    # Bulk formula.
    bulk_formula = bulk_structure.composition.reduced_formula

    # Defect chemical system from bulk composition.
    elems = sorted({el.symbol for el in bulk_structure.composition.elements})
    defect_chemsys = "-".join(elems)

    # Chemical potential limits count for Ga-N.
    # In pymatgen, chemical potential "limits" for a given chemical system can be obtained
    # by enumerating the facets (simplices) of the phase diagram that bound the stability
    # region of the target composition in that chemical subspace.
    #
    # Here we approximate the count as the number of unique facets in the Ga-N subdiagram.
    # This is consistent with interpreting "limits" as distinct equilibrium sets of phases
    # defining chemical potentials.
    ga_n_entries = [e for e in stable_entries if set([el.symbol for el in e.composition.elements]).issubset({"Ga", "N"})]
    ga_n_pd = PhaseDiagram(ga_n_entries)

    # Each facet corresponds to a distinct set of coexisting phases (a chemical potential limit).
    # For a binary system, facets correspond to tie-lines on the convex hull.
    chempot_limits_count = len(ga_n_pd.facets)

    return {
        "chempot_limits_count": int(chempot_limits_count),
        "defect_chemsys": str(defect_chemsys),
        "bulk_formula": str(bulk_formula),
    }
