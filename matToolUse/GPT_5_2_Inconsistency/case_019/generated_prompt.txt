from __future__ import annotations

from pathlib import Path
from typing import Any, Dict, List

from pymatgen.analysis.defects.ccd import HarmonicDefect, get_SRH_coefficient


def compute_srh_properties() -> Dict[str, Any]:
    """
    Compute SRH-related properties from local CCD/WSWQ inputs using Pymatgen.

    Parameters
    ----------
    None
        This function takes no required input arguments.

    Assumptions
    -----------
    The base directory `file_path` is assumed to be the current working directory.
    The WSWQ inputs are located under: v_Ga/ccd_0_-1/wswqs (relative to `file_path`).

    Returns
    -------
    dict
        A dictionary with the following keys:
        - "SRH_Coefficient" (List[float]): SRH coefficients evaluated at temperatures
          [300, 600, 900] K, in that order.
        - "RuntimeError_Check" (bool): True if a RuntimeError is raised when attempting
          to compute the SRH coefficient with an invalid defect-state configuration,
          otherwise False.
    """
    file_path = Path.cwd()
    wswq_dir = file_path / "v_Ga" / "ccd_0_-1" / "wswqs"

    # Build HarmonicDefect from the WSWQ directory.
    # Note: The exact constructor/helper may vary by pymatgen version; this uses the
    # canonical "from_dir" pattern used across pymatgen I/O-style objects.
    harmonic_defect = HarmonicDefect.from_dir(str(wswq_dir))

    temps: List[float] = [300.0, 600.0, 900.0]

    # Compute SRH coefficient for the valid configuration implied by the dataset.
    srh_vals = get_SRH_coefficient(harmonic_defect, temperatures=temps)
    srh_list = [float(x) for x in srh_vals]

    # RuntimeError check: attempt SRH with an invalid defect-state configuration.
    # We intentionally request a charge-state transition that is unlikely to exist
    # for the provided dataset (e.g., 0 -> +2), which should be inconsistent with
    # the available WSWQ charge states and trigger a RuntimeError.
    runtime_error_check = False
    try:
        _ = get_SRH_coefficient(
            harmonic_defect,
            temperatures=temps,
            initial_charge_state=0,
            final_charge_state=+2,
        )
    except RuntimeError:
        runtime_error_check = True

    return {
        "SRH_Coefficient": srh_list,
        "RuntimeError_Check": runtime_error_check,
    }
