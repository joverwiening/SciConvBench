from pathlib import Path
from typing import Dict

from pymatgen.core import Structure

def compute_defect_group_properties() -> Dict[str, bool]:

    - Mg_Ga charge-state directories (best-practice assumed names)
      Located at:
        file_path/Mg_Ga_-2/
        file_path/Mg_Ga_-1/
        file_path/Mg_Ga_0/
        file_path/Mg_Ga_+1/

    dict[str, bool]

    bulk_structure_path = base / "GaN.vasp"

    # Load bulk structure and bulk LOCPOT (used to demonstrate Freysoldt-associated inputs).

        base / "Mg_Ga_0",
