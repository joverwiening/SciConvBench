import os
import glob
from pymatgen.core import Structure
from pymatgen.analysis.defects.core import DefectEntry
from pymatgen.analysis.defects.parsers import DefectParser


def compute_defect_properties():
    """
    Compute defect properties for Mg_Ga defect entries.
    
    This function reads defect entries from the specified directory structure
    and checks that all entries in the Mg_Ga charge-state group share the
    same defect name.
    
    Parameters:
        None
        
    Returns:
        dict: A dictionary containing:
            - defect_name_consistency (bool): True if all defect entries in the
              Mg_Ga group have the same defect name, False otherwise
    """
    
    # Define file paths
    FILE_PATH = "."  # Base directory - adjust as needed for actual test case
    BULK_SC_PATH = os.path.join(FILE_PATH, "bulk_sc")
    BULK_STRUCTURE_PATH = os.path.join(FILE_PATH, "GaN.vasp")
    
    # Charge states for Mg_Ga defects
    CHARGE_STATES = [-2, -1, 0, 1]
    DEFECT_NAME = "Mg_Ga"
    
    # Load bulk structure
    bulk_structure = Structure.from_file(BULK_STRUCTURE_PATH)
    
    # Find bulk LOCPOT
    bulk_locpot_files = glob.glob(os.path.join(BULK_SC_PATH, "LOCPOT.gz"))
    if not bulk_locpot_files:
        return {"defect_name_consistency": False}
    bulk_locpot_path = bulk_locpot_files[0]
    
    # Collect defect entries
    defect_entries = []
    defect_names = []
    
    for charge in CHARGE_STATES:
        # Construct defect directory name
        defect_dir = f"{DEFECT_NAME}_{charge}"
        defect_path_full = os.path.join(FILE_PATH, defect_dir)
        
        # Find vasprun.xml.gz and LOCPOT.gz in defect directory
        vasprun_files = glob.glob(os.path.join(defect_path_full, "vasprun.xml.gz"))
        locpot_files = glob.glob(os.path.join(defect_path_full, "LOCPOT.gz"))
        
        # Skip if required files not found
        if not vasprun_files or not locpot_files:
            print(f"Warning: Missing files in {defect_dir}, skipping charge state {charge}")
            continue
        
        defect_vasprun_path = vasprun_files[0]
        defect_locpot_path = locpot_files[0]
        
        try:
            # Parse defect using DefectParser
            dp = DefectParser.from_paths(
                bulk_path=BULK_STRUCTURE_PATH,
                defect_path=defect_vasprun_path,
                bulk_locpot_path=bulk_locpot_path,
                defect_locpot_path=defect_locpot_path,
                charge=charge
            )
            
            # Extract defect entry and name
            defect_entry = dp.defect_entry
            defect_entries.append(defect_entry)
            defect_names.append(defect_entry.defect.name)
            
        except Exception as e:
            print(f"Error parsing defect in {defect_dir}: {e}")
            continue
    
    # Compute defect name consistency
    if len(defect_names) == 0:
        defect_name_consistency = False
    else:
        defect_name_consistency = len(set(defect_names)) == 1
    
    return {
        "defect_name_consistency": defect_name_consistency
    }
