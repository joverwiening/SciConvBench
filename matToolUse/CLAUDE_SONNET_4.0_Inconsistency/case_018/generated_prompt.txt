import numpy as np

class WSWQObject:
    def __init__(self, data):
        self.data = data

def compute_wswq_properties():
    """
    Computes WSWQ slopes for positive and negative distortion branches.
    
    This function generates fake WSWQ data consisting of three (3, 5) matrices
    with values 1, 0, and 1 respectively, wrapped in WSWQ-like objects.
    It then calculates slopes for both positive and negative distortion branches
    using the same WSWQ data but different distortion value assignments.
    
    Returns:
        dict: A dictionary containing:
            - 'wswq_slope_positive_distortion': numpy array of slopes when 
              distortion values are [0.5, 0.0, -0.5]
            - 'wswq_slope_negative_distortion': numpy array of slopes when
              distortion values are [-0.5, 0.0, 0.5]
    """
    
    # Generate fake WSWQ data
    matrix1 = np.ones((3, 5))
    matrix2 = np.zeros((3, 5))
    matrix3 = np.ones((3, 5))
    
    # Wrap matrices in WSWQ-like objects
    wswq1 = WSWQObject(matrix1)
    wswq2 = WSWQObject(matrix2)
    wswq3 = WSWQObject(matrix3)
    
    fake_wswqs = [wswq1, wswq2, wswq3]
    
    # Define distortion values for both branches
    positive_distortion = np.array([0.5, 0.0, -0.5])
    negative_distortion = np.array([-0.5, 0.0, 0.5])
    
    # Extract WSWQ data values
    wswq_values = np.array([wswq.data for wswq in fake_wswqs])
    
    # Calculate slopes for positive distortion branch
    wswq_slope_positive = np.zeros((3, 5))
    for i in range(3):
        for j in range(5):
            y_values = wswq_values[:, i, j]
            slope_pos = np.polyfit(positive_distortion, y_values, 1)[0]
            wswq_slope_positive[i, j] = slope_pos
    
    # Calculate slopes for negative distortion branch
    wswq_slope_negative = np.zeros((3, 5))
    for i in range(3):
        for j in range(5):
            y_values = wswq_values[:, i, j]
            slope_neg = np.polyfit(negative_distortion, y_values, 1)[0]
            wswq_slope_negative[i, j] = slope_neg
    
    return {
        'wswq_slope_positive_distortion': wswq_slope_positive,
        'wswq_slope_negative_distortion': wswq_slope_negative
    }
