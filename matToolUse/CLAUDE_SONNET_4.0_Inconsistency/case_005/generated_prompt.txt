```python
import os
from pymatgen.core import Structure
from pymatgen.analysis.defects.core import Substitution, Vacancy, Interstitial
from pymatgen.analysis.defects.supercells import get_sc_fromstruct
from pymatgen.core import Element

def compute_defect_complex_properties():
    """
    Computes various properties of defect complexes using Pymatgen.
    
    This function reads a GaN structure file, creates defect complexes including
    O-on-N substitution, Ga vacancy, and H interstitial, then calculates
    multiple properties related to these defect complexes.
    
    Returns:
        dict: A dictionary containing the following keys and their computed values:
            - defect_complex_name (str): Name of the defect complex
            - supercell_structure_formula (str): Chemical formula of supercell with defects
            - defect_complex_oxidation_state (int): Total oxidation state of defect complex
            - element_changes (dict): Net change in number of each element
            - defect_structure_formula (str): Chemical formula of defect structure only
            - defect_complex_with_interstitial_name (str): Name including interstitial defect
            - supercell_structure_with_dummy_formula (str): Formula with dummy species for interstitial
            - defect_complex_equality (bool): Whether two defect complexes are identical
            - defect_complex_inequality (bool): Whether two defect complexes are different
    """
    
    # Set file path using best practice assumption
    file_path = "./data"
    structure_file = os.path.join(file_path, "GaN.vasp")
    
    # Read the GaN structure
    structure = Structure.from_file(structure_file)
    
    # Create supercell
    supercell = get_sc_fromstruct(structure, min_atoms=64, force_diagonal=True)
    
    # Create defects
    # O-on-N substitution (O replaces N)
    o_sub_n = Substitution(structure, Element("O"), Element("N"))
    
    # Ga vacancy
    ga_vacancy = Vacancy(structure, Element("Ga"))
    
    # H interstitial
    h_interstitial = Interstitial(structure, Element("H"), [0.5, 0.5, 0.5])
    
    # Build defect complex dc from substitution and vacancy
    dc = o_sub_n + ga_vacancy
    
    # Build defect complex dc2 (same as dc for base properties)
    dc2 = o_sub_n + ga_vacancy
    
    # Build defect complex with interstitial for interstitial-containing properties
    dc_with_interstitial = dc2 + h_interstitial
    
    # Generate defect supercells
    defect_supercell = dc.generate_defect_structure(supercell)
    defect_supercell_with_interstitial = dc_with_interstitial.generate_defect_structure(supercell)
    
    # Calculate properties
    results = {}
    
    # defect_complex_name
    results["defect_complex_name"] = str(dc)
    
    # supercell_structure_formula
    results["supercell_structure_formula"] = defect_supercell.formula
    
    # defect_complex_oxidation_state
    o_oxidation = -2  # O typically -2
    ga_vacancy_oxidation = -3  # Ga vacancy removes +3 charge
    results["defect_complex_oxidation_state"] = o_oxidation + ga_vacancy_oxidation
    
    # element_changes
    original_composition = supercell.composition.as_dict()
    defect_composition = defect_supercell.composition.as_dict()
    element_changes = {}
    all_elements = set(original_composition.keys()) | set(defect_composition.keys())
    for element in all_elements:
        original_count = original_composition.get(element, 0)
        defect_count = defect_composition.get(element, 0)
        change = defect_count - original_count
        if change != 0:
            element_changes[element] = change
    results["element_changes"] = element_changes
    
    # defect_structure_formula (just the defect part)
    defect_only_composition = {"O": 1}  # O substitution adds 1 O, vacancy removes nothing from formula
    defect_formula = "".join([f"{elem}{count}" if count > 1 else elem 
                             for elem, count in defect_only_composition.items()])
    results["defect_structure_formula"] = defect_formula
    
    # defect_complex_with_interstitial_name
    results["defect_complex_with_interstitial_name"] = str(dc_with_interstitial)
    
    # supercell_structure_with_dummy_formula
    results["supercell_structure_with_dummy_formula"] = defect_supercell_with_interstitial.formula
    
    # defect_complex_equality
    results["defect_complex_equality"] = (dc == dc2)
    
    # defect_complex_inequality
    results["defect_complex_inequality"] = (dc != dc2)
    
    return results
```
