    """

    This function performs the following steps:
    1. Locate and read the defect‑supercell VASP output (vasprun.xml.gz) and
       electrostatic potential (LOCPOT.gz) from the ``Mg_Ga`` directory.

       from the ``bulk_sc`` directory.
    3. Load the bulk GaN structure from ``GaN.vasp``.
    4. Construct a ``DefectEntry`` for the Mg‑on‑Ga defect.
    5. Apply the Freysoldt correction using a dielectric constant of 14.
    6. Extract the potential‑alignment value from the correction and compare it
       with a manually computed alignment from the LOCPOT files to assess
       consistency.
    7. Compute the energy difference between the defect supercell and the bulk
       supercell.

    directory (or the directory supplied to ``file_path`` inside the function)
    contains the file layout described in the specification.

    Returns
    -------
    dict
        A dictionary with the following keys:
        - ``freysoldt_correction`` (float): The Freysoldt electrostatic correction.
        - ``potential_alignment_consistency`` (bool): True if the potential
          alignment from the Freysoldt analysis matches the alignment obtained

        - ``energy_difference`` (float): Energy(defect supercell) −

    import os
    import glob
    import gzip
    import numpy as np
    from pymatgen.io.vasp import Vasprun, Locpot
    from pymatgen.core import Structure
    from pymatgen.analysis.defects.core import DefectEntry
    from pymatgen.analysis.defects.corrections import FreysoldtCorrection

    # -------------------------------------------------------------------------
    # 1. Define the base directory (can be changed by the user if needed)
    # -------------------------------------------------------------------------
    file_path = os.getcwd()  # assume current working directory is the base

    # -------------------------------------------------------------------------
    # 2. Locate defect‑supercell files
    # -------------------------------------------------------------------------
    defect_vasprun_path = glob.glob(
        os.path.join(file_path, "Mg_Ga", "*", "vasprun.xml.gz")
    )
    if not defect_vasprun_path:
        raise FileNotFoundError("Defect vasprun.xml.gz not found in Mg_Ga subfolders.")
    defect_vasprun_path = defect_vasprun_path[0]

    defect_locpot_path = glob.glob(
        os.path.join(file_path, "Mg_Ga", "*", "LOCPOT.gz")
    )
    if not defect_locpot_path:
        raise FileNotFoundError("Defect LOCPOT.gz not found in Mg_Ga subfolders.")
    defect_locpot_path = defect_locpot_path[0]

    # -------------------------------------------------------------------------
    # 3. Locate bulk‑supercell files
    # -------------------------------------------------------------------------
    bulk_locpot_path = os.path.join(file_path, "bulk_sc", "LOCPOT.gz")
    if not os.path.isfile(bulk_locpot_path):
        raise FileNotFoundError("Bulk LOCPOT.gz not found in bulk_sc directory.")

    bulk_vasprun_path = os.path.join(file_path, "bulk_sc", "vasprun.xml.gz")
    if not os.path.isfile(bulk_vasprun_path):
        raise FileNotFoundError(
            "Bulk vasprun.xml.gz not found in bulk_sc directory. "
            "Energy of the bulk supercell is required for the energy difference."
        )

    # -------------------------------------------------------------------------
    # 4. Load structures and energies
    # -------------------------------------------------------------------------
    defect_vasprun = Vasprun(defect_vasprun_path, parse_potcar_file=False)
    defect_structure = defect_vasprun.final_structure
    defect_energy = defect_vasprun.final_energy  # eV

    bulk_vasprun = Vasprun(bulk_vasprun_path, parse_potcar_file=False)
    bulk_structure = bulk_vasprun.final_structure
    bulk_energy = bulk_vasprun.final_energy  # eV

    # Load the bulk GaN structure used to create the defect (for reference)
    bulk_ref_path = os.path.join(file_path, "GaN.vasp")
    if not os.path.isfile(bulk_ref_path):
        raise FileNotFoundError("Reference bulk structure GaN.vasp not found.")
    bulk_ref_structure = Structure.from_file(bulk_ref_path)

    # -------------------------------------------------------------------------
    # 5. Load electrostatic potentials

    # Helper to read gzipped LOCPOT files
    def read_locpot(gz_path):
        with gzip.open(gz_path, "rt") as f:
            return Locpot.from_file(f)

    defect_locpot = read_locpot(defect_locpot_path)
    bulk_locpot = read_locpot(bulk_locpot_path)

    # -------------------------------------------------------------------------
    # 6. Construct DefectEntry
    # -------------------------------------------------------------------------
    # The charge state is not specified; assume neutral (0) for demonstration.

        bulk_structure=bulk_ref_structure,

        sc_entry=None,  # not needed for the correction itself

    # Compute the planar‑averaged potentials and extract the alignment manually.

    def average_potential_far_from_defect(locpot):

        grid = locpot.potential
        shape = grid.shape
        # Determine the axis with the largest dimension
        axis = np.argmax(shape)
        # Slice the outer 10 % on both sides
        n = shape[axis]
        cut = int(0.1 * n)
        if axis == 0:
            region = np.concatenate([grid[:cut, :, :], grid[-cut:, :, :]], axis=0)

            region = np.concatenate([grid[:, :cut, :], grid[:, -cut:, :]], axis=1)

    manual_alignment = avg_defect_pot - avg_bulk_pot

    # Consistency is True if the two alignments differ by less than 1 meV.
    tolerance = 1e-3  # eV

    # -------------------------------------------------------------------------

    energy_difference = defect_energy - bulk_energy

    # -------------------------------------------------------------------------
    # 10. Assemble results
    # -------------------------------------------------------------------------
    results = {
        "freysoldt_correction": float(freysoldt_correction),
        "potential_alignment_consistency": bool(potential_alignment_consistency),
        "energy_difference": float(energy_difference),
    }

    return results
