def compute_chemical_potential_limits():
    """
    Compute the number of chemical‑potential limits in the Mg‑Ga defect formation‑energy
    diagram.

    Returns
    -------
    dict
        {"chemical_potential_limits_count": int}
        The integer is the number of distinct chemical‑potential limits considered.
    """
    import os
    import json
    from glob import glob

    from pymatgen.entries.computed_entries import ComputedEntry
    from pymatgen.analysis.defects.core import DefectEntry
    from pymatgen.analysis.defects.corrections import FreysoldtCorrection
    from pymatgen.analysis.defects.formation_energy import FormationEnergyDiagram
    from pymatgen.io.vasp import Vasprun, Locpot

    # ------------------------------------------------------------------
    # 1. Paths (relative to the base directory given by the env var FILE_PATH)
    # ------------------------------------------------------------------
    base_path = os.getenv("FILE_PATH", ".")
    stable_entries_path = os.path.join(base_path, "stable_entries_Mg_Ga_N.json")
    defect_dir = os.path.join(base_path, "Mg_Ga")
    bulk_vasprun_path = os.path.join(base_path, "bulk", "vasprun.xml.gz")
    bulk_locpot_path = os.path.join(base_path, "bulk", "LOCPOT.gz")

    # ------------------------------------------------------------------
    # 2. Load stable (elemental) entries
    # ------------------------------------------------------------------
    with open(stable_entries_path, "r") as f:
        stable_entries_json = json.load(f)
    stable_entries = [ComputedEntry.from_dict(d) for d in stable_entries_json]

    # ------------------------------------------------------------------
    # 3. Load bulk data (VBM and bulk reference for Freysoldt correction)
    # ------------------------------------------------------------------
    bulk_vasprun = Vasprun(bulk_vasprun_path, parse_potcar_file=False)
    bulk_vbm = bulk_vasprun.eigenvalue_band_properties[0]  # VBM
    bulk_entry = ComputedEntry(
        composition=bulk_vasprun.final_structure.composition,
        energy=bulk_vasprun.final_energy,
        parameters=bulk_vasprun.parameters,
        data={"run_type": "Bulk"},
    )
    bulk_locpot = Locpot.from_file(bulk_locpot_path)

    # ------------------------------------------------------------------
    # 4. Load defect entries, build DefectEntry objects for charges [-2, -1, 0, 1]
    #    and apply Freysoldt correction (dielectric = 14)
    # ------------------------------------------------------------------
    defect_entries = []
    # each subfolder in Mg_Ga corresponds to a defect (e.g. Mg_Ga/defect_Mg_0, etc.)
    subfolders = [d for d in glob(os.path.join(defect_dir, "*")) if os.path.isdir(d)]
    for sub in subfolders:
        # expected files inside each subfolder
        vasprun_path = os.path.join(sub, "vasprun.xml.gz")
        locpot_path = os.path.join(sub, "LOCPOT.gz")
        if not (os.path.isfile(vasprun_path) and os.path.isfile(locpot_path)):
            continue

        vr = Vasprun(vasprun_path, parse_potcar_file=False)
        locpot = Locpot.from_file(locpot_path)

        # charge is stored in the vasprun (or infer from folder name)
        charge = vr.parameters.get("NELECT", 0) - vr.final_structure.composition.total_electrons
        # restrict to the required charge states
        if charge not in [-2, -1, 0, 1]:
            continue

        defect_entry = DefectEntry(
            defect_structure=vr.final_structure,
            charge=charge,
            bulk_entry=bulk_entry,
            vbm=bulk_vbm,
        )

        # Freysoldt correction
        corr = FreysoldtCorrection(
            defect_entry=defect_entry,
            bulk_locpot=bulk_locpot,
            defect_locpot=locpot,
            dielectric=14,
        )
        defect_entry.correction = corr.get_correction()
        defect_entries.append(defect_entry)

    # ------------------------------------------------------------------
    # 5. Build the formation‑energy diagram
    # ------------------------------------------------------------------
    fed = FormationEnergyDiagram(
        defect_entries=defect_entries,
        chempots=stable_entries,
        bulk_entry=bulk_entry,
    )

    # ------------------------------------------------------------------
    # 6. Count chemical‑potential limits
    # ------------------------------------------------------------------
    chempot_limits = fed.get_chemical_potential_limits()
    limits_count = len(chempot_limits)

    # ------------------------------------------------------------------
    # 7. Return result
    # ------------------------------------------------------------------
    return {"chemical_potential_limits_count": limits_count}
