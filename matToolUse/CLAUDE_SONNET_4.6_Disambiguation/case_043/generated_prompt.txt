echo "=== Workflow complete ==="

File: extract_relaxed.py

import sys
import re

def extract_relaxed(outfile):
    with open(outfile) as f:
        content = f.read()

    cell_blocks = re.findall(
        r'CELL_PARAMETERS\s*\(angstrom\)\s*\n((?:\s*[-\d.]+\s+[-\d.]+\s+[-\d.]+\s*\n){3})',
        content)
    pos_blocks = re.findall(
        r'ATOMIC_POSITIONS\s*\(crystal\)\s*\n((?:\s*\w+\s+[-\d.]+\s+[-\d.]+\s+[-\d.]+\s*\n){4})',
        content)

    if not cell_blocks or not pos_blocks:
        print("ERROR: Could not find relaxed cell or positions in output.")
        sys.exit(1)

    cell_lines = cell_blocks[-1].strip().split('\n')
    pos_lines  = pos_blocks[-1].strip().split('\n')

    cell_placeholder = '\n'.join(
        ['  RELAXED_A{}_PLACEHOLDER'.format(i+1) for i in range(3)])
    pos_placeholder  = '  RELAXED_POSITIONS_PLACEHOLDER'

    cell_replacement = '\n'.join(['  ' + l.strip() for l in cell_lines])
    pos_replacement  = '\n'.join(['  ' + l.strip() for l in pos_lines])

    for fname in ['aln.scf.in', 'aln.nscf.in', 'aln.bands.in']:
        with open(fname) as f:
            txt = f.read()
        for i, line in enumerate(cell_lines):
            txt = txt.replace(
                '  RELAXED_A{}_PLACEHOLDER'.format(i+1),
                '  ' + line.strip())
        new_pos = '\n'.join(['  ' + l.strip() for l in pos_lines])
        txt = txt.replace('  RELAXED_POSITIONS_PLACEHOLDER', new_pos)
        with open(fname, 'w') as f:
            f.write(txt)
        print("Updated {}".format(fname))

if __name__ == '__main__':
    extract_relaxed(sys.argv[1])
