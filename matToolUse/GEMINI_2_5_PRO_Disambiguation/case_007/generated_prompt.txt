import os
from pathlib import Path
from pymatgen.core import Structure
from pymatgen.io.vasp.outputs import Vasprun
from pymatgen.analysis.defects.core import DefectEntry
from pymatgen.analysis.defects.utils import get_freysoldt_correction, check_freysoldt_scaling

def compute_mg_ga_defect_properties():
    """
    Computes defect properties for a Mg_Ga substitution in a host material.

    This function calculates the Freysoldt correction, potential alignment
    consistency, and the energy difference between a defect and bulk supercell.
    It assumes a specific directory structure for VASP output files and uses
    a charge state of +1 for the defect calculations as a representative example.

    The required file structure relative to a base `file_path` is:
    - file_path/bulk_sc/vasprun.xml
    - file_path/bulk_sc/OUTCAR
    - file_path/Mg_Ga/charge_1/vasprun.xml
    - file_path/Mg_Ga/charge_1/OUTCAR

    Returns:
        dict: A dictionary containing the calculated properties:
              - 'freysoldt_correction' (float): The Freysoldt correction value.
              - 'potential_alignment_consistency' (bool): True if potential
                alignment is consistent.
              - 'energy_difference' (float): The energy difference between
                the defect and bulk supercells.
    """
    file_path = Path(".")
    dielectric_constant = 14.0
    defect_charge = 1

    bulk_dir = file_path / "bulk_sc"
    defect_dir = file_path / "Mg_Ga" / f"charge_{defect_charge}"

    bulk_vasprun_path = bulk_dir / "vasprun.xml"
    bulk_outcar_path = bulk_dir / "OUTCAR"
    defect_vasprun_path = defect_dir / "vasprun.xml"
    defect_outcar_path = defect_dir / "OUTCAR"

    bulk_vasprun = Vasprun(bulk_vasprun_path)
    bulk_structure = bulk_vasprun.final_structure
    bulk_energy = bulk_vasprun.final_energy

    defect_vasprun = Vasprun(defect_vasprun_path)
    defect_energy = defect_vasprun.final_energy

    energy_difference = defect_energy - bulk_energy

    defect_entry = DefectEntry(
        defect_structure=defect_vasprun.final_structure,
        charge_state=defect_charge,
        sc_energy=defect_energy,
        bulk_structure=bulk_structure,
    )

    freysoldt_correction_dict = get_freysoldt_correction(
        defect_entry=defect_entry,
        dielectric=dielectric_constant,
        defect_outcar_path=str(defect_outcar_path),
        bulk_outcar_path=str(bulk_outcar_path),
    )
    freysoldt_correction = freysoldt_correction_dict["freysoldt_correction"]

    potential_alignment_consistency = check_freysoldt_scaling(
        bulk_outcar_path, defect_outcar_path
    )

    results = {
        "freysoldt_correction": freysoldt_correction,
        "potential_alignment_consistency": potential_alignment_consistency,
        "energy_difference": energy_difference,
    }

    return results
