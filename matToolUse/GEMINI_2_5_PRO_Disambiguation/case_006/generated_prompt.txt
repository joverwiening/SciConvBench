import os
from pymatgen.core import Structure, Composition
from pymatgen.analysis.defects.core import NamedDefect, Defect
from pymatgen.core.periodic_table import Specie

def compute_defect_properties(file_path: str) -> dict:
    """
    Computes properties of a NamedDefect object using Pymatgen.

    This function loads a bulk and a defect structure, creates a structure-derived
    NamedDefect, and a manual reference NamedDefect. It then calculates the
    change in elements, a string representation, and equality/inequality
    between the two defect objects. The site index for the manual defect is
    assumed to be the index of the first 'Ga' atom found in the bulk structure.

    Args:
        file_path (str): The base directory path containing the structure files.

    Returns:
        dict: A dictionary containing the calculated defect properties:
              - 'element_changes' (dict): Change in the number of atoms for each element.
              - 'defect_string_representation' (str): String representation of the defect.
              - 'defect_equality' (bool): Result of the equality check (==).
              - 'defect_inequality' (bool): Result of the inequality check (!=).
    """
    # Construct file paths
    bulk_path = os.path.join(file_path, "Mg_Ga", "bulk_sc", "CONTCAR.gz")
    defect_path = os.path.join(file_path, "Mg_Ga", "q=0", "CONTCAR.gz")

    # Load structures
    bulk_structure = Structure.from_file(bulk_path)
    defect_structure = Structure.from_file(defect_path)

    # Create a structure-derived NamedDefect
    # This defect is generated by comparing the bulk and defect structures
    structure_derived_defect = NamedDefect.from_structures(bulk_structure, defect_structure)

    # --- Calculate properties based on the structure-derived defect ---

    # 1. Calculate element_changes
    element_changes = structure_derived_defect.element_changes

    # 2. Calculate defect_string_representation
    defect_string_representation = str(structure_derived_defect)

    # --- Perform equality checks by creating a manual reference defect ---

    # Find the site index of the first Ga atom in the bulk structure (our assumption)
    ga_site_index = -1
    for i, site in enumerate(bulk_structure):
        if site.specie.symbol == "Ga":
            ga_site_index = i
            break
    
    if ga_site_index == -1:
        raise ValueError("No 'Ga' atom found in the bulk structure.")

    # Get the site from the bulk structure using the found index
    defect_site = bulk_structure[ga_site_index]

    # Create the manual reference NamedDefect for an Mg substitution on a Ga site
    manual_defect = NamedDefect(
        bulk_structure.composition,
        defect_site,
        defect_type="substitutions",
        substitution_specie=Specie("Mg")
    )

    # 3. Check for equality
    defect_equality = (structure_derived_defect == manual_defect)

    # 4. Check for inequality
    defect_inequality = (structure_derived_defect != manual_defect)

    # Compile results into the final dictionary
    results = {
        "element_changes": element_changes,
        "defect_string_representation": defect_string_representation,
        "defect_equality": defect_equality,
        "defect_inequality": defect_inequality,
    }

    return results
