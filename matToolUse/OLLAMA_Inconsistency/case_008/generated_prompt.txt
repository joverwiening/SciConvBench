import os
import json
from typing import Dict

from pymatgen.core import Structure
from pymatgen.entries.computed_entries import ComputedEntry
from pymatgen.analysis.phase_diagram import PhaseDiagram


def compute_gan_defect_properties() -> Dict[str, object]:
    """
    Compute defect-related properties for a GaN system using Pymatgen.

    Returns
    -------
    dict
        A dictionary with the following keys:
        - "chempot_limits_count" (int): Number of distinct chemical‑potential limits
          in the formation‑energy phase diagram for the Ga‑N chemical system.
        - "defect_chemsys" (str): Chemical system of the defects expressed as
          element symbols separated by hyphens (e.g., "Ga-N").
        - "bulk_formula" (str): Reduced chemical formula of the bulk GaN
          structure (e.g., "GaN").
    """
    # ------------------------------------------------------------------
    # 1. Locate the input files.  `file_path` is assumed to be defined in the
    #    environment and point to the directory containing the data files.
    # ------------------------------------------------------------------
    base_dir = globals().get("file_path")
    if base_dir is None:
        raise RuntimeError("The variable `file_path` must be defined and point to the data directory.")

    structure_path = os.path.join(base_dir, "GaN.vasp")
    entries_path = os.path.join(base_dir, "stable_entries_Mg_Ga_N.json")

    # ------------------------------------------------------------------
    # 2. Load the bulk structure.
    # ------------------------------------------------------------------
    bulk_structure = Structure.from_file(structure_path)

    # ------------------------------------------------------------------
    # 3. Load the stable entries from the JSON file.
    #    The JSON file is expected to contain a list of dictionaries that can
    #    be converted to ComputedEntry objects via `ComputedEntry.from_dict`.
    # ------------------------------------------------------------------
    with open(entries_path, "r") as f:
        entry_dicts = json.load(f)

    stable_entries = [ComputedEntry.from_dict(d) for d in entry_dicts]

    # ------------------------------------------------------------------
    # 4. Build the phase diagram for the Ga‑N chemical system.
    # ------------------------------------------------------------------
    # Filter entries to keep only those that contain Ga or N (the Ga‑N system).
    ga_n_elements = {"Ga", "N"}
    ga_n_entries = [
        e for e in stable_entries if set(e.composition.elements).issubset(ga_n_elements)
    ]

    phase_diagram = PhaseDiagram(ga_n_entries)

    # ------------------------------------------------------------------
    # 5. Determine the number of chemical‑potential limits.
    #    `phase_diagram.chempot_limits` is a list of dictionaries, each
    #    representing a vertex of the stability region.  Its length gives the
    #    count of distinct limits.
    # ------------------------------------------------------------------
    chempot_limits_count = len(phase_diagram.chempot_limits)

    # ------------------------------------------------------------------
    # 6. Determine the defect chemical system.
    #    For this example we assume defects are considered within the same
    #    Ga‑N chemical system as the bulk, so we construct the string from the
    #    element symbols present in the bulk composition.
    # ------------------------------------------------------------------
    defect_elements = sorted({el.symbol for el in bulk_structure.composition.elements})
    defect_chemsys = "-".join(defect_elements)

    # ------------------------------------------------------------------
    # 7. Get the bulk formula in reduced form.
    # ------------------------------------------------------------------
    bulk_formula = bulk_structure.composition.reduced_formula

    # ------------------------------------------------------------------
    # 8. Return the results.
    # ------------------------------------------------------------------
    return {
        "chempot_limits_count": chempot_limits_count,
        "defect_chemsys": defect_chemsys,
        "bulk_formula": bulk_formula,
    }
