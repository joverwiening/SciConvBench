```python
def compute_named_defect_properties():
    """
    Compute properties of a NamedDefect using Pymatgen.

    Returns
    -------
    dict
        A dictionary containing:
        - "element_changes" (dict): Change in atom counts for each element
          derived from the structure-derived Ga vacancy (e.g., {"Ga": -1}).
        - "defect_string_representation" (str): String combining the bulk
          reduced formula and the defect name, formatted as
          "<bulk_formula>_<defect_name>".
        - "defect_equality" (bool): True if the structure-derived defect
          equals the manually defined Ga‑vacancy NamedDefect.
        - "defect_inequality" (bool): True if the structure-derived defect
          does not equal the manually defined Mg‑on‑Ga antisite NamedDefect.
    """
    import os
    from pymatgen.core import Structure
    from pymatgen.analysis.defects.named_defects import NamedDefect

    # Base directory containing the input files – adjust as needed.
    file_path = "./"

    # File locations
    bulk_file = os.path.join(file_path, "Mg_Ga", "bulk_sc", "CONTCAR.gz")
    defect_file = os.path.join(file_path, "Mg_Ga", "q=0", "CONTCAR.gz")

    # Load structures
    bulk_structure = Structure.from_file(bulk_file)
    defect_structure = Structure.from_file(defect_file)

    # 1. Structure‑derived Ga vacancy (named as v_Ga)
    nd0 = NamedDefect(bulk_structure, defect_structure, name="v_Ga")

    # 2. Manual reference for Ga vacancy
    v_Ga_manual = NamedDefect(
        bulk_structure,
        defect_structure,
        name="v_Ga",
        element_changes={"Ga": -1},
    )

    # 3. Manual reference for Mg‑on‑Ga antisite
    Mg_Ga_manual = NamedDefect(
        bulk_structure,
        defect_structure,
        name="Mg_Ga",
        element_changes={"Mg": 1, "Ga": -1},
    )

    # Compute element changes for nd0
    bulk_comp = nd0.bulk_structure.composition
    defect_comp = nd0.defect_structure.composition
    delta_comp = defect_comp - bulk_comp
    element_changes = {el: int(delta_comp[el]) for el in delta_comp.elements if delta_comp[el] != 0}

    # String representation: "<bulk_formula>_<defect_name>"
    bulk_formula = nd0.bulk_structure.composition.reduced_formula
    defect_string_representation = f"{bulk_formula}_{nd0.name}"

    # Equality / inequality checks
    defect_equality = nd0 == v_Ga_manual
    defect_inequality = nd0 != Mg_Ga_manual

    # Return results
    return {
        "element_changes": element_changes,
        "defect_string_representation": defect_string_representation,
        "defect_equality": defect_equality,
        "defect_inequality": defect_inequality,
    }
```
