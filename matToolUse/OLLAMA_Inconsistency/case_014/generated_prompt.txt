def compute_gan_properties():

    import os
    import json
    from pymatgen.core.composition import Composition
    from pymatgen.entries.computed_entries import ComputedEntry
    from pymatgen.analysis.phase_diagram import PhaseDiagram, ensure_stable_bulk

    # Base directory containing the input files

    entries = []

        comp = Composition(entry['composition'])

    # Build the phase diagram
    pd = PhaseDiagram(entries)

    gan_comp = Composition('GaN')

    # Create a slightly unstable GaN entry (+2 eV)
    unstable_energy = hull_energy + 2.0
    unstable_entry = ComputedEntry(gan_comp, unstable_energy)

    # Ensure the entry is stable in the bulk
    stable_entry = ensure_stable_bulk(unstable_entry, pd)

    # Determine if GaN is now among the stable entries
    gan_is_stable = any(e.composition.reduced_formula == 'GaN' for e in pd.stable_entries)
