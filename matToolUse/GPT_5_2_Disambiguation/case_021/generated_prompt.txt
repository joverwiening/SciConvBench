def compute_freysoldt_properties():
    """
    Compute defect correction properties for the Mg_Ga dataset using Pymatgen.

    Parameters
    ----------
    None
        This function takes no required input arguments.

    Data/Filesystem Assumptions
    ---------------------------
    - A base directory named `file_path` is assumed to be the current working directory.
    - The Mg_Ga dataset directory is assumed to exist at: <file_path>/Mg_Ga
    - The function recursively scans subfolders under <file_path>/Mg_Ga to find a defect-state
      calculation folder containing:
        - vasprun.xml or vasprun.xml.gz
        - LOCPOT or LOCPOT.gz
    - The first matching folder (lexicographically sorted by full path) is used.

    Physical Assumptions
    --------------------
    - Defect charge state: q = -1
    - Dielectric constant: isotropic epsilon = 10.0, represented as a 3x3 diagonal tensor.

    Returns
    -------
    dict
        Dictionary with keys:
        - "freysoldt_correction_energy" (float): Freysoldt correction energy in eV.
    """
    import os
    import numpy as np

    from pymatgen.io.vasp.outputs import Vasprun, Locpot
    from pymatgen.analysis.defects.corrections.freysoldt import get_freysoldt_correction

    # Base path assumption (best practice default: current working directory)
    file_path = os.getcwd()
    mg_ga_root = os.path.join(file_path, "Mg_Ga")

    if not os.path.isdir(mg_ga_root):
        raise FileNotFoundError(
            f"Expected Mg_Ga directory at '{mg_ga_root}', but it does not exist."
        )

    # Recursively find candidate directories containing vasprun and LOCPOT
    candidate_dirs = []
    for root, dirs, files in os.walk(mg_ga_root):
        files_set = set(files)

        # Prefer gzipped if present
        vasprun_path = None
        if "vasprun.xml.gz" in files_set:
            vasprun_path = os.path.join(root, "vasprun.xml.gz")
        elif "vasprun.xml" in files_set:
            vasprun_path = os.path.join(root, "vasprun.xml")

        locpot_path = None
        if "LOCPOT.gz" in files_set:
            locpot_path = os.path.join(root, "LOCPOT.gz")
        elif "LOCPOT" in files_set:
            locpot_path = os.path.join(root, "LOCPOT")

        if vasprun_path and locpot_path:
            candidate_dirs.append((root, vasprun_path, locpot_path))

    if not candidate_dirs:
        raise FileNotFoundError(
            f"No calculation directory under '{mg_ga_root}' contained both "
            f"vasprun.xml(.gz) and LOCPOT(.gz)."
        )

    # Select first matching directory deterministically
    candidate_dirs.sort(key=lambda x: x[0])
    selected_root, vasprun_file, locpot_file = candidate_dirs[0]

    # Load VASP outputs
    vr = Vasprun(vasprun_file, parse_potcar_file=False)
    structure = vr.final_structure
    locpot = Locpot.from_file(locpot_file)

    # Assumed defect charge and dielectric tensor
    q = -1
    epsilon = 10.0
    dielectric_tensor = np.diag([epsilon, epsilon, epsilon])

    # Compute Freysoldt correction
    corr = get_freysoldt_correction(
        locpot=locpot,
        structure=structure,
        charge=q,
        dielectric=dielectric_tensor,
    )

    # Extract energy robustly
    energy = None
    if isinstance(corr, dict):
        energy = corr.get("energy", None)
    else:
        energy = getattr(corr, "energy", None)

    if energy is None:
        raise RuntimeError(
            f"Could not extract correction energy from get_freysoldt_correction return "
            f"type '{type(corr)}'."
        )

    return {"freysoldt_correction_energy": float(energy)}
