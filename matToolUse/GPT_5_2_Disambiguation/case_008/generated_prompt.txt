def compute_defect_formation_diagram_properties():
    """
    Compute basic properties derived from a defect formation energy diagram using Pymatgen's defects tooling.

    Parameters
    ----------
    None

    Data sources (resolved under a base directory `file_path`)
    ---------------------------------------------------------
    - GaN.vasp: Bulk structure file read with pymatgen.core.Structure.
    - stable_entries_Mg_Ga_N.json: JSON containing stable entries; used to select the bulk entry.
    - defect_entries.json (optional): JSON containing defect entries serialized with Monty/Pymatgen.
    - band_edges.json (optional): JSON containing band-edge data; expected to contain a float under key "vbm".

    Returns
    -------
    dict
        Dictionary with the following keys:
        - "chempot_limits_count" (int): Number of chemical potential limits/environments in the formation energy diagram.
          If required inputs to build the diagram are missing, this is returned as 0.
        - "defect_chemsys" (str): Chemical system string (element symbols separated by hyphens) derived from the bulk structure.
        - "bulk_formula" (str): Reduced formula of the bulk structure.
    """
    import os
    import json
    from pathlib import Path

    from monty.json import MontyDecoder
    from pymatgen.core import Structure
    from pymatgen.analysis.defects.thermodynamics import FormationEnergyDiagram

    # Base directory resolution (best-practice default)
    try:
        file_path = Path(__file__).resolve().parent
    except NameError:
        file_path = Path(os.getcwd())

    structure_path = file_path / "GaN.vasp"
    stable_entries_path = file_path / "stable_entries_Mg_Ga_N.json"
    defect_entries_path = file_path / "defect_entries.json"
    band_edges_path = file_path / "band_edges.json"

    # Bulk structure-derived properties
    structure = Structure.from_file(str(structure_path))
    bulk_formula = str(structure.composition.reduced_formula)
    defect_chemsys = "-".join(sorted([el.symbol for el in structure.composition.elements]))

    # Optional defect entries
    defect_entries = []
    if defect_entries_path.exists():
        with open(defect_entries_path, "r") as f:
            defect_entries = json.load(f, cls=MontyDecoder)

    # Optional band edges (VBM)
    vbm = 0.0
    if band_edges_path.exists():
        with open(band_edges_path, "r") as f:
            be = json.load(f)
        if isinstance(be, dict) and "vbm" in be:
            try:
                vbm = float(be["vbm"])
            except Exception:
                vbm = 0.0

    # Bulk entry from stable entries
    bulk_entry = None
    if stable_entries_path.exists():
        with open(stable_entries_path, "r") as f:
            stable_entries = json.load(f, cls=MontyDecoder)
        # stable_entries may be a list or dict depending on how it was serialized
        if isinstance(stable_entries, dict):
            candidates = list(stable_entries.values())
        else:
            candidates = list(stable_entries)

        for e in candidates:
            try:
                if str(e.composition.reduced_formula) == bulk_formula:
                    bulk_entry = e
                    break
            except Exception:
                continue

    # Build formation energy diagram if possible and count chempot limits
    chempot_limits_count = 0
    if bulk_entry is not None and defect_entries:
        fed = None
        # Try common constructor patterns across pymatgen-analysis-defects versions
        try:
            fed = FormationEnergyDiagram(defect_entries=defect_entries, bulk_entry=bulk_entry, vbm=vbm)
        except TypeError:
            try:
                fed = FormationEnergyDiagram(defect_entries, bulk_entry, vbm)
            except TypeError:
                # Last-resort: try keyword variants
                fed = FormationEnergyDiagram(defect_entries=defect_entries, bulk_entry=bulk_entry, vbm=vbm, band_gap=None)

        # Count chemical potential limits/environments
        if hasattr(fed, "chempot_limits") and fed.chempot_limits is not None:
            chempot_limits_count = len(fed.chempot_limits)
        elif hasattr(fed, "chempots") and fed.chempots is not None:
            chempot_limits_count = len(fed.chempots)
        else:
            chempot_limits_count = 0

    return {
        "chempot_limits_count": int(chempot_limits_count),
        "defect_chemsys": defect_chemsys,
        "bulk_formula": bulk_formula,
    }
