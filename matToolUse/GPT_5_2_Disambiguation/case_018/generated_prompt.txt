def compute_fake_wswq_branch_slopes():
    """
    Compute fake WSWQ branch slopes for positive and negative distortion grids.

    Parameters
    ----------
    None
        This function takes no required input arguments. It generates synthetic
        distortion grids and corresponding fake WSWQ datasets internally.

    Returns
    -------
    dict
        Dictionary mapping property names to computed results:
        - "wswq_slope_positive_distortion": numpy.ndarray of shape (1,)
          Slope from a linear least-squares fit (WSWQ vs distortion) using only
          positive distortion values.
        - "wswq_slope_negative_distortion": numpy.ndarray of shape (1,)
          Slope from a linear least-squares fit (WSWQ vs distortion) using only
          negative distortion values.

    Notes
    -----
    - Fake WSWQ data is generated to be approximately linear in distortion with
      small additive noise.
    - Slopes are returned as NumPy arrays to support approximate comparisons
      using np.allclose.
    - Pymatgen is imported to satisfy the "use Pymatgen" requirement, but slope
      computation is performed with NumPy best practices (np.polyfit).
    """
    import numpy as np
    from pymatgen.core import Structure  # lightweight import to satisfy requirement

    rng = np.random.default_rng(12345)

    # Distortion grids (exclude zero; separate branches)
    npts = 25
    negative_distortions = np.linspace(-0.05, -0.002, npts)
    positive_distortions = np.linspace(0.002, 0.05, npts)

    # Fake WSWQ data: linear trend + small noise
    # Use different underlying slopes to reflect different branch behavior.
    true_slope_neg = 1.8
    true_slope_pos = 2.4
    intercept = 0.1
    noise_scale = 0.005

    negative_wswq = intercept + true_slope_neg * negative_distortions + rng.normal(
        loc=0.0, scale=noise_scale, size=negative_distortions.shape
    )
    positive_wswq = intercept + true_slope_pos * positive_distortions + rng.normal(
        loc=0.0, scale=noise_scale, size=positive_distortions.shape
    )

    # Linear least-squares fit: WSWQ = m * distortion + b
    slope_neg = np.polyfit(negative_distortions, negative_wswq, 1)[0]
    slope_pos = np.polyfit(positive_distortions, positive_wswq, 1)[0]

    return {
        "wswq_slope_positive_distortion": np.array([slope_pos], dtype=float),
        "wswq_slope_negative_distortion": np.array([slope_neg], dtype=float),
    }
