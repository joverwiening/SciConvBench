from __future__ import annotations

from pathlib import Path
from collections import Counter
from typing import Dict, Any

from pymatgen.core import Structure
from pymatgen.core.periodic_table import DummySpecie


def compute_defect_complex_properties() -> Dict[str, Any]:
    """
    Compute defect-complex properties for GaN using pymatgen.

    Parameters
    ----------
    None

    Data sources / assumptions
    --------------------------
    - Reads a pristine structure from a VASP POSCAR/CONTCAR-like file named "GaN.vasp".
    - The base directory `file_path` is assumed to be the directory containing this script
      (Path(__file__).resolve().parent) when available, otherwise the current working directory.
    - Oxidation states are assigned by best-practice assumptions for GaN and introduced species:
      Ga = +3, N = -3, O = -2, H = +1.
    - Defects are constructed from the loaded structure:
      * Substitution: site index 3 (N) replaced by O (O-on-N).
      * Vacancy: site index 0 (Ga) removed (V_Ga).
      * Interstitial (optional): H added at fractional coordinates [0, 0, 0.75] (H_i).
    - Two defect complexes are built:
      1) O_N + V_Ga
      2) O_N + V_Ga + H_i

    Returns
    -------
    dict
        Dictionary mapping property names to computed values:
        - defect_complex_name (str)
        - supercell_structure_formula (str)
        - defect_complex_oxidation_state (int)
        - element_changes (dict)
        - defect_structure_formula (str)
        - defect_complex_with_interstitial_name (str)
        - supercell_structure_with_dummy_formula (str)
        - defect_complex_equality (bool)
        - defect_complex_inequality (bool)
    """

    # Resolve file path robustly
    try:
        file_path = Path(__file__).resolve().parent
    except NameError:
        file_path = Path.cwd()

    gan_path = file_path / "GaN.vasp"
    pristine = Structure.from_file(str(gan_path))

    # Assign oxidation states
    oxi = {"Ga": 3, "N": -3, "O": -2, "H": 1}
    pristine = pristine.copy()
    pristine.add_oxidation_state_by_element(oxi)

    # Helper: element counts
    def _element_counter(struct: Structure) -> Counter:
        c = Counter()
        for site in struct.sites:
            # site.specie is a Specie with oxidation state after decoration; use symbol
            c[site.specie.symbol] += 1
        return c

    pristine_counts = _element_counter(pristine)

    # --- Build defect complex #1: O_N + V_Ga ---
    complex1 = pristine.copy()

    # Substitution at index 3: N -> O
    # Keep oxidation state decoration consistent by using Specie-like string with oxidation state
    complex1[3] = "O2-"

    # Vacancy at index 0: remove Ga
    complex1.remove_sites([0])

    complex1_counts = _element_counter(complex1)

    # --- Build defect complex #2: O_N + V_Ga + H_i ---
    complex2 = pristine.copy()
    complex2[3] = "O2-"
    complex2.remove_sites([0])
    complex2.append("H+", [0, 0, 0.75], coords_are_cartesian=False)

    # --- Names ---
    defect_complex_name = "O_N + V_Ga"
    defect_complex_with_interstitial_name = "O_N + V_Ga + H_i"

    # --- Formulas ---
    supercell_structure_formula = complex1.composition.reduced_formula

    # For "dummy" formula: optionally replace H with a dummy species X to indicate interstitial accounting.
    # Here we demonstrate adding a dummy species in addition to H to "account" for the interstitial.
    # If you prefer to literally replace H with X, change the logic accordingly.
    complex2_with_dummy = complex2.copy()
    # Add a dummy at the same fractional coordinate to explicitly "account" for the interstitial
    complex2_with_dummy.append(DummySpecie("X"), [0, 0, 0.75], coords_are_cartesian=False)
    supercell_structure_with_dummy_formula = complex2_with_dummy.composition.reduced_formula

    # --- element_changes for complex #1 relative to pristine ---
    all_elems = set(pristine_counts) | set(complex1_counts)
    element_changes = {el: int(complex1_counts.get(el, 0) - pristine_counts.get(el, 0)) for el in sorted(all_elems)}
    # Remove zero entries for cleanliness
    element_changes = {k: v for k, v in element_changes.items() if v != 0}

    # --- defect_structure_formula (defect-only representation) ---
    # Since a chemical formula cannot contain negative stoichiometries, represent defect content as a compact tag.
    # Example: substitution contributes +O and -N; vacancy contributes -Ga.
    # We encode as "O1_N-1_Ga-1" style string.
    parts = []
    for el, delta in element_changes.items():
        parts.append(f"{el}{delta:+d}")
    defect_structure_formula = "_".join(parts) if parts else "no_change"

    # --- defect_complex_oxidation_state for complex #1 ---
    # Compute as sum of oxidation-state changes implied by element_changes.
    # This is a defect "charge" proxy based on formal oxidation states.
    defect_complex_oxidation_state = int(sum(oxi[el] * delta for el, delta in element_changes.items()))

    # --- Equality / inequality ---
    # Define identity as same species at same fractional coordinates within a tolerance after sorting.
    # Here, complex #1 and #2 differ by an interstitial, so they should not be identical.
    def _structure_signature(struct: Structure, tol: float = 1e-6):
        items = []
        for s in struct.sites:
            frac = s.frac_coords
            items.append(
                (
                    s.specie.symbol,
                    round(frac[0] / tol) * tol,
                    round(frac[1] / tol) * tol,
                    round(frac[2] / tol) * tol,
                )
            )
        return tuple(sorted(items))

    defect_complex_equality = _structure_signature(complex1) == _structure_signature(complex2)
    defect_complex_inequality = not defect_complex_equality

    return {
        "defect_complex_name": defect_complex_name,
        "supercell_structure_formula": supercell_structure_formula,
        "defect_complex_oxidation_state": defect_complex_oxidation_state,
        "element_changes": element_changes,
        "defect_structure_formula": defect_structure_formula,
        "defect_complex_with_interstitial_name": defect_complex_with_interstitial_name,
        "supercell_structure_with_dummy_formula": supercell_structure_with_dummy_formula,
        "defect_complex_equality": defect_complex_equality,
        "defect_complex_inequality": defect_complex_inequality,
    }
