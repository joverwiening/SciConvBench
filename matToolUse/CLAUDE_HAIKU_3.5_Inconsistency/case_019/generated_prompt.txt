```python
import os
import numpy as np
from pymatgen.analysis.defects.ccd import HarmonicDefect, get_SRH_coefficient

def compute_semiconductor_defect_properties():
    """
    Compute Shockley-Read-Hall (SRH) coefficient and runtime error verification for semiconductor defects.

    Returns:
        dict: A dictionary containing:
            - 'SRH_Coefficient': List of float values representing SRH coefficient at [300, 600, 900] K
            - 'RuntimeError_Check': Boolean indicating if RuntimeError is raised with invalid configuration
    """
    # Construct file paths
    base_dir = os.path.join(os.getcwd(), 'v_Ga', 'ccd_0_-1', 'wswqs')
    
    # Locate vaspruns and procar files (assuming standard naming conventions)
    vaspruns_path = os.path.join(base_dir, 'VASP', 'vaspruns')
    procar_path = os.path.join(base_dir, 'VASP', 'PROCAR')
    
    # Create HarmonicDefect object in neutral state
    harmonic_defect = HarmonicDefect(
        vaspruns_file=vaspruns_path, 
        procar_file=procar_path, 
        charge_state=0
    )
    
    # Load WSWQ files into the HarmonicDefect object
    harmonic_defect.load_wswq_files(base_dir)
    
    # Compute SRH coefficient for specified temperatures
    temperatures = [300, 600, 900]
    srh_coefficients = [
        get_SRH_coefficient(harmonic_defect, temperature) 
        for temperature in temperatures
    ]
    
    # Verify RuntimeError handling
    runtime_error_check = False
    try:
        # Attempt to calculate with an intentionally invalid configuration
        get_SRH_coefficient(harmonic_defect, temperature=-1)
    except RuntimeError:
        runtime_error_check = True
    
    # Return results as dictionary
    return {
        'SRH_Coefficient': srh_coefficients,
        'RuntimeError_Check': runtime_error_check
    }
```

5. Includes RuntimeError verification
