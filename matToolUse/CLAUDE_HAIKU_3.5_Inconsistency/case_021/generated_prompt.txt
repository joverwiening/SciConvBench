import os
import gzip
from typing import Dict, Any
from pymatgen.io.vasp.outputs import Vasprun
from pymatgen.analysis.defects.corrections.freysoldt import get_freysoldt_correction

def compute_mg_ga_defect_properties(file_path: str = '.') -> Dict[str, Any]:
    """
    Compute Freysoldt correction energies for Mg_Ga defect systems.

    This function recursively searches the specified directory for Mg_Ga defect 
    calculation files and computes Freysoldt correction energies.

    Args:
        file_path (str, optional): Base directory containing Mg_Ga defect calculations. 
                                   Defaults to current directory.

    Returns:
        Dict[str, Any]: A dictionary where keys are relative file paths and values 
                        are Freysoldt correction energy calculations.
    """
    results = {}

    # Recursively search for vasprun.xml.gz and LOCPOT.gz file pairs
    for root, _, files in os.walk(os.path.join(file_path, 'Mg_Ga')):
        vasprun_files = [f for f in files if f == 'vasprun.xml.gz']
        locpot_files = [f for f in files if f == 'LOCPOT.gz']

        for vasprun_file in vasprun_files:
            vasprun_path = os.path.join(root, vasprun_file)
            locpot_path = os.path.join(root, 'LOCPOT.gz')

            # Verify LOCPOT exists for this vasprun
            if os.path.exists(locpot_path):
                try:
                    # Load vasprun and LOCPOT
                    with gzip.open(vasprun_path, 'rb') as f:
                        vasprun = Vasprun(f)
                    
                    with gzip.open(locpot_path, 'rb') as f:
                        locpot = f.read()  # Raw LOCPOT data

                    # Compute Freysoldt correction
                    freysoldt_correction = get_freysoldt_correction(
                        vasprun.final_structure, 
                        vasprun.final_energy, 
                        locpot_data=locpot
                    )

                    # Store result using relative path as key
                    relative_path = os.path.relpath(root, file_path)
                    results[relative_path] = {
                        'freysoldt_correction_energy': freysoldt_correction
                    }

                except Exception as e:
                    print(f"Error processing {vasprun_path}: {e}")

    return results
