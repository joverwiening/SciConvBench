import numpy as np
import pymatgen as pmg

def compute_wswq_slopes():
    """
    Generate fake WSWQ data and compute slopes for positive and negative distortions.

    Returns:
        dict: A dictionary containing computed WSWQ slope properties.
            - 'wswq_slope_positive_distortion': Slope for positive distortion branch
            - 'wswq_slope_negative_distortion': Slope for negative distortion branch
    """
    # Distortion values for both branches
    positive_distortions = np.array([0.5, 0.0, -0.5])
    negative_distortions = np.array([-0.5, 0.0, 0.5])

    # Generate fake WSWQ data (example synthetic data)
    fake_wswqs_positive = np.array([1.2, 0.8, 0.3])
    fake_wswqs_negative = np.array([0.3, 0.8, 1.2])

    # Calculate slopes using linear regression
    wswq_slope_positive = np.polyfit(positive_distortions, fake_wswqs_positive, 1)[0]
    wswq_slope_negative = np.polyfit(negative_distortions, fake_wswqs_negative, 1)[0]

    # Return results as a dictionary
    return {
        'wswq_slope_positive_distortion': wswq_slope_positive,
        'wswq_slope_negative_distortion': wswq_slope_negative
    }
