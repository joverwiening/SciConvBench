    Method:
    - Uses participation ratio to identify localized bands
    - Localization is determined by low participation ratio (< 0.2)
    - Analyzes band structures from multiple vasprun.xml files
    - Pairs vasprun.xml files with corresponding PROCAR files

    Returns:
    dict: A dictionary containing localized band sets for different configurations
        - 'localized_bands_set_1': Set of localized band indices for first configuration
        - 'localized_bands_set_2': Set of localized band indices for second configuration
    """
    # Paths for different configurations
    config_1_path = os.path.join(file_path, 'v_Ga', 'ccd_0_-1')
    config_2_path = os.path.join(file_path, 'v_Ga', 'ccd_-1_0')

    def get_localized_bands(vasprun_path, procar_path):
        """
        Identify localized bands using participation ratio.
        
        Args:
            vasprun_path (str): Path to vasprun.xml
            procar_path (str): Path to PROCAR file
        
        Returns:
            set: Indices of localized bands
        """
        # Load vasprun and PROCAR
        vrun = Vasprun(vasprun_path)
        procar = Procar(procar_path)
        
        # Extract projected weights
        proj_weights = procar.data['weight']
        
        # Calculate participation ratio for each band
        participation_ratios = np.sum(proj_weights**2, axis=(1,2))
        
        # Identify localized bands (low participation ratio)
        localized_bands = {
            idx for idx, pr in enumerate(participation_ratios) 
            if pr < 0.2  # Threshold for localization
        }
        
        return localized_bands

    # Analyze first configuration
    localized_bands_set_1 = get_localized_bands(

    )

    return {
        'localized_bands_set_1': localized_bands_set_1,
