```python
import os
from typing import Dict
from pymatgen.core import Structure
from pymatgen.analysis.defects.formation_energy_diagram import FormationEnergyDiagram

def analyze_mg_ga_properties() -> Dict[str, int]:
    """
    Analyze Mg-Ga properties using Pymatgen.

    Loads GaN structure, creates Mg substitution defect, and computes formation energy diagram
    properties with a dielectric constant of 10.

    Returns:
        Dict[str, int]: A dictionary containing computed properties:
            - 'directory_map_length': Number of directories in the formation energy diagram
            - 'transition_count': Number of transition states in the formation energy diagram
    """
    # Assume base file path (this would be replaced with actual path in real implementation)
    file_path = os.path.join(os.getcwd(), 'Mg_Ga')
    
    # Load GaN structure
    structure_path = os.path.join(file_path, 'GaN.vasp')
    structure = Structure.from_file(structure_path)
    
    # Create Mg substitution defect on Ga site
    defect_structure = structure.copy()
    defect_structure.replace_species({'Ga': 'Mg'})
    
    # Construct directory paths
    directories = {
        'bulk_sc': os.path.join(file_path, 'bulk_sc'),
        'q_-1': os.path.join(file_path, 'q_-1'),
        'q_0': os.path.join(file_path, 'q_0'),
        'q_1': os.path.join(file_path, 'q_1')
    }
    
    # Create Formation Energy Diagram
    formation_diagram = FormationEnergyDiagram.with_directories(
        directories=directories,
        dielectric_constant=10
    )
    
    # Compute properties
    directory_map_length = len(formation_diagram.directories)
    transition_count = len(formation_diagram.transition_states)
    
    return {
        'directory_map_length': directory_map_length,
        'transition_count': transition_count
    }
```

1. Uses Pymatgen to load the GaN structure
2. Creates a Mg substitution defect on the Ga site
3. Constructs a directory map with the specified directories
4. Creates a `FormationEnergyDiagram` with a dielectric constant of 10
5. Calculates `directory_map_length` using `len(formation_diagram.directories)`
6. Calculates `transition_count` using `len(formation_diagram.transition_states)`
7. Returns a dictionary with the computed integer properties
