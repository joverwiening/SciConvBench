import os
import gzip
import json
from typing import Dict, List, Set

import numpy as np
from pymatgen.core import Composition
from pymatgen.entries.computed_entries import ComputedEntry
from pymatgen.io.vasp.outputs import Vasprun
from pymatgen.analysis.phase_diagram import PhaseDiagram
from pymatgen.analysis.defects.thermo import FormationEnergyDiagram

def compute_defect_properties(file_path: str = None) -> Dict:
    """
    Compute defect-related properties for Mg-Ga-N system using Pymatgen.

    Loads stable entries, defect entries, and computes formation energy diagram 
    with competing phases at chemical potential limits.

    Args:
        file_path (str, optional): Base directory containing input files. 
                                   Defaults to current working directory.

    Returns:
        Dict: A dictionary containing computed properties with the following keys:
            - 'competing_phases_at_chempot_limits': Dictionary of competing phases
              at different chemical potential limits
    """
    # Set default file path if not provided
    if file_path is None:
        file_path = os.getcwd()

    # 1. Load stable entries
    stable_entries_path = os.path.join(file_path, 'stable_entries_Mg_Ga_N.json')
    with open(stable_entries_path, 'r') as f:
        stable_entries_data = json.load(f)

    # Convert stable entries to ComputedEntries
    stable_entries = [
        ComputedEntry(
            Composition(entry['composition']), 
            entry['energy'], 
            parameters=entry.get('parameters', {})
        ) for entry in stable_entries_data
    ]

    # 2. Collect elemental entries
    elemental_entries = []
    elements = set()
    for entry in stable_entries:
        elements.update(entry.composition.elements)

    # Load elemental entries (assuming they are in the same directory)
    for element in elements:
        elem_entry_path = os.path.join(file_path, f'{element}_entry.json')
        if os.path.exists(elem_entry_path):
            with open(elem_entry_path, 'r') as f:
                elem_data = json.load(f)
                elemental_entries.append(
                    ComputedEntry(
                        Composition(element), 
                        elem_data['energy'], 
                        parameters=elem_data.get('parameters', {})
                    )
                )

    # 3. Load bulk entry and VBM
    bulk_vasprun_path = os.path.join(file_path, 'bulk', 'vasprun.xml.gz')
    with gzip.open(bulk_vasprun_path, 'rb') as f:
        bulk_vasprun = Vasprun(f)
    
    bulk_entry = ComputedEntry(
        bulk_vasprun.final_structure.composition, 
        bulk_vasprun.final_energy
    )
    vbm = bulk_vasprun.eigenvalue_band_properties[1]

    # 4. Collect defect entries for all charge states
    defect_entries = []
    charge_states = [-2, -1, 0, 1]

    for charge in charge_states:
        charge_dir = os.path.join(file_path, 'Mg_Ga', f'charge_{charge}')
        
        # Find all defect calculation directories
        for defect_calc_dir in os.listdir(charge_dir):
            vasprun_path = os.path.join(charge_dir, defect_calc_dir, 'vasprun.xml.gz')
            locpot_path = os.path.join(charge_dir, defect_calc_dir, 'LOCPOT.gz')
            
            # Skip if files are missing
            if not (os.path.exists(vasprun_path) and os.path.exists(locpot_path)):
                continue
            
            # Load vasprun
            with gzip.open(vasprun_path, 'rb') as f:
                defect_vasprun = Vasprun(f)
            
            # Create defect entry with charge state
            defect_entry = ComputedEntry(
                defect_vasprun.final_structure.composition, 
                defect_vasprun.final_energy,
                parameters={
                    'charge': charge,
                    'defect_type': defect_calc_dir
                }
            )
            defect_entries.append(defect_entry)

    # 5. Create Phase Diagram
    all_entries = stable_entries + elemental_entries + [bulk_entry] + defect_entries
    phase_diagram = PhaseDiagram(all_entries)

    # 6. Create Formation Energy Diagram
    formation_diagram = FormationEnergyDiagram(
        phase_diagram, 
        defect_entries,
        bulk_entry=bulk_entry,
        vbm=vbm
    )

    # 7. Compute competing phases at chemical potential limits
    chempot_limits = {
        'Mg-rich': {'Mg': -1.0, 'Ga': 0.0, 'N': 0.0},
        'Ga-rich': {'Mg': 0.0, 'Ga': -1.0, 'N': 0.0},
        'N-rich': {'Mg': 0.0, 'Ga': 0.0, 'N': -1.0}
    }

    competing_phases = {}
    for limit_name, limit_values in chempot_limits.items():
        competing_phases[limit_name] = set(
            phase.name for phase in formation_diagram.get_competing_phases(limit_values)
        )

    return {
        'competing_phases_at_chempot_limits': competing_phases
    }
