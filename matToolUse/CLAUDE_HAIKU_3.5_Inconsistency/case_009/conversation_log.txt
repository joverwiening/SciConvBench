from pymatgen.analysis.defects.core import DefectEntry
from pymatgen.analysis.defects.corrections import FreysoldtCorrection
from pymatgen.analysis.defects.formation_energy import FormationEnergyDiagram

def compute_chemical_potential_properties(file_path: str = None) -> Dict[str, Any]:
    """
    Compute chemical potential properties for the Mg-Ga-N defect system.

    Args:
        file_path (str, optional): Base directory containing defect entries. 
                                   Defaults to current working directory.

    Returns:
        Dict containing chemical potential limits count and related properties.
    """
    # Set default file path
    if file_path is None:
        file_path = os.getcwd()

    # Load bulk supercell calculation
    bulk_locpot_path = os.path.join(file_path, 'bulk_sc', 'LOCPOT.gz')
    bulk_vasprun_path = os.path.join(file_path, 'bulk_sc', 'vasprun.xml.gz')
    
    # Parse bulk calculation
    with gzip.open(bulk_vasprun_path, 'rb') as f:
        bulk_vasprun = Vasprun(f)
    
    bulk_structure = bulk_vasprun.structure
    bulk_vbm = bulk_vasprun.eigenvalue_band_properties[1]  # VBM energy
    bulk_entry = bulk_vasprun.get_computed_entry()

    # Dielectric constant
    dielectric_constant = 14.0

    # Prepare DefectEntries for each charge state
    defect_entries = []
    charge_states = [-2, -1, 0, 1]

    for charge in charge_states:
        # Load charge state specific files
        charge_dir = os.path.join(file_path, 'charge_states', f'q{charge}')
        vasprun_path = os.path.join(charge_dir, 'vasprun.xml.gz')
        locpot_path = os.path.join(charge_dir, 'LOCPOT.gz')

        # Parse defect calculation
        with gzip.open(vasprun_path, 'rb') as f:
            defect_vasprun = Vasprun(f)
        
        defect_structure = defect_vasprun.structure
        defect_entry = defect_vasprun.get_computed_entry()

        # Apply Freysoldt correction
        freysoldt_correction = FreysoldtCorrection(
            defect_structure, 
            bulk_structure, 
            charge, 
            dielectric_constant
        )

        # Compute correction parameters
        correction_data = freysoldt_correction.get_correction_data(
            bulk_vbm=bulk_vbm
        )

        # Create DefectEntry
        defect_entry_obj = DefectEntry(
            defect_structure, 
            defect_entry.composition, 
            charge, 
            bulk_entry,
            correction_data=correction_data
        )

        defect_entries.append(defect_entry_obj)

    # Create Formation Energy Diagram
    formation_energy_diagram = FormationEnergyDiagram(
        bulk_entry, 
        defect_entries
    )

    # Compute chemical potential limits count
    chemical_potential_limits_count = len(formation_energy_diagram.chemical_potential_limits)

    return {
        'chemical_potential_limits_count': chemical_potential_limits_count
    }
