def compute_fermi_level(dos, temperature, kb=8.617333262e-5):
    """
    Compute Fermi level using density of states and temperature.

    Args:
        dos (CompleteDos): Complete density of states object
        temperature (float): Temperature in Kelvin
        kb (float): Boltzmann constant in eV/K

    Returns:
        float: Fermi level ensuring value > 0
    """
    # Extract energy and DOS values
    energies = dos.energies
    total_dos = dos.get_total_dos()
    
    # Compute Fermi level using statistical mechanics approach
    fermi_level = dos.efermi  # Start with DFT calculated Fermi level
    
    # Simple temperature correction (simplified model)
    thermal_correction = kb * temperature * np.log(
        np.sum(total_dos.densities[Spin.up] * np.exp(-(energies - fermi_level) / (kb * temperature)))
    )
    
    fermi_level_corrected = fermi_level + thermal_correction
    
    return max(fermi_level_corrected, 0.001)  # Ensure positive value
