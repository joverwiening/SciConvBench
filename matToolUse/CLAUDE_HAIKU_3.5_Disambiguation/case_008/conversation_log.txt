import json
import os
from pymatgen.core import Structure
from pymatgen.entries.computed_entries import ComputedEntry
from pymatgen.analysis.defects.formation_energy_diagram import DefectFormationEnergyDiagram

def compute_defect_properties():
    """
    Compute defect-related properties using Pymatgen for a GaN system.

    Returns:
        dict: A dictionary containing computed defect properties:
            - 'chempot_limits_count': Number of chemical potential limits
            - 'defect_chemsys': Chemical system of defects
            - 'bulk_formula': Chemical formula of the bulk material
    """
    # Define file paths
    file_path = os.path.dirname(__file__)
    structure_file = os.path.join(file_path, 'GaN.vasp')
    entries_file = os.path.join(file_path, 'stable_entries_Mg_Ga_N.json')

    # Read structure
    bulk_structure = Structure.from_file(structure_file)

    # Read and convert stable entries
    with open(entries_file, 'r') as f:
        entry_data = json.load(f)
        stable_entries = [ComputedEntry.from_dict(entry) for entry in entry_data]

    # Create formation energy diagram
    defect_diagram = DefectFormationEnergyDiagram(
        bulk_structure, stable_entries
    )

    # Compute properties
    properties = {
        'chempot_limits_count': len(defect_diagram.chemical_potential_limits),
        'defect_chemsys': '-'.join(sorted(set(bulk_structure.composition.elements))),
        'bulk_formula': bulk_structure.composition.reduced_formula
    }

    return properties
