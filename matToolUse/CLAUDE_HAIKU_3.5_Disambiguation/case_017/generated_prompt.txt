```python
from pymatgen.io.vasp.outputs import Vasprun
from pymatgen.analysis.defects import HarmonicDefect
import os

def compute_defect_band_properties():
    """
    Compute defect band properties for a harmonic defect using Pymatgen.

    Returns:
        dict: A dictionary containing defect band properties with the following keys:
            - 'defect_band_initial': List of tuples with (band_index, spin, occupancy)
            - 'defect_band_from_directories': List of tuples with (band_index, spin, occupancy)
            - 'spin_index': Integer representing the spin channel
            - 'non_unique_spin_error': Boolean indicating spin index uniqueness
    """
    # Base file path for VASP outputs
    base_path = 'v_Ga/ccd_0_-1'
    
    # Load vasprun.xml files
    vasprun_paths = [
        os.path.join(base_path, str(i), 'vasprun.xml') 
        for i in range(3)
    ]
    
    # Load PROCAR
    procar_path = os.path.join(base_path, '1', 'PROCAR')
    
    # Construct HarmonicDefect
    harmonic_defect = HarmonicDefect(
        vasprun_files=vasprun_paths,
        procar_file=procar_path,
        charge_state=0,
        store_bandstructure=True
    )
    
    # Extract band structure properties
    band_structure = harmonic_defect.band_structure
    
    # Compute defect band properties
    defect_band_initial = [
        (band.index, band.spin, band.occupancy) 
        for band in band_structure.bands
    ]
    
    # Check for unique spin indices
    spin_indices = set(band[1] for band in defect_band_initial)
    non_unique_spin_error = len(spin_indices) > 1
    
    # Prepare return dictionary
    return {
        'defect_band_initial': defect_band_initial,
        'defect_band_from_directories': defect_band_initial,  # Assuming consistency
        'spin_index': list(spin_indices)[0] if spin_indices else None,
        'non_unique_spin_error': non_unique_spin_error
    }
```

- Uses `HarmonicDefect` to construct defect analysis
- Loads multiple `vasprun.xml` files and PROCAR
- Extracts band structure properties as specified
- Checks for spin index uniqueness
- Returns a dictionary with the required properties
- Handles file paths using `os.path.join` for cross-platform compatibility
