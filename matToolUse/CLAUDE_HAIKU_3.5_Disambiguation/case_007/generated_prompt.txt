import os
import numpy as np
from pymatgen.core import Structure
from pymatgen.io.vasp import Vasprun, Locpot
from pymatgen.analysis.defects.core import DefectEntry
from pymatgen.analysis.defects.corrections import FreysoldtCorrection

def compute_mg_ga_defect_properties(file_path):
    """
    Compute defect properties for Mg substitutional defect in GaN system.

    Returns a dictionary with the following properties:
    - freysoldt_correction: Electrostatic correction for charged defect
    - potential_alignment_consistency: Consistency of potential alignment
    - energy_difference: Energy difference between defect and bulk supercells

    Args:
        file_path (str): Base directory containing defect and bulk data files

    Returns:
        dict: Computed defect properties
    """
    # Load base structure
    base_structure = Structure.from_file(os.path.join(file_path, 'GaN.vasp'))

    # Charge states to analyze
    charge_states = [-2, -1, 0, 1]

    # Compute Freysoldt corrections and potential alignment
    freysoldt_corrections = []
    potential_alignments = []

    for charge in charge_states:
        # Load defect vasprun and LOCPOT
        defect_dir = os.path.join(file_path, 'Mg_Ga', f'charge_{charge}')
        vasprun = Vasprun(os.path.join(defect_dir, 'vasprun.xml.gz'))
        locpot = Locpot.from_file(os.path.join(defect_dir, 'LOCPOT.gz'))

        # Create DefectEntry
        defect_entry = DefectEntry(base_structure, charge)

        # Compute Freysoldt correction
        freysoldt_corr = FreysoldtCorrection(defect_entry, locpot).correction
        freysoldt_corrections.append(freysoldt_corr)

        # Extract potential alignment
        potential_alignments.append(vasprun.final_energy)

    # Compute Freysoldt correction (average of charge states)
    freysoldt_correction = np.mean(freysoldt_corrections)

    # Check potential alignment consistency
    potential_alignment_consistency = np.std(potential_alignments) < 0.05

    # Compute energy difference
    defect_energy = Vasprun(os.path.join(file_path, 'Mg_Ga', 'charge_0', 'vasprun.xml.gz')).final_energy
    bulk_energy = Vasprun(os.path.join(file_path, 'bulk_sc', 'LOCPOT.gz')).final_energy
    energy_difference = abs(defect_energy - bulk_energy)

    return {
        'freysoldt_correction': freysoldt_correction,
        'potential_alignment_consistency': potential_alignment_consistency,
        'energy_difference': energy_difference
    }
