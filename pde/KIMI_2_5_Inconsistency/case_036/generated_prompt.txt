import numpy as np
import matplotlib.pyplot as plt
from scipy.sparse import diags
from scipy.sparse.linalg import spsolve

# Parameters
alpha = 0.1
T_final = 0.5
Nx = 200
dx = 1.0 / Nx
dt = 0.0005
Nt = int(T_final / dt)  # Should be 1000, but let's verify: 0.5/0.0005 = 1000

# Grid
x = np.linspace(0, 1, Nx + 1)  # Nx+1 points including boundaries

# Initial condition
u = np.zeros(Nx + 1)

# Crank-Nicolson setup
# Interior points: i = 1, ..., Nx-1
# r = alpha * dt / (2 * dx^2) for CN
r = alpha * dt / (2 * dx**2)

# Coefficients for tridiagonal system
# (1 + 2r) * u_i^{n+1} - r * u_{i-1}^{n+1} - r * u_{i+1}^{n+1} = (1 - 2r) * u_i^n + r * u_{i-1}^n + r * u_{i+1}^n

# Build matrices for interior (size Nx-1)
n_int = Nx - 1  # number of interior points

# Left-hand side matrix (implicit part)
main_diag_lhs = np.full(n_int, 1 + 2*r)
off_diag_lhs = np.full(n_int - 1, -r)
A = diags([off_diag_lhs, main_diag_lhs, off_diag_lhs], [-1, 0, 1], format='csr')

# Right-hand side matrix (explicit part)
main_diag_rhs = np.full(n_int, 1 - 2*r)
off_diag_rhs = np.full(n_int - 1, r)
B = diags([off_diag_rhs, main_diag_rhs, off_diag_rhs], [-1, 0, 1], format='csr')

# Time stepping
for n in range(Nt):
    t = n * dt
    t_next = (n + 1) * dt
    
    # Boundary values at current and next time
    u_left_n = np.sin(2 * np.pi * t)      # u(0, t_n)
    u_left_np1 = np.sin(2 * np.pi * t_next)  # u(0, t_{n+1})
    u_right_n = 0.0                        # u(1, t_n)
    u_right_np1 = 0.0                     # u(1, t_{n+1})
    
    # Interior values at time n
    u_int = u[1:Nx]  # u_1 to u_{Nx-1}
    
    # Compute RHS: B * u_int + boundary contributions
    rhs = B.dot(u_int)
    
    # Add boundary contributions to RHS
    # For i=1 (leftmost interior): + r * u_0^n from explicit part
    rhs[0] += r * u_left_n
    # For i=Nx-1 (rightmost interior): + r * u_Nx^n from explicit part
    rhs[-1] += r * u_right_n
    
    # Solve for interior at n+1
    u_int_new = spsolve(A, rhs)
    
    # Update solution
    u[0] = u_left_np1      # Enforce left BC
    u[1:Nx] = u_int_new    # Interior
    u[Nx] = u_right_np1    # Enforce right BC

# Final time should be T=0.5
print(f"Final time: {Nt * dt}")

# Compute diagnostics
u_min = np.min(u)
u_max = np.max(u)
L2_norm = np.sqrt(np.sum(u**2) * dx)

print(f"min(u) = {u_min}")
print(f"max(u) = {u_max}")
print(f"L2 norm = {L2_norm}")

# Save to NPZ
np.savez("solution_agent_heat_07.npz", 
         x=x, 
         u=u, 
         T=T_final, 
         dx=dx, 
         dt=dt)

# Save plot
plt.figure(figsize=(10, 6))
plt.plot(x, u, 'b-', linewidth=1.5)
plt.xlabel('x')
plt.ylabel('u(x, T=0.5)')
plt.title('Solution of Heat Equation at T=0.5')
plt.grid(True)
plt.tight_layout()
plt.savefig("solution_agent_heat_07.png", dpi=150)
plt.close()

print("Files saved: solution_agent_heat_07.npz, solution_agent_heat_07.png")
