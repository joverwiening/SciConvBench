Solve the 2D wave equation u_tt = u_xx + u_yy (wave speed c=1.0) numerically using finite differences.

Domain is (x,y) in [0,1]x[0,1], t in [0,1.0].

Use Dirichlet boundary conditions u=0 on all four boundaries.

Initial condition is u(x,y,0) = sin(pi*x)*sin(pi*y) and u_t(x,y,0) = 0.

Use Nx=Ny=120 uniform intervals so the spatial grid has 121x121 points including endpoints.

Set dx = dy = 1.0/120.

Compute dt0 = 0.9 / (c * sqrt(1/dx**2 + 1/dy**2)), set Nt = ceil(1.0/dt0), then dt = 1.0/Nt so the simulation hits T=1.0 exactly.

Use the 5-point Laplacian stencil: u_xx + u_yy at interior point (i,j) approximated as (u[j,i+1] - 2*u[j,i] + u[j,i-1])/dx**2 + (u[j+1,i] - 2*u[j,i] + u[j-1,i])/dy**2.

Use 2nd-order central difference in time: u_new = 2*u_cur - u_old + dt**2 * laplacian(u_cur).

Initialize the first time step using the Taylor expansion: u1[j,i] = u0[j,i] + 0*dt + 0.5*dt**2 * laplacian(u0)[j,i], since u_t(x,y,0)=0 so the velocity term vanishes.

Enforce u=0 on all boundary nodes (i=0, i=Nx, j=0, j=Ny) at every time step including after the Taylor start step.

After completing all Nt time steps, the array u holds u(x,y,T=1.0).

Compute the exact solution u_exact[j,i] = cos(1.0*sqrt(2)*pi*T)*sin(pi*x[i])*sin(pi*y[j]) on the same grid.

Print min(u), max(u), and discrete L2 norm sqrt(sum(u**2)*dx*dy).

Print max absolute error max(abs(u - u_exact)) and L2 error sqrt(sum((u - u_exact)**2)*dx*dy).

Save file "solution_agent_wave_07.npz" containing: x as 1D array of length 121, y as 1D array of length 121, u as 2D array of shape (121,121) where u[j,i] = u(x[i],y[j]) at T, and scalar metadata T=1.0, dx, dy, dt.

Save a heatmap or filled contour plot of u(x,y,T) as "solution_agent_wave_07.png" with labeled axes and a colorbar.

Use Python with numpy and matplotlib. Use '^' interpreted as '**' in all math expressions when writing code.

Here is the complete Python implementation:

```python
import numpy as np
import matplotlib.pyplot as plt
import math

# Parameters
c = 1.0
T = 1.0
Nx = 120
Ny = 120

dx = 1.0 / Nx
dy = 1.0 / Ny

# Grid points (121 x 121)
x = np.linspace(0.0, 1.0, Nx + 1)
y = np.linspace(0.0, 1.0, Ny + 1)

# Time step from CFL condition
dt0 = 0.9 / (c * math.sqrt(1.0/dx**2 + 1.0/dy**2))
Nt = math.ceil(T / dt0)
dt = T / Nt

print(f"dx={dx:.6f}, dy={dy:.6f}, dt={dt:.6f}, Nt={Nt}")
print(f"CFL = {c*dt*math.sqrt(1/dx**2+1/dy**2):.6f}")

# Meshgrid: u[j, i] = u(x[i], y[j])
X, Y = np.meshgrid(x, y)  # shape (Ny+1, Nx+1)

# Initial condition
u0 = np.sin(np.pi * X) * np.sin(np.pi * Y)

# Laplacian function (interior only, boundaries stay 0)
def laplacian(u):
    lap = np.zeros_like(u)
    lap[1:-1, 1:-1] = (
        (u[1:-1, 2:] - 2.0*u[1:-1, 1:-1] + u[1:-1, :-2]) / dx**2 +
        (u[2:, 1:-1] - 2.0*u[1:-1, 1:-1] + u[:-2, 1:-1]) / dy**2
    )
    return lap

# Taylor start for first step (u_t = 0 at t=0)
lap0 = laplacian(u0)
u1 = u0 + 0.5 * dt**2 * lap0
# Enforce Dirichlet BC
u1[0, :] = 0.0
u1[-1, :] = 0.0
u1[:, 0] = 0.0
u1[:, -1] = 0.0

# Time stepping
u_old = u0.copy()
u_cur = u1.copy()

for n in range(1, Nt):
    lap_cur = laplacian(u_cur)
    u_new = 2.0 * u_cur - u_old + dt**2 * lap_cur
    # Enforce Dirichlet BC
    u_new[0, :] = 0.0
    u_new[-1, :] = 0.0
    u_new[:, 0] = 0.0
    u_new[:, -1] = 0.0
    u_old = u_cur
    u_cur = u_new

u_final = u_cur

# Exact solution at T
u_exact = np.cos(1.0 * math.sqrt(2) * np.pi * T) * np.sin(np.pi * X) * np.sin(np.pi * Y)

# Diagnostics
print(f"min(u) = {u_final.min():.8f}")
print(f"max(u) = {u_final.max():.8f}")
L2_num = math.sqrt(np.sum(u_final**2) * dx * dy)
print(f"Discrete L2 norm = {L2_num:.8f}")

max_abs_err = np.max(np.abs(u_final - u_exact))
L2_err = math.sqrt(np.sum((u_final - u_exact)**2) * dx * dy)
print(f"Max absolute error = {max_abs_err:.8e}")
print(f"L2 error vs exact  = {L2_err:.8e}")

# Save npz
np.savez("solution_agent_wave_07.npz",
         x=x, y=y, u=u_final,
         T=np.float64(T),
         dx=np.float64(dx),
         dy=np.float64(dy),
         dt=np.float64(dt))
print("Saved solution_agent_wave_07.npz")

# Save plot
fig, ax = plt.subplots(figsize=(6, 5))
cp = ax.contourf(X, Y, u_final, levels=50, cmap='RdBu_r')
fig.colorbar(cp, ax=ax, label='u(x,y,T)')
ax.set_xlabel('x')
ax.set_ylabel('y')
ax.set_title(f'Wave equation solution at T={T}')
plt.tight_layout()
plt.savefig("solution_agent_wave_07.png", dpi=150)
plt.close()
print("Saved solution_agent_wave_07.png")
```
