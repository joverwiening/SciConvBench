Import numpy and matplotlib.pyplot.
Set Nx=512, dx=1/Nx, dt=0.001, Nt=1000, nu=10.0, rho=5.0, T=1.0.
Create spatial grid x = np.arange(Nx)*dx.
Initialize u = 0.25 + 0.1*np.sin(2*np.pi*x).
Create frequency array k = 2*np.pi*np.fft.fftfreq(Nx, dx).
Create Fourier diffusion operator exp_diff = np.exp(-nu*k**2*dt/2).
For each time step from 0 to Nt-1:
    Apply diffusion half-step: u = np.real(np.fft.ifft(np.fft.fft(u) * exp_diff)).
    Apply reaction full-step: u = (u*np.exp(rho*dt))/(1 + u*(np.exp(rho*dt) - 1)).
    Apply diffusion half-step: u = np.real(np.fft.ifft(np.fft.fft(u) * exp_diff)).
Save arrays x, u and scalars nu, rho, T, dx, dt, Nx, Nt to solution_pdebench_reacdiff_nu10_rho5.npz.
Create line plot of u vs x and save as solution_pdebench_reacdiff_nu10_rho5.png.
Print min(u), max(u), mean(u), and sqrt(sum(u**2)*dx) at final time.
