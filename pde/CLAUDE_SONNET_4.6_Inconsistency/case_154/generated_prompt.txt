```python
import numpy as np
import matplotlib.pyplot as plt

# ── parameters ────────────────────────────────────────────────────────────────
nu   = 0.004
T    = 1.0
Nx   = 512
CFL  = 0.4

# ── grid ──────────────────────────────────────────────────────────────────────
dx   = 1.0 / Nx
x    = np.arange(Nx) * dx          # x_j = j*dx, j = 0 … Nx-1

# ── time step (CFL-based, adjusted so Nt*dt == T exactly) ────────────────────
dt0  = CFL * dx / 1.0              # max speed ≈ 1
Nt   = int(np.ceil(T / dt0))
dt   = T / Nt

print(f"Nx={Nx}, dx={dx:.6e}, Nt={Nt}, dt={dt:.6e}, nu={nu}")

# ── wavenumbers for spectral diffusion ───────────────────────────────────────
k    = 2.0 * np.pi * np.fft.fftfreq(Nx, d=dx)   # shape (Nx,)

# ── precompute diffusion multipliers ─────────────────────────────────────────
diff_half = np.exp(-nu * k**2 * (dt / 2.0))     # half-step multiplier
# (full-step would be diff_half**2, but we apply two half-steps)

# ── helper: diffusion half-step in Fourier space ─────────────────────────────
def diffusion_half(u):
    u_hat = np.fft.fft(u)
    u_hat *= diff_half
    return np.fft.ifft(u_hat).real

# ── helper: Rusanov finite-volume advection full-step ────────────────────────
def advection_full(u):
    # f(u) = 0.5*u**2
    f      = 0.5 * u**2

    # right-shifted u: u_right[j] = u[j+1]  (periodic)
    u_r    = np.roll(u, -1)
    f_r    = np.roll(f, -1)

    # Rusanov flux at right interface of cell j  →  F_plus[j]
    alpha_r        = np.maximum(np.abs(u), np.abs(u_r))
    F_plus         = 0.5 * (f + f_r) - 0.5 * alpha_r * (u_r - u)

    # flux at left interface of cell j  =  F_plus[j-1]  (periodic)
    F_minus        = np.roll(F_plus, 1)

    return u - (dt / dx) * (F_plus - F_minus)

# ── initial condition ─────────────────────────────────────────────────────────
u = np.sin(2.0 * np.pi * x)

# ── time integration: Strang splitting  A – B – A ────────────────────────────
for n in range(Nt):
    u = diffusion_half(u)      # (A) diffusion  dt/2
    u = advection_full(u)      # (B) advection  dt
    u = diffusion_half(u)      # (C) diffusion  dt/2

# ── diagnostics ───────────────────────────────────────────────────────────────
u_min   = u.min()
u_max   = u.max()
u_mean  = u.mean()
u_L2    = np.sqrt(np.sum(u**2) * dx)

print(f"T        = {T}")
print(f"min(u)   = {u_min:.6e}")
print(f"max(u)   = {u_max:.6e}")
print(f"mean(u)  = {u_mean:.6e}   (should be ≈ 0)")
print(f"L2 norm  = {u_L2:.6e}")

# ── save .npz ─────────────────────────────────────────────────────────────────
np.savez(
    "solution_pdebench_burgers_nu4em3.npz",
    x   = x,
    u   = u,
    nu  = nu,
    T   = T,
    dx  = dx,
    dt  = dt,
    Nx  = Nx,
    Nt  = Nt,
)
print("Saved solution_pdebench_burgers_nu4em3.npz")

# ── save plot ─────────────────────────────────────────────────────────────────
fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(x, u, linewidth=1.2, color="steelblue")
ax.set_xlabel("x")
ax.set_ylabel("u(x, T)")
ax.set_title(
    f"1D Viscous Burgers  |  nu={nu},  T={T},  Nx={Nx},  Nt={Nt}\n"
    f"min={u_min:.4f},  max={u_max:.4f},  mean={u_mean:.2e},  L2={u_L2:.4f}"
)
ax.grid(True, linewidth=0.4, alpha=0.6)
fig.tight_layout()
fig.savefig("solution_pdebench_burgers_nu4em3.png", dpi=150)
plt.close(fig)
print("Saved solution_pdebench_burgers_nu4em3.png")
```
