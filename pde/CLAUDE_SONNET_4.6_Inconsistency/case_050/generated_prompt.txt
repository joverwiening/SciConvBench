```python
import numpy as np
import matplotlib.pyplot as plt

# Parameters
c = 1.0
m = 0.5
T = 1.1
Nx = 400
x = np.linspace(0, 1, Nx + 1)  # 401 points
dx = 1.0 / Nx

# Time stepping
dt0 = 0.9 * dx / c
Nt = int(np.ceil(T / dt0))
dt = T / Nt

print(f"dx = {dx}, dt = {dt}, Nt = {Nt}, CFL = {c*dt/dx:.4f}")

# Initial conditions
u_old = np.cos(np.pi * x)  # u(x, 0)
# v0 = u_t(x, 0) = 0

# Compute u_xx of u0 using central differences with ghost points (Neumann BC)
def apply_laplacian(u, dx):
    u_xx = np.zeros_like(u)
    # Interior points
    u_xx[1:-1] = (u[:-2] - 2*u[1:-1] + u[2:]) / dx**2
    # Boundary points using ghost points (mirror)
    # x=0: ghost u_{-1} = u_{1}
    u_xx[0] = (u[1] - 2*u[0] + u[1]) / dx**2  # u_{-1}=u_{1}
    # x=1: ghost u_{Nx+1} = u_{Nx-1}
    u_xx[-1] = (u[-2] - 2*u[-1] + u[-2]) / dx**2  # u_{Nx+1}=u_{Nx-1}
    return u_xx

# Taylor start: u1 = u0 + dt*v0 + 0.5*dt^2*(c^2*u_xx - m^2*u0)
u_xx0 = apply_laplacian(u_old, dx)
u_tt0 = c**2 * u_xx0 - m**2 * u_old
u_new = u_old + 0.0 * dt + 0.5 * dt**2 * u_tt0  # v0=0

# Time stepping (leapfrog)
for n in range(1, Nt):
    u_xx = apply_laplacian(u_new, dx)
    u_next = 2*u_new - u_old + dt**2 * (c**2 * u_xx - m**2 * u_new)
    u_old = u_new
    u_new = u_next

u = u_new

# Exact solution
omega = np.sqrt((np.pi)**2 + m**2)
u_exact = np.cos(omega * T) * np.cos(np.pi * x)

# Diagnostics
print(f"min(u) = {np.min(u):.6e}")
print(f"max(u) = {np.max(u):.6e}")
L2_u = np.sqrt(np.sum(u**2) * dx)
print(f"L2 norm of u = {L2_u:.6e}")

max_abs_err = np.max(np.abs(u - u_exact))
L2_err = np.sqrt(np.sum((u - u_exact)**2) * dx)
print(f"Max absolute error = {max_abs_err:.6e}")
print(f"L2 error = {L2_err:.6e}")

# Save npz
np.savez("solution_agent_kg_03.npz", x=x, u=u, T=T, dx=dx, dt=dt)
print("Saved solution_agent_kg_03.npz")

# Save plot
plt.figure(figsize=(8, 5))
plt.plot(x, u, label=f"Numerical u(x, T={T})")
plt.plot(x, u_exact, '--', label="Exact solution")
plt.xlabel("x")
plt.ylabel("u")
plt.title(f"Klein-Gordon solution at T={T}")
plt.legend()
plt.tight_layout()
plt.savefig("solution_agent_kg_03.png", dpi=150)
plt.close()
print("Saved solution_agent_kg_03.png")
```
