```python
import numpy as np
import matplotlib.pyplot as plt
import math

# Parameters
a = -0.7   # advection speed in x
b = 1.2    # advection speed in y
T = 0.6    # final time (from output requirement)

# Grid
Nx = 256
Ny = 256
dx = 1.0 / Nx
dy = 1.0 / Ny

x = np.arange(Nx) * dx  # shape (Nx,)
y = np.arange(Ny) * dy  # shape (Ny,)

# Time stepping
cfl_target = 0.9
dt0 = cfl_target / (abs(a)/dx + abs(b)/dy)
Nt = math.ceil(T / dt0)
dt = T / Nt

print(f"dx={dx}, dy={dy}, dt0={dt0:.6f}, Nt={Nt}, dt={dt:.6f}")
print(f"CFL number: {abs(a)*dt/dx + abs(b)*dt/dy:.6f}")

# Initial condition: u(x,y,0) = exp(-80*((x-0.2)^2 + (y-0.6)^2))
# u[j, i] = u(x[i], y[j])
X, Y = np.meshgrid(x, y)  # X shape (Ny, Nx), Y shape (Ny, Nx)
u = np.exp(-80.0 * ((X - 0.2)**2 + (Y - 0.6)**2))

# Time integration: Forward Euler with upwind scheme
# a < 0: right-biased in x => u_x ~ (u[j, i+1] - u[j, i]) / dx
# b > 0: left-biased in y  => u_y ~ (u[j, i] - u[j-1, i]) / dy

for n in range(Nt):
    # Upwind differences
    # x-direction: a < 0, right-biased: (u[:, i+1] - u[:, i]) / dx
    u_x = (np.roll(u, -1, axis=1) - u) / dx  # roll left => i+1
    # y-direction: b > 0, left-biased: (u[j, :] - u[j-1, :]) / dy
    u_y = (u - np.roll(u, 1, axis=0)) / dy   # roll right => j-1

    u = u - dt * (a * u_x + b * u_y)

# Diagnostics
u_min = np.min(u)
u_max = np.max(u)
l2_norm = np.sqrt(np.sum(u**2) * dx * dy)

print(f"min(u)    = {u_min:.6e}")
print(f"max(u)    = {u_max:.6e}")
print(f"L2 norm   = {l2_norm:.6e}")

# Save npz
np.savez("solution_agent_advection_05.npz",
         x=x, y=y, u=u,
         T=np.float64(T),
         dx=np.float64(dx),
         dy=np.float64(dy),
         dt=np.float64(dt))

print("Saved solution_agent_advection_05.npz")

# Save plot
fig, ax = plt.subplots(figsize=(7, 6))
cp = ax.contourf(X, Y, u, levels=50, cmap='viridis')
fig.colorbar(cp, ax=ax, label='u(x,y,T)')
ax.set_xlabel('x')
ax.set_ylabel('y')
ax.set_title(f'u(x,y,T={T}) — 2D Advection (upwind + Forward Euler)')
plt.tight_layout()
plt.savefig("solution_agent_advection_05.png", dpi=150)
plt.close()
print("Saved solution_agent_advection_05.png")
```
