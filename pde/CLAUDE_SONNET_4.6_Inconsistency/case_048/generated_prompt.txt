```python
import numpy as np
import matplotlib.pyplot as plt
from math import ceil, sqrt, pi

# ── parameters ────────────────────────────────────────────────────────────────
c   = 2.0
m   = 1.0
T   = 1.0
Nx  = 400
dx  = 1.0 / Nx
dt0 = 0.9 * dx / c
Nt  = ceil(T / dt0)
dt  = T / Nt

print(f"Nx={Nx}, dx={dx:.6e}, Nt={Nt}, dt={dt:.6e}, CFL={c*dt/dx:.4f}")

# ── grid ──────────────────────────────────────────────────────────────────────
x = np.linspace(0.0, 1.0, Nx + 1)          # 401 points

# ── initial conditions ────────────────────────────────────────────────────────
u0 = np.sin(pi * x)                         # u(x,0)
v0 = np.zeros(Nx + 1)                       # u_t(x,0) = 0

# ── helper: interior Laplacian (2nd-order central differences) ────────────────
def laplacian(u):
    """Returns c^2 * u_xx - m^2 * u at interior points (indices 1..Nx-1)."""
    uxx = (u[:-2] - 2.0*u[1:-1] + u[2:]) / dx**2
    return c**2 * uxx - m**2 * u[1:-1]

# ── Taylor start (2nd-order) ──────────────────────────────────────────────────
u_prev = u0.copy()
u_curr = u0.copy()
u_curr[1:-1] = (u0[1:-1]
                + dt * v0[1:-1]
                + 0.5 * dt**2 * laplacian(u0))
# boundaries stay zero
u_curr[0]  = 0.0
u_curr[-1] = 0.0

# ── leapfrog time-stepping ────────────────────────────────────────────────────
u_next = np.empty(Nx + 1)

for n in range(1, Nt):          # we already have levels 0 and 1; march to Nt
    u_next[1:-1] = (2.0 * u_curr[1:-1]
                    - u_prev[1:-1]
                    + dt**2 * laplacian(u_curr))
    u_next[0]  = 0.0
    u_next[-1] = 0.0
    u_prev, u_curr = u_curr, u_next.copy()

u_final = u_curr                # solution at t = T

# ── exact solution ────────────────────────────────────────────────────────────
omega   = sqrt((pi)**2 + m**2)          # sqrt((1.0*pi)^2 + (1.0)^2)
u_exact = np.cos(omega * T) * np.sin(pi * x)

# ── diagnostics ───────────────────────────────────────────────────────────────
u_min  = u_final.min()
u_max  = u_final.max()
l2_num = sqrt(np.sum(u_final**2) * dx)

max_err = np.max(np.abs(u_final - u_exact))
l2_err  = sqrt(np.sum((u_final - u_exact)**2) * dx)

print(f"min(u)          = {u_min:.6e}")
print(f"max(u)          = {u_max:.6e}")
print(f"L2 norm (num)   = {l2_num:.6e}")
print(f"Max abs error   = {max_err:.6e}")
print(f"L2 error        = {l2_err:.6e}")

# ── save .npz ─────────────────────────────────────────────────────────────────
np.savez("solution_agent_kg_01.npz",
         x=x, u=u_final, T=T, dx=dx, dt=dt)
print("Saved solution_agent_kg_01.npz")

# ── save plot ─────────────────────────────────────────────────────────────────
fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(x, u_final,  lw=2,   label="Numerical")
ax.plot(x, u_exact,  lw=1.5, ls="--", label="Exact")
ax.set_xlabel("x")
ax.set_ylabel("u(x, T)")
ax.set_title(f"Klein-Gordon solution at T={T}  (Nx={Nx}, Nt={Nt})")
ax.legend()
fig.tight_layout()
fig.savefig("solution_agent_kg_01.png", dpi=150)
plt.close(fig)
print("Saved solution_agent_kg_01.png")
```
