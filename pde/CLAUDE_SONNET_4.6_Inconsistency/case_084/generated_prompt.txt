Solve the 1D wave equation u_tt = (1.0)^2 * u_xx numerically with the following specifications.

The spatial domain is x in [0, 1] (closed interval).
The time domain is t in [0, 0.8].

For all subsequent time steps n >= 1, advance using the leapfrog scheme: u^{n+1}_i = 2*u^n_i - u^{n-1}_i + (c*dt/dx)**2 * (u^n_{i+1} - 2*u^n_i + u^n_{i-1}) for interior nodes i = 1 to Nx-1.

```python
import numpy as np
import matplotlib.pyplot as plt

# ── parameters ────────────────────────────────────────────────────────────────
c      = 2.0          # wave speed used in PDE / CFL
T      = 0.8
Nx     = 600
x      = np.linspace(0.0, 1.0, Nx + 1)   # 601 points
dx     = 1.0 / Nx

dt0    = 0.9 * dx / c
Nt     = int(np.ceil(T / dt0))
dt     = T / Nt                           # adjusted so t_Nt = T exactly
r      = (c * dt / dx) ** 2              # Courant number squared

print(f"Nx={Nx}, dx={dx:.6e}, Nt={Nt}, dt={dt:.6e}, CFL={c*dt/dx:.6f}")

# ── initial conditions ────────────────────────────────────────────────────────
u0 = np.sin(np.pi * x) + 0.5 * np.sin(3.0 * np.pi * x)
v0 = np.zeros_like(x)                    # u_t(x,0) = 0

# ── helper: interior Laplacian (2nd-order central difference) ─────────────────
def laplacian(u):
    """Returns c^2 * u_xx at interior nodes 1..Nx-1 (length Nx-1)."""
    return c**2 * (u[:-2] - 2.0*u[1:-1] + u[2:]) / dx**2

# ── time level 0 ──────────────────────────────────────────────────────────────
u_prev = u0.copy()

# ── time level 1  (Taylor / Störmer–Verlet start) ────────────────────────────
u_curr = np.empty_like(u0)
u_curr[0]  = 0.0                          # Dirichlet
u_curr[-1] = 0.0
u_curr[1:-1] = (u_prev[1:-1]
                + dt * v0[1:-1]
                + 0.5 * dt**2 * laplacian(u_prev))

# ── leapfrog march ────────────────────────────────────────────────────────────
for n in range(1, Nt):
    u_next = np.empty_like(u_curr)
    u_next[0]  = 0.0
    u_next[-1] = 0.0
    u_next[1:-1] = (2.0 * u_curr[1:-1]
                    - u_prev[1:-1]
                    + r * (u_curr[:-2] - 2.0*u_curr[1:-1] + u_curr[2:]))
    u_prev = u_curr
    u_curr = u_next

u = u_curr   # solution at t = T

# ── exact solution (c_exact = 1.0 as written in the problem) ─────────────────
# u_exact = cos(pi*t)*sin(pi*x) + 0.5*cos(3*pi*t)*sin(3*pi*x)
u_exact = (np.cos(np.pi * T) * np.sin(np.pi * x)
           + 0.5 * np.cos(3.0 * np.pi * T) * np.sin(3.0 * np.pi * x))

# ── diagnostics ───────────────────────────────────────────────────────────────
L2_u      = np.sqrt(np.sum(u**2) * dx)
max_err   = np.max(np.abs(u - u_exact))
L2_err    = np.sqrt(np.sum((u - u_exact)**2) * dx)

print(f"min(u)        = {u.min():.8e}")
print(f"max(u)        = {u.max():.8e}")
print(f"L2 norm of u  = {L2_u:.8e}")
print(f"Max abs error = {max_err:.8e}")
print(f"L2 error      = {L2_err:.8e}")

# ── save npz ──────────────────────────────────────────────────────────────────
np.savez("solution_agent_wave_03.npz",
         x=x, u=u, T=np.float64(T), dx=np.float64(dx), dt=np.float64(dt))
print("Saved solution_agent_wave_03.npz")

# ── save plot ─────────────────────────────────────────────────────────────────
fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(x, u,       label=f"Numerical  (t={T})", linewidth=1.5)
ax.plot(x, u_exact, label="Exact",  linewidth=1.5, linestyle="--")
ax.set_xlabel("x")
ax.set_ylabel("u(x, T)")
ax.set_title(f"Wave equation solution at T={T}  (Nx={Nx}, Nt={Nt})")
ax.legend()
fig.tight_layout()
fig.savefig("solution_agent_wave_03.png", dpi=150)
plt.close(fig)
print("Saved solution_agent_wave_03.png")
```
