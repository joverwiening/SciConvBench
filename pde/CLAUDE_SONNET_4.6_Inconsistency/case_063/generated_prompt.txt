Solve the Laplace PDE u_xx + u_yy = 0 on the domain (x,y) in [0,1] x [0,1].
Use Dirichlet boundary conditions: u(x,0)=0, u(x,1)=sin(pi*x), u(0,y)=0, u(1,y)=0.
Ignore the periodic boundary condition statement; it contradicts the Dirichlet conditions and the exact solution, so use only the Dirichlet conditions.
The exact reference solution is u(x,y)=sin(pi*x)*sinh(pi*y)/sinh(pi).
Use a uniform grid with Nx=Ny=100 intervals, giving 101x101 grid points including boundaries.
Set dx=dy=1/100, x_i=i*dx for i=0..100, y_j=j*dy for j=0..100.
Discretize the Laplacian using the standard 5-point finite difference stencil (2nd-order centered differences) on interior nodes only.
Assemble the resulting sparse linear system for the (Nx-1)*(Ny-1)=99*99 interior unknowns.
Apply Dirichlet boundary values as known right-hand-side contributions when assembling the system.
Solve the sparse linear system using scipy.sparse.linalg.spsolve.
Construct the full (101x101) solution array u with shape (len(y), len(x)) so that u[j,i] = u(x_i, y_j), filling boundary rows/columns from the Dirichlet conditions and interior from the solver result.
Save "solution_agent_laplace_07.npz" containing: x (1D array of length 101), y (1D array of length 101), u (2D array of shape (101,101)), dx (scalar), dy (scalar).
Save a heatmap or filled-contour plot of u(x,y) as "solution_agent_laplace_07.png" with labeled axes and a colorbar.
Print min(u), max(u), and the discrete L2 norm computed as sqrt(sum(u**2)*dx*dy) over all grid points.
Compute the exact solution u_exact[j,i]=sin(pi*x_i)*sinh(pi*y_j)/sinh(pi) on the same 101x101 grid.
Print the max absolute error max(|u - u_exact|) and the L2 error sqrt(sum((u - u_exact)**2)*dx*dy).
Use Python with numpy and scipy. Use radians for all trigonometric functions. Use ** for exponentiation in code.

Here is the complete Python script implementing the specification above:

```python
import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# ── Grid ──────────────────────────────────────────────────────────────────────
Nx = Ny = 100
dx = dy = 1.0 / Nx
x = np.linspace(0, 1, Nx + 1)   # length 101
y = np.linspace(0, 1, Ny + 1)   # length 101

# ── Interior indices ──────────────────────────────────────────────────────────
# Interior nodes: i = 1..Nx-1, j = 1..Ny-1  (99 x 99 = 9801 unknowns)
Nix = Nx - 1   # 99
Niy = Ny - 1   # 99
N   = Nix * Niy

def idx(i, j):
    """Flat index for interior node (i,j), i=1..Nx-1, j=1..Ny-1."""
    return (j - 1) * Nix + (i - 1)

# ── Assemble sparse system ────────────────────────────────────────────────────
rows, cols, vals = [], [], []
rhs = np.zeros(N)

for j in range(1, Ny):          # j = 1..99
    for i in range(1, Nx):      # i = 1..99
        k = idx(i, j)
        # Centre coefficient
        rows.append(k); cols.append(k); vals.append(-4.0 / dx**2)

        # East  (i+1, j)
        if i + 1 <= Nx - 1:
            rows.append(k); cols.append(idx(i+1, j)); vals.append(1.0 / dx**2)
        else:   # i+1 = Nx  → right boundary u(1,y)=0
            pass   # contribution is 0

        # West  (i-1, j)
        if i - 1 >= 1:
            rows.append(k); cols.append(idx(i-1, j)); vals.append(1.0 / dx**2)
        else:   # i-1 = 0  → left boundary u(0,y)=0
            pass   # contribution is 0

        # North (i, j+1)
        if j + 1 <= Ny - 1:
            rows.append(k); cols.append(idx(i, j+1)); vals.append(1.0 / dy**2)
        else:   # j+1 = Ny → top boundary u(x,1)=sin(pi*x)
            rhs[k] -= np.sin(np.pi * x[i]) / dy**2

        # South (i, j-1)
        if j - 1 >= 1:
            rows.append(k); cols.append(idx(i, j-1)); vals.append(1.0 / dy**2)
        else:   # j-1 = 0  → bottom boundary u(x,0)=0
            pass   # contribution is 0

A = sp.csr_matrix((vals, (rows, cols)), shape=(N, N))

# ── Solve ─────────────────────────────────────────────────────────────────────
u_int = spla.spsolve(A, rhs)

# ── Assemble full solution array u[j,i] ──────────────────────────────────────
u = np.zeros((Ny + 1, Nx + 1))

# Boundary conditions
u[0,  :]  = 0.0                          # bottom  u(x,0)=0
u[-1, :]  = np.sin(np.pi * x)            # top     u(x,1)=sin(pi*x)
u[:,  0]  = 0.0                          # left    u(0,y)=0
u[:, -1]  = 0.0                          # right   u(1,y)=0

# Interior
for j in range(1, Ny):
    for i in range(1, Nx):
        u[j, i] = u_int[idx(i, j)]

# ── Exact solution ────────────────────────────────────────────────────────────
X, Y = np.meshgrid(x, y)   # shape (101,101)
u_exact = np.sin(np.pi * X) * np.sinh(np.pi * Y) / np.sinh(np.pi)

# ── Diagnostics ───────────────────────────────────────────────────────────────
print(f"min(u)          = {u.min():.6e}")
print(f"max(u)          = {u.max():.6e}")
print(f"L2 norm of u    = {np.sqrt(np.sum(u**2) * dx * dy):.6e}")

abs_err = np.abs(u - u_exact)
l2_err  = np.sqrt(np.sum((u - u_exact)**2) * dx * dy)
print(f"Max abs error   = {abs_err.max():.6e}")
print(f"L2 error        = {l2_err:.6e}")

# ── Save .npz ─────────────────────────────────────────────────────────────────
np.savez("solution_agent_laplace_07.npz",
         x=x, y=y, u=u, dx=dx, dy=dy)
print("Saved solution_agent_laplace_07.npz")

# ── Save plot ─────────────────────────────────────────────────────────────────
fig, ax = plt.subplots(figsize=(6, 5))
cp = ax.contourf(X, Y, u, levels=50, cmap='viridis')
fig.colorbar(cp, ax=ax, label='u(x,y)')
ax.set_xlabel('x')
ax.set_ylabel('y')
ax.set_title('Laplace solution  u(x,y)')
fig.tight_layout()
fig.savefig("solution_agent_laplace_07.png", dpi=150)
plt.close(fig)
print("Saved solution_agent_laplace_07.png")
```
