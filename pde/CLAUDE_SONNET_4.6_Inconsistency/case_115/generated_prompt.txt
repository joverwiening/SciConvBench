Solve the 1D Klein-Gordon-type PDE numerically using the specifications below.

The PDE is u_tt + (q2**2)*(q1**4)*u = (q1**2)*u_xx.
The spatial domain is x in [0, 1) with periodic boundary conditions.
The time domain is t in [0, 1.0].
Set q1 = 1.0 and q2 = 0.5.
Use a uniform periodic grid with Nx = 256 points on [0, 1), defined as x_j = j * dx for j = 0, ..., 255, with dx = 1.0 / 256.
Convert the PDE to a first-order system: let v = u_t, so u_t = v and v_t = (q1**2) * u_xx - (q2**2) * (q1**4) * u.
Compute u_xx using the Fourier spectral method via FFT on the periodic grid.
Apply RK4 time integration to the first-order system.
Use dt = 0.0005 and Nt = 2000 steps so the simulation ends exactly at T = 1.0.
Set the initial condition u(x, 0) = sin(2 * pi * x) and u_t(x, 0) = 0 for all grid points.
The exact solution for verification is u(x, t) = cos(omega * t) * sin(2 * pi * x) where omega = sqrt(q2**2 * q1**4 + q1**2 * (2 * pi)**2) = 6.303048278758258.
At T = 1.0, compute u on the full periodic grid.
Print min(u), max(u), mean(u), and the discrete L2 norm sqrt(sum(u**2) * dx).
Save a file named "solution_molk_014.npz" containing the 1D array x of length 256, the 1D array u of length 256, and the scalars T, dx, Nx, q1, q2.
Save a line plot of u(x, T) as "solution_molk_014.png" with x on the horizontal axis and u on the vertical axis.
Use Python with numpy and matplotlib.
Use radians for all trigonometric functions.
Use ** for exponentiation in code.

```python
import numpy as np
import matplotlib.pyplot as plt

# Parameters
q1 = 1.0
q2 = 0.5
Nx = 256
dx = 1.0 / Nx
x = np.array([j * dx for j in range(Nx)])
dt = 0.0005
Nt = 2000
T = 1.0

# Wavenumbers for spectral derivative on periodic domain of length 1
k = np.fft.fftfreq(Nx, d=1.0/Nx)  # k = 0, 1, ..., Nx/2-1, -Nx/2, ..., -1
k_wave = 2.0 * np.pi * k           # actual wavenumbers

def compute_uxx_spectral(u):
    u_hat = np.fft.fft(u)
    uxx_hat = -(k_wave**2) * u_hat
    return np.fft.ifft(uxx_hat).real

# Initial conditions
u0 = np.sin(2.0 * np.pi * x)
v0 = np.zeros(Nx)

def rhs(u, v):
    uxx = compute_uxx_spectral(u)
    du_dt = v
    dv_dt = (q1**2) * uxx - (q2**2) * (q1**4) * u
    return du_dt, dv_dt

def rk4_step(u, v, dt):
    k1u, k1v = rhs(u, v)
    k2u, k2v = rhs(u + 0.5*dt*k1u, v + 0.5*dt*k1v)
    k3u, k3v = rhs(u + 0.5*dt*k2u, v + 0.5*dt*k2v)
    k4u, k4v = rhs(u + dt*k3u, v + dt*k3v)
    u_new = u + (dt/6.0) * (k1u + 2.0*k2u + 2.0*k3u + k4u)
    v_new = v + (dt/6.0) * (k1v + 2.0*k2v + 2.0*k3v + k4v)
    return u_new, v_new

# Time integration
u = u0.copy()
v = v0.copy()

for n in range(Nt):
    u, v = rk4_step(u, v, dt)

# Diagnostics
u_min = np.min(u)
u_max = np.max(u)
u_mean = np.mean(u)
l2_norm = np.sqrt(np.sum(u**2) * dx)

print(f"min(u)   = {u_min:.10f}")
print(f"max(u)   = {u_max:.10f}")
print(f"mean(u)  = {u_mean:.10f}")
print(f"L2 norm  = {l2_norm:.10f}")

# Exact solution for comparison
omega = np.sqrt(q2**2 * q1**4 + q1**2 * (2.0 * np.pi)**2)
u_exact = np.cos(omega * T) * np.sin(2.0 * np.pi * x)
l2_error = np.sqrt(np.sum((u - u_exact)**2) * dx)
print(f"omega    = {omega:.15f}")
print(f"L2 error vs exact = {l2_error:.6e}")

# Save npz
np.savez("solution_molk_014.npz",
         x=x,
         u=u,
         T=np.float64(T),
         dx=np.float64(dx),
         Nx=np.int64(Nx),
         q1=np.float64(q1),
         q2=np.float64(q2))

# Save plot
fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(x, u, label=f"Numerical u(x, T={T})", color="steelblue", linewidth=1.5)
ax.plot(x, u_exact, label="Exact solution", color="tomato", linewidth=1.5, linestyle="--")
ax.set_xlabel("x")
ax.set_ylabel("u(x, T)")
ax.set_title(f"1D Klein-Gordon solution at T={T} (Nx={Nx}, dt={dt})")
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig("solution_molk_014.png", dpi=150)
plt.close()
print("Files saved: solution_molk_014.npz, solution_molk_014.png")
```
