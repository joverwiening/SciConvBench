dx = 1.0 / Nx

# Time step
dt0 = 0.9 * dx / c_cfl
Nt = math.ceil(T / dt0)
dt = T / Nt

print(f"Nx={Nx}, dx={dx:.6e}")
print(f"Nt={Nt}, dt={dt:.6e}")
print(f"CFL = c_cfl*dt/dx = {c_cfl*dt/dx:.6f}")

# Initial conditions
u0 = np.sin(4 * np.pi * x)
v0 = np.zeros(Nx + 1)

# Enforce Dirichlet BCs on initial condition
u0[0] = 0.0
u0[Nx] = 0.0

# Function to compute c_pde^2 * u_xx using 2nd-order central differences (interior only)
def compute_uxx(u, dx, c_pde):
    """Returns c_pde**2 * u_xx at interior points, zero at boundaries."""
    rhs = np.zeros_like(u)
    rhs[1:-1] = c_pde**2 * (u[2:] - 2.0*u[1:-1] + u[:-2]) / dx**2

The leapfrog update is `u^{n+1} = 2*u^n - u^{n-1} + dt**2 * c_pde**2 * u_xx^n`.

Dirichlet BCs (`u=0`) are enforced at indices `0` and `800` at every step.
