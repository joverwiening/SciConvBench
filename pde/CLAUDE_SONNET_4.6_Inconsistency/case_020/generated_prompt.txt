Solve the 2D variable-coefficient diffusion PDE numerically and save the results as specified.

The PDE is u_t = div(D(x,y) grad(u)) where D(x,y) = 0.05*(1 + 0.5*sin(2*pi*x)*sin(2*pi*y)).
The spatial domain is (x,y) in [0,1]x[0,1] and the time domain is t in [0,0.1].
Use Nx=Ny=80 uniform intervals in each spatial direction, giving 81 grid points per axis including both endpoints.
Set dx = 1/80 and dy = 1/80.
Set x_i = i*dx for i=0..80 and y_j = j*dy for j=0..80.
Use dt=0.0005 and take 200 time steps to reach T=0.1. Wait — 0.1/0.0005 = 200 steps, not 240. Use 200 steps of dt=0.0005 to reach T=0.1 exactly.
Apply Dirichlet boundary conditions u=0 on all four edges (i=0, i=80, j=0, j=80) for all time.
Set the initial condition u(x_i, y_j, 0) = sin(pi*x_i)*sin(pi*y_j) at all grid points.
For interior nodes i=1..79, j=1..79, discretize the diffusion operator in conservative flux form using arithmetic face averages.
Compute D_{i+1/2,j} = 0.5*(D(x_i,y_j) + D(x_{i+1},y_j)) and similarly for D_{i-1/2,j}, D_{i,j+1/2}, D_{i,j-1/2}.
The discrete Laplacian operator L applied to u at interior node (i,j) equals (1/dx**2)*(D_{i+1/2,j}*(u[i+1,j]-u[i,j]) - D_{i-1/2,j}*(u[i,j]-u[i-1,j])) + (1/dy**2)*(D_{i,j+1/2}*(u[i,j+1]-u[i,j]) - D_{i,j-1/2}*(u[i,j]-u[i,j-1])).
Use Implicit Euler time integration: at each step solve the sparse linear system (I - dt*L) u^{n+1} = u^n.
Enforce Dirichlet u=0 boundary conditions by fixing boundary rows in the linear system (identity row, zero right-hand side).
Build the sparse matrix once before the time loop since L does not change in time, then reuse the factorization or solver each step.
Use scipy.sparse for the matrix and scipy.sparse.linalg.spsolve (or a factored solver) to solve each step.
After 200 steps, the solution u at T=0.1 is stored as a 2D array of shape (81,81) with indexing u[j,i] = u(x_i, y_j, T).
Print min(u), max(u), and the discrete L2 norm sqrt(sum(u**2)*dx*dy) at T=0.1.
Save "solution_agent_diffusion_04.npz" containing: x as a 1D array of length 81, y as a 1D array of length 81, u as a 2D array of shape (81,81), and scalar metadata T=0.1, dx=1/80, dy=1/80, dt=0.0005.
Save a heatmap or filled-contour plot of u(x,y,T=0.1) as "solution_agent_diffusion_04.png" with labeled axes and a colorbar.

```python
import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# ── Grid ──────────────────────────────────────────────────────────────────────
Nx = Ny = 80
dx = 1.0 / Nx
dy = 1.0 / Ny
x = np.linspace(0.0, 1.0, Nx + 1)   # 81 points
y = np.linspace(0.0, 1.0, Ny + 1)   # 81 points
XX, YY = np.meshgrid(x, y)           # shape (81,81), YY[j,i]=y_j, XX[j,i]=x_i

dt = 0.0005
T  = 0.1
nsteps = int(round(T / dt))          # 200

# ── Diffusion coefficient on the full grid ────────────────────────────────────
D_grid = 0.05 * (1.0 + 0.5 * np.sin(2*np.pi*XX) * np.sin(2*np.pi*YY))
# D_grid[j,i] = D(x_i, y_j)

# ── Arithmetic face averages ──────────────────────────────────────────────────
# x-faces: D_{i+1/2, j}  shape (Ny+1, Nx)  — for i=0..Nx-1
Dxf = 0.5 * (D_grid[:, :-1] + D_grid[:, 1:])   # [j, i] = D_{i+1/2,j}
# y-faces: D_{i, j+1/2}  shape (Ny, Nx+1)  — for j=0..Ny-1
Dyf = 0.5 * (D_grid[:-1, :] + D_grid[1:, :])   # [j, i] = D_{i,j+1/2}

# ── Build sparse operator L (interior nodes only, but full (Ny+1)*(Nx+1) DOFs)
# DOF index: k = j*(Nx+1) + i
N = (Nx + 1) * (Ny + 1)   # 81*81 = 6561

rows, cols, vals = [], [], []

def add(r, c, v):
    rows.append(r); cols.append(c); vals.append(v)

for j in range(Ny + 1):
    for i in range(Nx + 1):
        k = j * (Nx + 1) + i
        # boundary → identity row (handled separately)
        if i == 0 or i == Nx or j == 0 or j == Ny:
            add(k, k, 1.0)
            continue
        # interior node (i=1..Nx-1, j=1..Ny-1)
        # x-direction contributions
        Dip = Dxf[j, i]       # D_{i+1/2, j}
        Dim = Dxf[j, i-1]     # D_{i-1/2, j}
        # y-direction contributions
        Djp = Dyf[j, i]       # D_{i, j+1/2}
        Djm = Dyf[j-1, i]     # D_{i, j-1/2}

        cx = 1.0 / dx**2
        cy = 1.0 / dy**2

        diag = -(Dip + Dim) * cx - (Djp + Djm) * cy
        add(k, k,                    diag)
        add(k, k + 1,                Dip * cx)          # i+1
        add(k, k - 1,                Dim * cx)          # i-1
        add(k, k + (Nx + 1),         Djp * cy)          # j+1
        add(k, k - (Nx + 1),         Djm * cy)          # j-1

L = sp.csr_matrix((vals, (rows, cols)), shape=(N, N))

# ── Build implicit-Euler system matrix A = I - dt*L ───────────────────────────
I_mat = sp.eye(N, format='csr')
A = I_mat - dt * L

# Factorise once (LU)
solver = spla.factorized(A.tocsc())

# ── Initial condition ─────────────────────────────────────────────────────────
u2d = np.sin(np.pi * XX) * np.sin(np.pi * YY)   # shape (81,81)
# Enforce BCs (already 0 from sin, but be explicit)
u2d[0, :]  = 0.0
u2d[-1, :] = 0.0
u2d[:, 0]  = 0.0
u2d[:, -1] = 0.0

u_vec = u2d.ravel()   # row-major: index k = j*(Nx+1)+i

# ── Time integration ──────────────────────────────────────────────────────────
for step in range(nsteps):
    rhs = u_vec.copy()
    # boundary RHS stays 0 (Dirichlet)
    # interior RHS = u^n  (already set by copy; boundary entries are 0)
    # Force boundary entries of rhs to 0
    # (they are already 0 if BCs are maintained, but be safe)
    u_vec = solver(rhs)
    # Re-enforce Dirichlet (should be machine-zero already)
    u_vec = u_vec.reshape(Ny + 1, Nx + 1)
    u_vec[0, :]  = 0.0
    u_vec[-1, :] = 0.0
    u_vec[:, 0]  = 0.0
    u_vec[:, -1] = 0.0
    u_vec = u_vec.ravel()

u_final = u_vec.reshape(Ny + 1, Nx + 1)   # u_final[j,i]

# ── Diagnostics ───────────────────────────────────────────────────────────────
u_min  = u_final.min()
u_max  = u_final.max()
l2norm = np.sqrt(np.sum(u_final**2) * dx * dy)
print(f"min(u)  = {u_min:.6e}")
print(f"max(u)  = {u_max:.6e}")
print(f"L2 norm = {l2norm:.6e}")

# ── Save .npz ─────────────────────────────────────────────────────────────────
np.savez("solution_agent_diffusion_04.npz",
         x=x, y=y, u=u_final,
         T=np.float64(T),
         dx=np.float64(dx),
         dy=np.float64(dy),
         dt=np.float64(dt))
print("Saved solution_agent_diffusion_04.npz")

# ── Save plot ─────────────────────────────────────────────────────────────────
fig, ax = plt.subplots(figsize=(6, 5))
cf = ax.contourf(x, y, u_final, levels=50, cmap='viridis')
fig.colorbar(cf, ax=ax, label='u(x,y,T)')
ax.set_xlabel('x')
ax.set_ylabel('y')
ax.set_title(f'u(x,y,T={T})  —  variable-coeff diffusion')
fig.tight_layout()
fig.savefig("solution_agent_diffusion_04.png", dpi=150)
plt.close(fig)
print("Saved solution_agent_diffusion_04.png")
```
