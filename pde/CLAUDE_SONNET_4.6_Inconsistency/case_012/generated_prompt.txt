Solve the Allen-Cahn PDE numerically as described below.

The PDE is u_t = (0.015)**2 * (u_xx + u_yy) + u - u**3.
The spatial domain is (x,y) in [0,1] x [0,1] and time domain is t in [0, 0.2].
The parameter epsilon = 0.015.
Boundary conditions are zero Neumann on all four boundaries, enforced via ghost nodes (mirror values).
The initial condition is u(x,y,0) = sin(pi*x) * sin(pi*y).

Use a uniform grid with Nx = Ny = 200 intervals, giving 201 x 201 grid points including both endpoints.
dx = 1/200, dy = 1/200.
Grid points are x_i = i*dx for i = 0..200 and y_j = j*dy for j = 0..200.

Use a semi-implicit time integration scheme: (u^{n+1} - u^n) / dt = epsilon**2 * Delta(u^{n+1}) + u^n - (u^n)**3.
Rearrange to: u^{n+1} - dt * epsilon**2 * Delta(u^{n+1}) = u^n + dt * (u^n - (u^n)**3).
Discretize the Laplacian with the standard 5-point stencil.
Enforce zero Neumann boundary conditions via ghost nodes: mirror the interior value across each boundary edge before applying the stencil.
Use dt = 0.0002 and run for 1000 time steps to reach T = 0.2.
Assemble and solve a sparse linear system at each time step.

At T = 0.2, compute u(x,y,T) on the full 201x201 grid.
Save a file named "solution_agent_allen_cahn_06.npz" containing: x as a 1D array of length 201, y as a 1D array of length 201, u as a 2D array of shape (201, 201) where u[j,i] = u(x_i, y_j), and scalar metadata T=0.2, dx=1/200, dy=1/200, dt=0.0002.
Save a heatmap or contour plot of u(x,y,T) as "solution_agent_allen_cahn_06.png".
Print min(u), max(u), and the discrete L2 norm sqrt(sum(u**2) * dx * dy) at T.

```python
import numpy as np
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import matplotlib.pyplot as plt
import matplotlib
matplotlib.use('Agg')

# Parameters
epsilon = 0.015
eps2 = epsilon**2
Nx = Ny = 200
Lx = Ly = 1.0
dx = Lx / Nx
dy = Ly / Ny
dt = 0.0002
Nt = 1000
T = 0.2

# Grid points (201 x 201 including endpoints)
x = np.linspace(0, Lx, Nx+1)
y = np.linspace(0, Ly, Ny+1)
Npx = Nx + 1  # 201
Npy = Ny + 1  # 201
N = Npx * Npy  # total unknowns

# Initial condition
X, Y = np.meshgrid(x, y)  # shape (Npy, Npx)
u = np.sin(np.pi * X) * np.sin(np.pi * Y)  # shape (Npy, Npx)

# Build the Laplacian matrix with zero Neumann BCs via ghost nodes.
# For interior points: standard 5-point stencil with 1/dx^2 and 1/dy^2.
# For boundary points: ghost node mirror means the derivative term doubles
# the interior neighbor contribution (ghost = neighbor value).
#
# Index mapping: node (j, i) -> j * Npx + i
# u[j,i] is at (x_i, y_j)

# We build the sparse Laplacian L such that (L @ u_flat) approximates
# (u_xx + u_yy) at each node with zero Neumann BCs.

row_idx = []
col_idx = []
data_vals = []

def idx(j, i):
    return j * Npx + i

for j in range(Npy):
    for i in range(Npx):
        center = idx(j, i)
        
        # x-direction (second derivative w.r.t. x)
        # Left neighbor
        if i == 0:
            # ghost at i=-1 mirrors i=1 (zero Neumann)
            # u_xx ~ (u[j,1] - 2*u[j,0] + u[j,1]) / dx^2 = 2*(u[j,1]-u[j,0])/dx^2
            coeff_left = 0.0
            coeff_right = 2.0 / dx**2
            coeff_center_x = -2.0 / dx**2
        elif i == Npx - 1:
            # ghost at i=Npx mirrors i=Npx-2
            coeff_left = 2.0 / dx**2
            coeff_right = 0.0
            coeff_center_x = -2.0 / dx**2
        else:
            coeff_left = 1.0 / dx**2
            coeff_right = 1.0 / dx**2
            coeff_center_x = -2.0 / dx**2
        
        # y-direction (second derivative w.r.t. y)
        if j == 0:
            coeff_down = 0.0
            coeff_up = 2.0 / dy**2
            coeff_center_y = -2.0 / dy**2
        elif j == Npy - 1:
            coeff_down = 2.0 / dy**2
            coeff_up = 0.0
            coeff_center_y = -2.0 / dy**2
        else:
            coeff_down = 1.0 / dy**2
            coeff_up = 1.0 / dy**2
            coeff_center_y = -2.0 / dy**2
        
        coeff_center = coeff_center_x + coeff_center_y
        
        # Add center
        row_idx.append(center)
        col_idx.append(center)
        data_vals.append(coeff_center)
        
        # Left neighbor (i-1)
        if i > 0:
            row_idx.append(center)
            col_idx.append(idx(j, i-1))
            data_vals.append(coeff_left)
        
        # Right neighbor (i+1)
        if i < Npx - 1:
            row_idx.append(center)
            col_idx.append(idx(j, i+1))
            data_vals.append(coeff_right)
        
        # Down neighbor (j-1)
        if j > 0:
            row_idx.append(center)
            col_idx.append(idx(j-1, i))
            data_vals.append(coeff_down)
        
        # Up neighbor (j+1)
        if j < Npy - 1:
            row_idx.append(center)
            col_idx.append(idx(j+1, i))
            data_vals.append(coeff_up)

L = sp.csr_matrix((data_vals, (row_idx, col_idx)), shape=(N, N))

# Build the system matrix A = I - dt * eps2 * L
# This is constant across all time steps, so factor it once.
I_mat = sp.eye(N, format='csr')
A = I_mat - dt * eps2 * L

# Use SuperLU factorization for efficiency
print("Factorizing system matrix...")
A_factored = spla.splu(A.tocsc())
print("Factorization complete.")

# Time integration
u_flat = u.flatten()  # shape (N,)

print(f"Starting time integration: {Nt} steps, dt={dt}, T={T}")
for step in range(Nt):
    # Nonlinear explicit part: u^n - (u^n)^3
    nonlinear = u_flat - u_flat**3
    # RHS: u^n + dt * (u^n - (u^n)^3)
    rhs = u_flat + dt * nonlinear
    # Solve: A * u^{n+1} = rhs
    u_flat = A_factored.solve(rhs)
    
    if (step + 1) % 200 == 0:
        print(f"  Step {step+1}/{Nt}, t={( step+1)*dt:.4f}, "
              f"min={u_flat.min():.6f}, max={u_flat.max():.6f}")

# Reshape solution
u_final = u_flat.reshape(Npy, Npx)  # shape (201, 201), u_final[j,i] = u(x_i, y_j)

# Diagnostics
u_min = u_final.min()
u_max = u_final.max()
l2_norm = np.sqrt(np.sum(u_final**2) * dx * dy)
print(f"\nAt T={T}:")
print(f"  min(u) = {u_min:.8f}")
print(f"  max(u) = {u_max:.8f}")
print(f"  L2 norm = {l2_norm:.8f}")

# Save NPZ
np.savez("solution_agent_allen_cahn_06.npz",
         x=x, y=y, u=u_final,
         T=np.float64(T),
         dx=np.float64(dx),
         dy=np.float64(dy),
         dt=np.float64(dt))
print("Saved solution_agent_allen_cahn_06.npz")

# Save plot
fig, ax = plt.subplots(figsize=(7, 6))
cp = ax.contourf(X, Y, u_final, levels=50, cmap='RdBu_r')
plt.colorbar(cp, ax=ax, label='u(x,y,T)')
ax.set_xlabel('x')
ax.set_ylabel('y')
ax.set_title(f'Allen-Cahn solution u(x,y,T={T})')
plt.tight_layout()
plt.savefig("solution_agent_allen_cahn_06.png", dpi=150)
plt.close()
print("Saved solution_agent_allen_cahn_06.png")
```
