import numpy as np
import matplotlib.pyplot as plt

# Parameters
c = 1.0
m = 3.0
T = 0.8
Nx = 512
dx = 1.0 / Nx
CFL_target = 0.9

# Time stepping
dt0 = CFL_target * dx / c
Nt = int(np.ceil(T / dt0))
dt = T / Nt

print(f"Nx={Nx}, dx={dx:.6e}, Nt={Nt}, dt={dt:.6e}, CFL={c*dt/dx:.6f}")

# Grid
x = np.arange(Nx) * dx  # x_i = i*dx, i=0..Nx-1

# Initial conditions
u0 = np.cos(2 * np.pi * x)
ut0 = np.zeros(Nx)

# 2nd-order central difference for u_xx with periodic wrap-around
def laplacian_periodic(u, dx):
    return (np.roll(u, -1) - 2.0 * u + np.roll(u, 1)) / dx**2

# Taylor start for first step
uxx0 = laplacian_periodic(u0, dx)
rhs0 = uxx0 - m**2 * u0  # u_tt = c^2*u_xx - m^2*u, c=1
u_prev = u0.copy()
u_curr = u0 + dt * ut0 + 0.5 * dt**2 * rhs0

# Leapfrog time integration
for n in range(1, Nt):
    uxx = laplacian_periodic(u_curr, dx)
    rhs = c**2 * uxx - m**2 * u_curr
    u_next = 2.0 * u_curr - u_prev + dt**2 * rhs
    u_prev = u_curr
    u_curr = u_next

u_numerical = u_curr

# Exact solution at T
omega = np.sqrt((c * 2 * np.pi)**2 + m**2)
u_exact = np.cos(2 * np.pi * x) * np.cos(omega * T)

# Diagnostics
u_min = np.min(u_numerical)
u_max = np.max(u_numerical)
l2_norm = np.sqrt(np.sum(u_numerical**2) * dx)
max_abs_error = np.max(np.abs(u_numerical - u_exact))
l2_error = np.sqrt(np.sum((u_numerical - u_exact)**2) * dx)

print(f"min(u)       = {u_min:.8e}")
print(f"max(u)       = {u_max:.8e}")
print(f"L2 norm      = {l2_norm:.8e}")
print(f"Max abs error= {max_abs_error:.8e}")
print(f"L2 error     = {l2_error:.8e}")

# Save npz
np.savez("solution_agent_kg_04.npz", x=x, u=u_numerical, T=T, dx=dx, dt=dt)
print("Saved solution_agent_kg_04.npz")

# Save plot
fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(x, u_numerical, label=f"Numerical u(x, T={T})")
ax.plot(x, u_exact, '--', label="Exact solution", alpha=0.7)
ax.set_xlabel("x")
ax.set_ylabel("u")
ax.set_title(f"Klein-Gordon solution at T={T}")
ax.legend()
plt.tight_layout()
plt.savefig("solution_agent_kg_04.png", dpi=150)
plt.close()
